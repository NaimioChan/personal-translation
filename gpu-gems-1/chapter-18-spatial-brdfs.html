第 18 章　空间 BRDF

David McAllister
NVIDIA

空间双向反射分布函数（SBRDF）描述表面上每个不同点、在每一对入射角与出射角下的反射率。该函数以非常通用、细致的方式表现表面的外观。本章讨论一种紧凑的 SBRDF 表示，以及用离散光源或环境贴图照明 SBRDF 的渲染方法。

18.1 什么是 SBRDF？

SBRDF 是纹理映射与双向反射分布函数（BRDF）的结合。纹理贴图存储的是在二维表面上随空间变化的反射率或其他属性。BRDF 存储的是表面上单个点在所有入射角与出射角下的反射率。见图 18-1。

真实表面通常变化显著，既有表面上的空间变化，也有随所有光照与视线方向的角度变化。因此，要更真实地表现大多数表面，就应当把 BRDF 与纹理映射结合起来。这种结合就是 SBRDF。详见 McAllister 2002。

18.2 表示的细节

最直接的 SBRDF 表示，是在纹理贴图的每个纹素处存储该点 BRDF 的全部参数。任何 BRDF 表示都可以用，但紧凑且适合硬件渲染的一种是 Lafortune 表示（Lafortune 1997）。该表示由若干项之和构成：

Lafortune 表示在局部表面坐标中求值。x 轴与 y 轴沿各向异性方向（划痕或线纹方向），z 轴为法线。把矩阵 C 定义为 Cx = -1、Cy = -1、Cz = 1，会使 i 关于法线反射，得到标准的 Phong 波瓣。但 C 的系数可以取其他值，以剪切镜面波瓣，从而表现真实表面的散射行为。波瓣的峰值位于 C · i 方向上。对于各向同性 BRDF，Cx = Cy。对于偏离镜面方向的反射， ，把波瓣拉向切平面。对于回射，Cx > 0 且 Cy > 0。当 Cx 与 Cy 符号相反时，波瓣对平行于主各向异性方向射来的光做前向散射，对垂直于该方向射来的光做后向散射，这正好模拟线纹与划痕的行为。

Lafortune 表示能逐个瞄准并缩放散射波瓣，这种灵活性正是仅用少数几个波瓣就能近似 BRDF 的关键。这一性质也支撑了 18.4 节所述的光泽环境贴图技术。Lafortune 表示很贴合 BRDF 通常具有的形状。它紧凑，并且能表现有意思的 BRDF 特性，例如菲涅尔反射随角度增大的反射率、偏镜面峰值以及回射。

反照率 值是 RGB 三元组，而 Cx 、 Cy 、 Cz 与 n 的值由各通道共用，每个镜面波瓣共七个系数。视表面性质以及质量与速度之间的权衡，通常一到三个镜面项就足够。系数少至十个（三个漫反射加七个镜面反射）时，SBRDF 表示仅比 RGB 纹理贴图大约三倍，却真实得多。

SBRDF 数据可以从真实表面测量得到（McAllister 2002），也可以由美术用专门编写的绘制程序绘制。最简单的绘制方法是使用一个现有 BRDF 的调色板，把它们绘制进 SBRDF 纹理贴图。调色板条目可以混合现有 BRDF，可以从 SBRDF 数据库采样（McAllister 2003），也可以用对话框设置 、Cx 、 Cy 、 Cz 与 n 的值来定义。

18.3 使用离散光源渲染

两种渲染方式载入 SBRDF 到纹理内存的方式相同。漫反射颜色 d 存在一张纹理贴图中（其 alpha 通道可用于 alpha kill 或其他用途）。对每个波瓣，s 存在一张纹理贴图中（其 alpha 通道不使用，或用于各向异性方向），而 Cx 、 Cy 、 Cz 与 n 存在第二张贴图中。使用八位纹理贴图时，必须施加缩放与偏置，因为 Cx 与 Cy 的有用取值范围约为 -1.1 到 1.1。先乘 96，再加 128 的偏置，效果不错。视可用的内存带宽与片元着色器性能，把 SBRDF 数据存进浮点纹理而非八位纹理可能更高效，这样可避免缩放并提高精度。

用标准点光源或方向光照明 SBRDF 表面的 Cg 着色器很简单，如代码清单 18-1 所示。它类似于标准 Phong 模型与硬件光源的着色器。区别之一在于，所有 BRDF 参数都从纹理贴图采样，而非只有漫反射颜色。另一个区别是点积在局部表面坐标中求值，因此其分量可以用 Cx 、 Cy 与 Cz 缩放。这要求顶点程序以局部表面坐标提供光照向量。示例结果见图 18-3。

示例 18-1. 用于离散光源的 SBRDF 片元着色器

#define NUM_LOBES 3 #define NUM_LIGHTS 2
// Store Cx, Cy, Cz on range -1.3333 .. 1.3333.
#define SCL
(1.0 / (96.0 / 256.0)) #define BIAS(SCL / 2.0) #define EXSCL
    255.0 // 将指数缩放到 0 .. 255 范围。
    // 对视线向量和所有光源向量做光栅化。
    // 把光源颜色作为 uniform 参数传入。
    struct fromrast
{
  float2 TexUV : TEXCOORD0;
  // 表面纹理坐标
  float3 EyeVec : TEXCOORD1;
  // 指向眼睛的向量（局部空间）
  float3 LightVec[NUM_LIGHTS] : TEXCOORD2; // 光源（局部空间）
};
float4 main(fromrast I, uniform sampler2D tex_dif,
            // 漫反射
            uniform sampler2D tex_lshp[NUM_LOBES], // 波瓣形状
            uniform sampler2D tex_lalb[NUM_LOBES], // 波瓣反照率
            uniform float4 Expos, uniform float3 LightCol[NUM_LIGHTS])
    : COLOR
{
  // 从纹理中加载 BRDF 参数
  float4 lobe_shape[NUM_LOBES];
  float4 lobe_albedo[NUM_LOBES];
  for (float p = 0; p < NUM_LOBES; p++)
  {
    lobe_shape[p] =
        f4tex2D(tex_lshp[p], I.TexUV.xy) * float4(SCL, SCL, SCL, EXSCL) -
        float4(BIAS, BIAS, BIAS, 0.0);
    lobe_albedo[p] = f4tex2D(tex_lalb[p], I.TexUV.xy);
  }
  float4 dif_albedo = f4tex2D(tex_dif, I.TexUV.xy);
  // 局部空间中指向眼睛的向量。
  float3 toeye = normalize(I.EyeVec.xyz);
  // 累加各光源在该表面上的出射辐亮度
  float3 exrad = float3(0, 0, 0);
  for (int l = 0; l < NUM_LIGHTS; l++)
  {
    // 局部空间中指向光源的向量。
    float3 tolight = normalize(I.LightVec[l].xyz);
    // 对该点与方向对求值 SBRDF
    float3 refl = dif_albedo.xyz;
    for (float p = 0; p < NUM_LOBES; p++)
    {
      // 错切视线向量
      float3 Cwr_local = lobe_shape[p].xyz * toeye;
      float thedot = dot(Cwr_local, tolight);
      refl = refl + lit(thedot, thedot, lobe_shape[p].w).z * lobe_albedo[p].xyz;
    }
    // 该光源的辐照度（入射辐亮度乘以 NdotL）。
    float NdotL = max(0, tolight.z);
    float3 irrad = LightCol[l] * NdotL;
    // 反射率乘以辐照度即为出射辐亮度。
    exrad += refl * irrad;
  }
  float4 final_col = exrad.xyzz * Expos.xxxx; // 设置 HDR 曝光。
  final_col.w = dif_albedo.w; // 把贴图中的 alpha 写入像素。
  return final_col;
}

18.3.1 纹理采样

Cx 、 Cy 、 Cz 与 n 是一个非线性函数的参数，因此对这些纹理贴图做双线性或三线性插值会产生闪烁的瑕疵。最简单的解决办法是对这张贴图使用点采样。

18.4 使用环境贴图渲染

除了用点光源或方向光照明，SBRDF 还可以通过环境贴图接受来自各个方向的入射光。关键在于渲染前先把环境贴图与 BRDF 的一部分做卷积。对大多数 BRDF 表示，这必须针对每个不同的 BRDF 分别进行。但一个 SBRDF 可能包含上百万个不同的 BRDF，这样做是不可能的。

我们改为只把环境贴图与一组挑选了不同镜面指数的 Phong 波瓣做卷积，例如 n = 0、1、4、16、64 和 256. 这些贴图可以存进立方体 mipmap 的各个层，n = 0 的贴图放在最粗的 mipmap 层，n = 256 的贴图放在最细的 mipmap 层。SBRDF 纹素的 n 值随即指示采样立方体贴图相应 mipmap 层时所用的细节层次（LOD）值。

这组卷积后的环境贴图的另一种表示是单张三维纹理，其中 s 与 t 两个映射维度对应抛物面映射（Heidrich and Seidel 1999），r 维度对应 n。

示例 18-2. 将环境贴图与 Phong 波瓣卷积的伪代码

float3 S(float3 Wp, float n)
{
  float3 Sum = 0.xyz;
  float WgtSum = 0;
  for (int si = 0; si < 6; si++)
  {
    for (int yi = 0; yi < YMAX; yi++)
    {
      for (int xi = 0; xi < XMAX; xi++)
      {
        float3 Wi = VecFromCubeMap(xi, yi, si);
        float dp = dot(Wp, Wi);
        if (dp < 0)
          continue;
        float lobe_shape = pow(dp, n);
        float3 FN = FaceNormal(si);
        // 用立方体向量的长度缩放辐照度，
        // 以补偿立方体贴图分布的不规则。
        float3 Irrad = Li(Wi) * dp / length2(Wi);
        Sum += Irrad * lobe_shape;
        WgtSum += lobe_shape; // 计算波瓣的体积
      }
    }
  }
  return Sum / WgtSum;
}
// 用指数为 n 的 Phong 波瓣卷积立方体贴图 Li
CubeMap PrecomputeEnvMap(float n)
{
  CubeMap Smap;
  // 遍历立方体贴图的所有像素
  for (int s = 0; s < 6; s++)
  {
    for (int y = 0; y < YMAX; y++)
    {
      for (int x = 0; x < XMAX; x++)
      {
        float3 Wp = VecFromCubeMap(x, y, s);
        Smap(x, y, s) = S(Wp, n);
      }
    }
  }
  return Smap;
}

18.4.2 着色器代码

代码清单 18-3 给出用预卷积环境贴图渲染的示例代码。它调用函数 f3texCUBE_RGBE_Conv( env_tex, dir, n )，该函数在给定方向上采样预卷积环境贴图，并根据 n 计算 LOD。结果见图 18-5。

为了紧凑地表示大于 1.0 的入射辐亮度值，可以使用 RGBE 表示，解码为 float3 也在该函数内完成。

示例 18-3. 用于环境贴图的 SBRDF 片元着色器

#define NUM_LOBES 2
// Store Cx, Cy, Cz on range -1.3333 .. 1.3333.
#define SCL
(1.0 / (96.0 / 256.0)) #define BIAS(SCL / 2.0) #define EXSCL
    255.0 // 将指数缩放到 0 .. 255 范围。
    float3
    ApplyLobe(float3 toeye, float4 lobe_shape, float3 lobe_albedo, float3 Tan,
              float3 BiN, float3 Nrm, uniform samplerCUBE tex_s)
{
  // 在局部表面空间中反射视线向量得到 p(wr)
  float3 Cwr_local = lobe_shape.xyz * toeye;
  // 环境查找之前先把波瓣峰值变换到世界空间
  float3 Cwr_world = ToWorld(Cwr_local, Tan, BiN, Nrm);
  // 在波瓣峰值方向上采样立方体贴图。
  float sharpness = lobe_shape.w; // 这就是 n。
  float3 incrad = f3texCUBE_RGBE_Conv(tex_s, Cwr_world, sharpness);
  // (length^2)^(n*0.5) = length^n
  float lobelen = pow(dot(Cwr_world, Cwr_world), sharpness * 0.5);
  // 用峰值方向的衰减
  // 来近似所有点处的辐照度衰减。
  // 这就是局部空间中的 N dot Cwr。
  float3 irrad = incrad * (Cwr_local.z * lobelen);
  float3 radiance = irrad * lobe_albedo;
  return radiance;
}
struct fromrast
{
  float2 TexUV : TEXCOORD0;
  // 表面纹理坐标
  float3 EyeVec : TEXCOORD1; // 指向眼睛的向量（局部空间）
  // ——需要归一化
  float3 NrmVec : TEXCOORD2; // 法线（世界空间）
  // ——需要归一化
  float3 TanVec : TEXCOORD3; // 切线（世界空间）
  // ——需要归一化
};
float4 main(fromrast I, uniform sampler2D tex_dif,
            // 漫反射
            uniform sampler2D tex_lshp0,
            // 波瓣形状
            uniform sampler2D tex_lalb0,
            // 波瓣反照率
            uniform sampler2D tex_lshp1,
            // 波瓣形状
            uniform sampler2D tex_lalb1,
            // 波瓣反照率
            uniform samplerCUBE tex_envd, // 立方体漫反射环境贴图
            uniform samplerCUBE tex_envs, // 立方体镜面反射环境贴图
            uniform float4 Expos)
    : COLOR
{
  // 预加载所有 BRDF 参数。
  float4 lobe_shape[NUM_LOBES];
  float4 lobe_albedo[NUM_LOBES];
  for (float p = 0; p < NUM_LOBES; p++)
  {
    lobe_shape[p] =
        f4tex2D(tex_lshp[p], I.TexUV.xy) * float4(SCL, SCL, SCL, EXSCL) -
        float4(BIAS, BIAS, BIAS, 0.0);
    lobe_albedo[p] = f4tex2D(tex_lalb[p], I.TexUV.xy);
  }
  float4 dif_albedo = f4tex2D(tex_dif, I.TexUV.xy);
  float3 Nrm = normalize(I.NrmVec.xyz);
  float3 Tan = normalize(I.TanVec.xyz);
  float3 BiN = cross(Nrm, Tan);
  // 局部空间中指向眼睛的向量。
  float3 toeye = normalize(I.EyeVec.xyz);
  // 累加各波瓣在该表面上的出射辐亮度
  float3 exrad = float3(0, 0, 0);
  for (int l = 0; l < NUM_LOBES; l++)
    exrad += ApplyLobe(toeye, lobe_shape[l], lobe_albedo[l].xyz, Tan, BiN, Nrm,
                       tex_envs);
  // 加上漫反射项。tex_envd 保存了表面
  // 法线方向为 Nrm 时的辐照度。
  float3 irrad = f3texCUBE_RGBE(tex_envd, Nrm);
  exrad = exrad + dif_albedo.xyz * irrad;
  float4 final_col = exrad.xyzz * Expos.xxxx; // 设置 HDR 曝光。
  final_col.w = dif_albedo.w; // 把贴图中的 alpha 写入像素。
  return final_col;
}

18.5 结论

考虑到 SBRDF 的通用性之强，离散光源着色器的性能好得惊人。它的主要工作是对每个波瓣、每个光源做一次点积和一次 lit() 计算。

同样，环境贴图着色器的性能也相当好，因为它允许每个纹素拥有完全不同的 BRDF，却仍能用环境贴图照明。这个着色器的核心只是一次环境贴图纹理查找，但遗憾的是，计算查找坐标需要一次矩阵变换。两个着色器都有大量运算可以轻松地以半精度完成，而看不出视觉差异。

为简化起见，本章省略了随空间变化的各向异性方向。不过细节可以在 McAllister 2002 中找到，其中还有卷积环境贴图的示例代码、与这些片元着色器对应的顶点着色器，以及更多内容。
