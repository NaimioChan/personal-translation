Arch 模板

下面是一个 PKGBUILD 文件的模板。

它是名为 makepkg 的程序的基础。你可以把这个文件放在文件结构里任何你有写权限的地方。然后就在那个文件夹里运行 makepkg。

这个程序会读取 PKGBUILD 文件，下载源代码，进入一个安全的 chroot，编译，安装到名为 pkg 的子目录，然后用这些文件打包。这个包可以用 pacman -U pd-template-version.tar.xz 安装。

如果你把文件分享到 Arch User Repository（AUR），世界上每个 Arch 用户都能安装你的外部对象，例如用 yaourt。

# 贡献者：你的名字 <your.name at the domain of email dot com> 

pkgname=pd-template
pkgver=version
pkgrel=1
pkgdesc="External for Pure Data. Describe functionality in on sentence."
arch=('i686' 'x86_64')
url="http://your-site.html"
depends=('pd')
license=('GPL2')
groups=('pd-externals')
source=('http://soucre.at/${pkgname}-${pkgver}')

md5sums=('--------------------------------') # 用 makepkg -g >> PKGBUILD 生成

build() {
  cd ${srcdir}/template-version
  make
  make DESTDIR=$startdir/pkg prefix=/usr install
}
