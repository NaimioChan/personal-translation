panpot

declare name      "panpot";
declare version   "1.0";
declare author    "Grame";
declare license   "BSD";
declare copyright "(c)GRAME 2006";

//-------------------------------------------------
// 立体声声像控制
//-------------------------------------------------

panpot(x) = sqrt(c)*x, sqrt(1-c)*x
            with {
                c = (nentry("[1]pan[style:knob]",0,-90,90,1)-90.0)/-180.0;
            };

process = panpot;

spat

declare name      "spat";
declare version   "1.0";
declare author    "Grame";
declare license   "BSD";
declare copyright "(c)GRAME 2006";

//==========================================================
//
// GMEM SPAT
// L. Pottier 空间化器的实现
//
//==========================================================

import("stdfaust.lib");

//------------------------------------------------------
// 示例：单声道输入空间化到 8 路输出
//------------------------------------------------------

angle = hslider("angle", 0.0, 0, 1, 0.01);
distance = hslider("distance", 0.5, 0, 1, 0.01);

process = vgroup("Spatializer 1x8", sp.spat(8, angle, distance));

speakersDist = 0.0783;  

// 参考听音点（WFS 的中心位置）
xref = 0;
yref = 1;
zref = 0;

// 将 2 个声源空间化到 8 只扬声器
process = sp.wfs_ui(xref,yref,zref,speakersDist,2,8);
