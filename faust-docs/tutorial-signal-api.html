使用 signal API

signal API 在Faust 编译链内部提供了一个中间接入点。本教程通过代码示例介绍它。目标是展示如何在 signal API 之上构建新的音频 DSP 语言（文本式或图形式），并利用 Faust 编译器基础设施的一部分。

Faust 编译器结构

Faust 编译器由若干步骤组成：

编译链

从 DSP 源代码出发，语义阶段产生信号，信号在概念上是样本或控制值的无限流。随后在代码生成阶段把这些信号编译为命令式代码（C/C++、LLVM IR、WebAssembly 等）。

语义阶段本身由若干步骤组成：

语义阶段

使用框图代数（BDA）编写的初始 DSP 代码，在求值、lambda 演算步骤中被翻译为范式的扁平电路。

输出信号列表由符号传播步骤产生。每个输出信号随后被简化，并执行一组优化（范式计算与化简、延迟线共享、类型推导等），最终产生范式形式的输出信号列表。

代码生成阶段把信号翻译为名为 FIR（Faust Imperative Representation）的中间表示，再通过一组后端转换为最终目标语言（C/C++、LLVM IR、WebAssembly 等）。

访问 signal 阶段

在语义阶段中新增了一个公开的中间入口点，可以创建信号集（作为输出信号列表），从而利用编译链剩余的所有部分。signal API（或 C signal API 版本）让开发者能够以编程方式创建输出信号列表，然后编译它，生成可直接使用的 DSP，形式可以是 C++ 类，也可以是 LLVM、Interpreter 或 WebAssembly 工厂，以便与所有现有的架构文件配合使用。信号阶段完成的若干优化，将通过查看生成的 C++ 代码来演示。

注意，box API 可以访问编译链中的另一个阶段。

编译 signal 表达式

使用 signal API 需要执行以下步骤：

  • 使用 createLibContext 函数创建全局编译上下文

  • 使用 signal API 创建信号输出，通过组合更简单的表达式逐步构建更复杂的表达式

  • 使用 createCPPDSPFactoryFromSignals 函数编译输出列表以创建 DSP 工厂（或用 createDSPFactoryFromSignals 生成 LLVM 嵌入工厂，或用 createInterpreterDSPFactoryFromSignals 生成 Interpreter 嵌入工厂）

  • 最后使用 destroyLibContext 函数销毁编译上下文

DSP 工厂允许创建 DSP 实例，以便与音频和 UI 架构文件配合使用，在编译过程本身之外。DSP 实例和工厂最终在不再使用时必须释放。

工具

先定义一个 compile 函数，它使用 createCPPDSPFactoryFromSignals 函数并打印生成的 C++ 类：

static void compile(const string& name, 
                    tvec signals, 
                    int argc = 0, 
                    const char* argv[] = nullptr)
{
    string error_msg;
    dsp_factory_base* factory = createCPPDSPFactoryFromSignals(name, 
                                                              signals, 
                                                              argc, 
                                                              argv, 
                                                              error_msg);
    if (factory) {
        // 打印 C++ 类
        factory->write(&cout);
        delete(factory);
    } else {
        cerr << error_msg;
    }
}

一个封装所有必要步骤的宏：

#define COMPILER(exp)    \
{                        \
    createLibContext();  \
    exp                  \
    destroyLibContext(); \
}                        \

示例

每个示例都给出等价的 Faust DSP 程序和 SVG 图作为辅助。SVG 图展示编译 propagate 步骤的结果（即在任何信号规范化步骤之前），并清楚展示每个输出信号表达式必须如何创建。所有展示的 C++ 示例（以及一些额外示例）都定义在 signal-tester 工具中，可在 tools/benchmark 目录下用 make signal-tester 编译。

简单的常量信号

我们来创建一个生成常量值 0.5 的程序。以下是 Faust DSP 代码：

process = 0.5;

下面的代码创建一个输出信号向量（类型为 tvec），其中包含单个 sigReal(0.5) 信号，然后编译它并显示 C++ 类：

static void test1()
{
    COMPILER
    (
        tvec signals;
        signals.push_back(sigReal(0.5));

        compile("test1", signals);
    )
}

compute 方法则是：

virtual void compute(int count, FAUSTFLOAT** inputs, FAUSTFLOAT** outputs) 
{
    FAUSTFLOAT* output0 = outputs[0];
    for (int i0 = 0; (i0 < count); i0 = (i0 + 1)) {
        output0[i0] = FAUSTFLOAT(0.5f);
    }
}

对输入信号做一些数学运算

下面是一个对信号输入做数学运算的简单程序：

process = _ <: +(0.5), *(1.5);

第一个音频输入用 sigInput(0) 信号创建，然后用 sigAdd 和 sigMul 信号算子变换，产生两个输出：

static void test2()
{
    COMPILER
    (
        tvec signals;
        Signal in1 = sigInput(0);
        signals.push_back(sigAdd(in1, sigReal(0.5)));
        signals.push_back(sigMul(in1, sigReal(1.5)));

        compile("test2", signals);
     )
}

compute 方法则是：

virtual void compute(int count, FAUSTFLOAT** inputs, FAUSTFLOAT** outputs) 
{
    FAUSTFLOAT* input0 = inputs[0];
    FAUSTFLOAT* output0 = outputs[0];
    FAUSTFLOAT* output1 = outputs[1];
    for (int i0 = 0; (i0 < count); i0 = (i0 + 1)) {
        float fTemp0 = float(input0[i0]);
        output0[i0] = FAUSTFLOAT((fTemp0 + 0.5f));
        output1[i0] = FAUSTFLOAT((1.5f * fTemp0));
    }
}

注意，访问输入 N 只需使用 sigInput(N) 表达式。

定义延迟信号

下面是一个使用信号输入并对其做数学运算、然后延迟信号的简单程序：

process = _ <: @(+(0.5), 500), @(*(1.5), 3000);

sigDelay(x, y) 算子用第二个 y 参数延迟第一个参数 x，这里使用常量值：

static void test3()
{
    COMPILER
    (
        tvec signals;
        Signal in1 = sigInput(0);
        signals.push_back(sigDelay(sigAdd(in1, sigReal(0.5)), sigInt(500)));
        signals.push_back(sigDelay(sigMul(in1, sigReal(1.5)), sigInt(3000)));

        compile("test3", signals);
    )
}

compute 方法则是：

{
    FAUSTFLOAT* input0 = inputs[0];
    FAUSTFLOAT* output0 = outputs[0];
    FAUSTFLOAT* output1 = outputs[1];
    for (int i0 = 0; (i0 < count); i0 = (i0 + 1)) {
        float fTemp0 = float(input0[i0]);
        fVec0[(IOTA & 4095)] = fTemp0;
        fVec1[(IOTA & 511)] = (fTemp0 + 0.5f);
        output0[i0] = FAUSTFLOAT(fVec1[((IOTA - 500) & 511)]);
        output1[i0] = FAUSTFLOAT((1.5f * fVec0[((IOTA - 3000) & 4095)]));
        IOTA = (IOTA + 1);
    }
}

Faust 编译器的若干选项可以控制生成的 C++ 代码。默认情况下在单个循环中逐样本计算。但编译器也可以生成向量化和并行代码。下面的代码展示如何以向量模式编译：

static void test5()
{
    createLibContext();

    tvec signals;
    Signal in1 = sigInput(0);
    signals.push_back(sigDelay(sigAdd(in1, sigReal(0.5)), sigInt(500)));
    signals.push_back(sigDelay(sigMul(in1, sigReal(1.5)), sigInt(3000)));

    // 向量化编译
    compile("test5", signals, 3, (const char* []){ "-vec", "-lv", "1" });

    destroyLibContext();
}

compute 方法则是：

virtual void compute(int count, FAUSTFLOAT** inputs, FAUSTFLOAT** outputs) 
{
    FAUSTFLOAT* input0_ptr = inputs[0];
    FAUSTFLOAT* output0_ptr = outputs[0];
    FAUSTFLOAT* output1_ptr = outputs[1];
    for (int vindex = 0; (vindex < count); vindex = (vindex + 32)) {
        FAUSTFLOAT* input0 = &input0_ptr[vindex];
        FAUSTFLOAT* output0 = &output0_ptr[vindex];
        FAUSTFLOAT* output1 = &output1_ptr[vindex];
        int vsize = std::min<int>(32, (count - vindex));
        /* 可向量化循环 0 */
        /* 前置代码 */
        fYec0_idx = ((fYec0_idx + fYec0_idx_save) & 4095);
        /* 计算代码 */
        for (int i = 0; (i < vsize); i = (i + 1)) {
            fYec0[((i + fYec0_idx) & 4095)] = float(input0[i]);
        }
        /* 后置代码 */
        fYec0_idx_save = vsize;
        /* 可向量化循环 1 */
        /* 前置代码 */
        fYec1_idx = ((fYec1_idx + fYec1_idx_save) & 1023);
        /* 计算代码 */
        for (int i = 0; (i < vsize); i = (i + 1)) {
            fYec1[((i + fYec1_idx) & 1023)] = (float(input0[i]) + 0.5f);
        }
        /* 后置代码 */
        fYec1_idx_save = vsize;
        /* 可向量化循环 2 */
        /* 计算代码 */
        for (int i = 0; (i < vsize); i = (i + 1)) {
            output0[i] = FAUSTFLOAT(fYec1[(((i + fYec1_idx) - 500) & 1023)]);
        }
        /* 可向量化循环 3 */
        /* 计算代码 */
        for (int i = 0; (i < vsize); i = (i + 1)) {
            output1[i] = FAUSTFLOAT((1.5f * fYec0[(((i + fYec0_idx) - 3000) & 4095)]));
        }
    }
}

如果 C++ 编译器能自动向量化，可能会更快。

等价的 signal 表达式

非常重要的一点是，语法等价的 signal 表达式会在内部由同一个内存结构表示（使用哈希共享 hash consing），因此在后续编译步骤中以相同方式处理。所以下面的代码中，创建 s1 变量来定义 sigAdd(sigDelay(sigInput(0), sigInt(500)), sigReal(0.5)) 表达式，然后在两个输出中使用：

static void equivalent1()
{
    COMPILER
    (
        tvec signals;
        Signal s1 = sigAdd(sigDelay(sigInput(0), sigInt(500)), sigReal(0.5))
        signals.push_back(s1);
        signals.push_back(s1);

        compile("equivalent1", signals);
    )
}

其行为与下面的代码完全相同，其中 sigAdd(sigDelay(sigInput(0), sigInt(500)), sigReal(0.5)) 表达式被使用了两次：

static void equivalent2()
{
    COMPILER
    (
        tvec signals;
        signals.push_back(sigAdd(sigDelay(sigInput(0), sigInt(500)), sigReal(0.5)));
        signals.push_back(sigAdd(sigDelay(sigInput(0), sigInt(500)), sigReal(0.5)));

        compile("equivalent2", signals);
    )
}

在 signal API 之上创建 DSL 时，这是值得记住的一个特性。

定义递归信号

递归信号可以用 sigRecursion 函数构建递归，用 sigSelf 函数引用递归信号本身。会自动创建一个样本的延迟以产生有效的计算。下面是一个简单例子：

process = + ~ _;

用下面的代码构建：

static void test10()
{
    COMPILER
    (
        tvec signals;
        Signal in1 = sigInput(0);
        signals.push_back(sigRecursion(sigAdd(sigSelf(), in1)));

        compile("test10", signals);
    )
}

compute 方法显示了保存延迟信号的 fRec0 变量：

virtual void compute(int count, FAUSTFLOAT** inputs, FAUSTFLOAT** outputs) 
{
        FAUSTFLOAT* input0 = inputs[0];
        FAUSTFLOAT* output0 = outputs[0];
        for (int i0 = 0; (i0 < count); i0 = (i0 + 1)) {
            fRec0[0] = (float(input0[i0]) + fRec0[1]);
            output0[i0] = FAUSTFLOAT(fRec0[0]);
            fRec0[1] = fRec0[0];
        }
}

等价的代码也可以用更通用的 sigRecursionN 和 sigSelfN 函数定义，它们允许构建一组（可能相互依赖的）递归信号：

static void test10bis()
{
    COMPILER
    (
        tvec signals;
        Signal in1 = sigInput(0);
        signals.push_back(sigRecursionN(sigAdd(sigSelfN(0), in1)));

        compile("test10", signals);
    )
}

下面是相互依赖的递归信号定义示例：

static void test10ter()
{
    COMPILER
    (
        Signal in0 = sigInput(0);
        Signal in1 = sigInput(1);
        tvec ins;
        ins.push_back(sigAdd(sigMul(sigSelfN(1), sigReal(0.5)), in0));
        ins.push_back(sigAdd(sigMul(sigSelfN(0), sigReal(0.9)), in1));
        tvec outs = sigRecursionN(ins);

        compile("test10ter", outs);
    )
}

以及下面的 compute 方法：

virtual void compute(int count, FAUSTFLOAT** RESTRICT inputs, FAUSTFLOAT** RESTRICT outputs) {
    FAUSTFLOAT* input0 = inputs[0];
    FAUSTFLOAT* input1 = inputs[1];
    FAUSTFLOAT* output0 = outputs[0];
    FAUSTFLOAT* output1 = outputs[1];
    for (int i0 = 0; i0 < count; i0 = i0 + 1) {
        fRec0[0] = float(input0[i0]) + 0.5f * fRec1[1];
        fRec1[0] = float(input1[i0]) + 0.9f * fRec0[1];
        output0[i0] = FAUSTFLOAT(fRec0[0]);
        output1[i0] = FAUSTFLOAT(fRec1[0]);
        fRec0[1] = fRec0[0];
        fRec1[1] = fRec1[0];
    }
}

访问全局上下文

在 Faust 中，底层音频引擎的采样率和缓冲区大小通过外部函数与常量机制访问。这些值也可以在 signal 语言中用以下辅助函数使用：

// 在 platform.lib 中复现 'SR' 定义
// SR = min(192000.0, max(1.0, fconstant(int fSampleFreq, )));
inline Signal SR()
{
    return sigMin(sigReal(192000.0), 
           sigMax(sigReal(1.0), 
           sigFConst(SType::kSInt, "fSampleFreq", "<dummy.h>")));
}

// 在 platform.lib 中复现 'BS' 定义
// BS = fvariable(int count, );
inline Signal BS()
{
    return sigFVar(SType::kSInt, "count", "<dummy.h>");
}

所以下面的 DSP 程序：

import("stdfaust.lib"); 
process = ma.SR, ma.BS;

可以在 signal API 层面写成：

static void test11()
{
    COMPILER
    (
        tvec signals;
        signals.push_back(SR());
        signals.push_back(BS());

        compile("test11", signals);
    )
}

生成的 C++ 类包含：

virtual void instanceConstants(int sample_rate) 
{
    fSampleRate = sample_rate;
    fConst0 = std::min<float>(192000.0f, std::max<float>(1.0f, float(fSampleRate)));
}

以及：

virtual void compute(int count, FAUSTFLOAT** inputs, FAUSTFLOAT** outputs) 
{
    FAUSTFLOAT* output0 = outputs[0];
    FAUSTFLOAT* output1 = outputs[1];
    int iSlow0 = count;
    for (int i0 = 0; (i0 < count); i0 = (i0 + 1)) {
        output0[i0] = FAUSTFLOAT(fConst0);
        output1[i0] = FAUSTFLOAT(iSlow0);
    }
}

创建表

可以创建只读表和读写表。只读表信号用 sigReadOnlyTable 创建，接受：

  • 第一个参数为长度
  • 第二个参数为内容
  • 第三个参数为读取索引（介于 0 与 size-1 之间）

并产生被索引的表内容作为其唯一输出。下面这个简单的 DSP 示例：

process = 10,1,int(_) : rdtable;

可以用以下代码写出：

static void test20()
{
    COMPILER
    (
        tvec signals;
        signals.push_back(sigReadOnlyTable(sigInt(10), sigInt(1), sigIntCast(sigInput(0))));

        compile("test20", signals);
    )
}

生成的 C++ 代码包含 itbl0mydspSIG0 静态表定义：

static int itbl0mydspSIG0[10];

将在初始化时调用一次的表填充代码：

void fillmydspSIG0(int count, int* table) 
{
    for (int i1 = 0; (i1 < count); i1 = (i1 + 1)) {
        table[i1] = 1;
    }
}

以及访问 itbl0mydspSIG0 表的 compute 方法：

virtual void compute(int count, FAUSTFLOAT** inputs, FAUSTFLOAT** outputs) 
{
    FAUSTFLOAT* input0 = inputs[0];
    FAUSTFLOAT* output0 = outputs[0];
    for (int i0 = 0; (i0 < count); i0 = (i0 + 1)) {
        output0[i0] = FAUSTFLOAT(itbl0mydspSIG0[int(float(input0[i0]))]);
    }
}

读写表信号用 sigWriteReadTable 创建，接受：

  • 第一个参数为长度
  • 第二个参数为内容
  • 第三个参数为写入索引（介于 0 与 size-1 之间）
  • 第四个参数为表的输入
  • 第五个参数为读取索引（介于 0 与 size-1 之间）

并产生被索引的表内容作为其唯一输出。下面这个 DSP 示例：

process = 10,1,int(_),int(_),int(_) : rwtable;

可以用以下代码写出：

static void test20()
{
    COMPILER
    (
        tvec signals;
        signals.push_back(sigWriteReadTable(sigInt(10), 
                                            sigInt(1), 
                                            sigIntCast(sigInput(0)), 
                                            sigIntCast(sigInput(1)),    
                                            sigIntCast(sigInput(2))));

        compile("test21", signals);
    )
}

生成的 C++ 代码在 mydsp 类中包含 itbl0 定义作为字段：

int itbl0[10];

将在初始化时调用一次的表填充代码：

void fillmydspSIG0(int count, int* table) 
{
    for (int i1 = 0; (i1 < count); i1 = (i1 + 1)) {
        table[i1] = 1;
    }
}

以及从 itbl0 表读取和写入的 compute 方法：

virtual void compute(int count, FAUSTFLOAT** inputs, FAUSTFLOAT** outputs) 
{
    FAUSTFLOAT* input0 = inputs[0];
    FAUSTFLOAT* input1 = inputs[1];
    FAUSTFLOAT* input2 = inputs[2];
    FAUSTFLOAT* output0 = outputs[0];
    for (int i0 = 0; (i0 < count); i0 = (i0 + 1)) {
        itbl0[int(float(input0[i0]))] = int(float(input1[i0]));
        output0[i0] = FAUSTFLOAT(itbl0[int(float(input2[i0]))]);
    }
}

创建波形

下面这个定义波形的 DSP 程序：

process = waveform { 0, 100, 200, 300, 400 };

可以用以下代码写出，其中波形的长度是第一个输出，波形内容本身是用 sigWaveform 创建的第二个输出，以遵循波形语义：

static void test12()
{
    COMPILER
    (
        tvec waveform;
        // 填充波形内容向量
        for (int i = 0; i < 5; i++) {
            waveform.push_back(sigReal(100*i));
        }
        tvec signals;
        signals.push_back(sigInt(waveform.size())); // 波形长度
        signals.push_back(sigWaveform(waveform)); // 波形内容

        compile("test12", signals);
     )
}

生成的 C++ 代码中，fmydspWave0 波形被定义为静态表：

const static float fmydspWave0[5] = {0.0f,100.0f,200.0f,300.0f,400.0f};

并在下面的 compute 方法中使用：

virtual void compute(int count, FAUSTFLOAT** inputs, FAUSTFLOAT** outputs) 
{
    FAUSTFLOAT* output0 = outputs[0];
    FAUSTFLOAT* output1 = outputs[1];
    for (int i0 = 0; (i0 < count); i0 = (i0 + 1)) {
        output0[i0] = FAUSTFLOAT(5);
        output1[i0] = FAUSTFLOAT(fmydspWave0[fmydspWave0_idx]);
        fmydspWave0_idx = ((1 + fmydspWave0_idx) % 5);
    }
}

创建声音文件

soundfile 基元可以访问一组外部定义的声音资源，这些资源用其文件名或完整路径的列表描述。它接受：

  • 声音编号（作为 0 到 255 之间的整数，即常量数值表达式）
  • 声音中的读取索引（如果读取索引大于声音长度，将访问声音的最后一个样本）

生成的块具有：

  • 两个固定输出：第一个是当前访问的声音长度（以帧为单位），第二个是当前访问的声音标称采样率
  • 声音声道本身还有若干额外输出，作为常量数值表达式

soundfile 块用 sigSoundfile 创建，但不能直接使用。它必须与以下函数配合：

  • sigSoundfileLength 用于访问声音长度（以帧为单位）
  • sigSoundfileRate 用于访问声音采样率（以 Hz 为单位）
  • sigSoundfileBuffer 用于访问实际的样本

因此下面的 DSP 代码：

process = 0,0 : soundfile("sound[url:{'tango.wav'}]", 1);

将用 signal API 这样创建：

static void test19()
{
    COMPILER
    (
        tvec signals;
        // soundfile 定义
        Signal sf = sigSoundfile("sound[url:{'tango.wav'}]");
        // 为简化代码使用简单的读取索引 0
        Signal rdx = sigInt(0);
        // 第 0 部分
        Signal part = sigInt(0);
        // 包装索引以避免读取超出缓冲区
        Signal wridx = sigIntCast(sigMax(sigInt(0), 
                       sigMin(rdx, sigSub(sigSoundfileLength(sf, 
                       sigInt(0)),
                       sigInt(1)))));
        // 访问第 0 部分
        signals.push_back(sigSoundfileLength(sf, part));
        // 访问第 0 部分
        signals.push_back(sigSoundfileRate(sf, part));
        // 访问声道 0 和第 0 部分，使用包装后的读取索引
        signals.push_back(sigSoundfileBuffer(sf, sigInt(0), part, wridx));

        compile("test19", signals);
    )
}

并生成下面的 compute 方法：

virtual void compute(int count, FAUSTFLOAT** inputs, FAUSTFLOAT** outputs) 
{
    FAUSTFLOAT* output0 = outputs[0];
    FAUSTFLOAT* output1 = outputs[1];
    FAUSTFLOAT* output2 = outputs[2];
    Soundfile* fSoundfile0ca = fSoundfile0;
    int* fSoundfile0ca_le0 = fSoundfile0ca->fLength;
    int iSlow0 = fSoundfile0ca_le0[0];
    int* fSoundfile0ca_ra0 = fSoundfile0ca->fSR;
    int iSlow1 = fSoundfile0ca_ra0[0];
    int iSlow2 = std::max<int>(0, std::min<int>(0, (iSlow0 + -1)));
    int* fSoundfile0ca_of0 = fSoundfile0ca->fOffset;
    float** fSoundfile0ca_bu0 = static_cast<float**>(fSoundfile0ca->fBuffers);
    float* fSoundfile0ca_bu_ch0 = fSoundfile0ca_bu0[0];
    for (int i0 = 0; (i0 < count); i0 = (i0 + 1)) {
        output0[i0] = FAUSTFLOAT(iSlow0);
        output1[i0] = FAUSTFLOAT(iSlow1);
        output2[i0] = FAUSTFLOAT(fSoundfile0ca_bu_ch0[(fSoundfile0ca_of0[0] + iSlow2)]);
    }
    fSoundfile0 = fSoundfile0ca;
}

定义更复杂的表达式：相位器与振荡器

可以定义更复杂的信号表达式，用辅助定义创建信号。所以下面的 DSP 程序：

import("stdfaust.lib");
process = phasor(440)
with {
     decimalpart(x) = x-int(x);
     phasor(f) = f/ma.SR : (+ : decimalpart) ~ _;
};

可以用以下辅助函数构建，这里用 C 编写：

static Signal decimalpart(Signal x)
{
    return sigSub(x, sigIntCast(x));
}

static Signal phasor(Signal f)
{
    return sigRecursion(decimalpart(sigAdd(sigSelf(), sigDiv(f, SR()))));
}

以及组合它们的主函数：

static void test17()
{
    COMPILER
    (
        tvec signals;
        signals.push_back(phasor(sigReal(440.0)));

        compile("test17", signals);
    )
}

它产生下面的 compute 方法：

virtual void compute(int count, FAUSTFLOAT** inputs, FAUSTFLOAT** outputs) 
{
    FAUSTFLOAT* output0 = outputs[0];
    for (int i0 = 0; (i0 < count); i0 = (i0 + 1)) {
        fRec0[0] = (fConst0 + (fRec0[1] - float(int((fConst0 + fRec0[1])))));
        output0[i0] = FAUSTFLOAT(fRec0[0]);
        fRec0[1] = fRec0[0];
    }
}

现在看下面的振荡器：

 import("stdfaust.lib");
 process = osc(440), osc(440)
 with {
    decimalpart(x) = x-int(x);
    phasor(f) = f/ma.SR : (+ : decimalpart) ~ _;
    osc(f) = sin(2 * ma.PI * phasor(f));
 };

可以用以下代码构建：

static Signal osc(Signal f)
{
    return sigSin(sigMul(phasor(f), sigMul(sigReal(2.0), sigReal(3.141592653))));
}

static void test18()
{
    COMPILER
    (
        tvec signals;
        signals.push_back(osc(sigReal(440.0)));
        signals.push_back(osc(sigReal(440.0)));

        compile("test18", signals);
    )
}

它产生下面的 compute 方法，可以看到，由于两个输出上使用的是同一个振荡器信号，它实际上只计算一次并复制两次：

virtual void compute(int count, FAUSTFLOAT** inputs, FAUSTFLOAT** outputs) 
{
    FAUSTFLOAT* output0 = outputs[0];
    FAUSTFLOAT* output1 = outputs[1];
    for (int i0 = 0; (i0 < count); i0 = (i0 + 1)) {
        fRec0[0] = (fConst0 + (fRec0[1] - float(int((fConst0 + fRec0[1])))));
        float fTemp0 = std::sin((6.28318548f * fRec0[0]));
        output0[i0] = FAUSTFLOAT(fTemp0);
        output1[i0] = FAUSTFLOAT(fTemp0);
        fRec0[1] = fRec0[0];
    }
}

复音 MIDI 可控的简单合成器

下面是一个 MIDI 可控的简单合成器，先是 DSP 代码：

import("stdfaust.lib");
process = organ, organ
with {
    decimalpart(x) = x-int(x);
    phasor(f) = f/ma.SR : (+ : decimalpart) ~ _;
    osc(f) = sin(2 * ma.PI * phasor(f));
    freq = nentry("freq", 100, 100, 3000, 0.01);
    gate = button("gate");
    gain = nentry("gain", 0.5, 0, 1, 0.01);
    organ = gate * (osc(freq) * gain + osc(2 * freq) * gain);

};

然后是使用 signal API 的 C++ 代码：

// 简单的复音 DSP。
static void test24(int argc, char* argv[])
{
    interpreter_dsp_factory* factory = nullptr;
    string error_msg;

    createLibContext();
    {
        tvec signals;

        // 遵循 freq/gate/gain 约定，
        // 见：../manual/midi/#standard-polyphony-parameters
        Signal freq = sigNumEntry("freq", 
                                  sigReal(100), 
                                  sigReal(100), 
                                  sigReal(3000), 
                                  sigReal(0.01));
        Signal gate = sigButton("gate");
        Signal gain = sigNumEntry("gain", 
                                  sigReal(0.5), 
                                  sigReal(0), 
                                  sigReal(1), 
                                  sigReal(0.01));
        Signal organ = sigMul(gate, sigAdd(sigMul(osc(freq), gain), 
                                           sigMul(osc(sigMul(freq, sigInt(2))), gain)));
        // 立体声
        signals.push_back(organ);
        signals.push_back(organ);

        factory = createInterpreterDSPFactoryFromSignals("FaustDSP", 
                                                         signals, 
                                                         0, nullptr, 
                                                         error_msg);
    }
    destroyLibContext();

    // 在 createLibContext/destroyLibContext 作用域之外使用工厂
    if (factory) {
        dsp* dsp = factory->createDSPInstance();
        assert(dsp);

        // 分配复音 DSP
        dsp = new mydsp_poly(dsp, 8, true, true);

        // 分配 MIDI/音频驱动
        jackaudio_midi audio;
        audio.init("Organ", dsp);

        // 创建 GUI
        GTKUI gtk_ui = GTKUI("Organ", &argc, &argv);
        dsp->buildUserInterface(&gtk_ui);

        // 创建 MIDI 控制器
        MidiUI midi_ui = MidiUI(&audio);
        dsp->buildUserInterface(&midi_ui);

        // 启动实时处理
        audio.start();

        // 启动 MIDI
        midi_ui.run();

        // 启动 GUI
        gtk_ui.run();

        // 清理
        audio.stop();
        delete dsp;
        deleteInterpreterDSPFactory(factory);
    } else {
        cerr << error_msg;
    }
}

C API 示例

signal API 也以纯 C API 形式提供。下面是前面某个示例用 C API 重写，以创建信号，其中 LLVM 后端与 C 版本的 createCDSPFactoryFromSignals 函数一起使用（见 llvm-dsp-c.h）来生成 DSP 工厂，然后生成 DSP 实例：

/*
 import("stdfaust.lib");
 process = phasor(440)
 with {
     decimalpart(x) = x-int(x);
     phasor(f) = f/ma.SR : (+ : decimalpart) ~ _;
 };
 */

static Signal decimalpart(Signal x)
{
    return CsigSub(x, CsigIntCast(x));
}

static Signal phasor(Signal f)
{
    return CsigRecursion(decimalpart(CsigAdd(CsigSelf(), CsigDiv(f, SR()))));
}

static void test1()
{
    createLibContext();
    {
        Signal signals[2];
        signals[0] = phasor(CsigReal(2000));
        signals[1] = NULL; // 以 null 结尾的数组

        char error_msg[4096];
        llvm_dsp_factory* factory = createCDSPFactoryFromSignals("test1", 
                                                                 signals, 
                                                                 0, NULL, 
                                                                 "", 
                                                                 error_msg, 
                                                                 -1);

        if (factory) {

            llvm_dsp* dsp = createCDSPInstance(factory);
            assert(dsp);

            // 渲染音频
            render(dsp);

            // 清理
            deleteCDSPInstance(dsp);
            deleteCDSPFactory(factory);

        } else {
            printf("Cannot create factory : %s\n", error_msg);
        }
    }
    destroyLibContext();
}

基于此 API 创建 signal 语言

直接使用 signal API 生成复杂表达式很快就会变得非常棘手且不实用。因此通常需要一种构建在 signal API 之上的语言。这正是框图代数的全部内容，也是整个 Faust 语言本身。

但也可以尝试其他一些方法。例如 Elementary audio language 就构建在类似的 signal language 之上，并使用 JavaScript 作为上层语言，帮助以编程方式创建复杂的输出信号列表。使用基于图形的工具的其他方法当然也可以尝试。
