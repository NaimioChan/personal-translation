autopan

// --------------------------autopan----------------------------
//
// 对立体声输入，左右声道的增益
// 按与速率同步的正弦振荡器调节。本示例
// 反推自 Ableton Live 的「Auto Pan」音频
// 效果。我们没有实现「spin」功能，
// 即让两个 LFO 具有不同频率。
// 也没有使用三角波、方波、噪声等其它 LFO。
//
// 用法
//
// 
// si.bus(2) : autopan(amount, rate, phase, shape) : si.bus(2)
// 
//
// 其中：
//
// - amount：调节 LFO 调制量，该调制量会
// 施加到输入信号上
// - rate：设置 LFO 的频率，单位为赫兹
// - phase：调节左右声道波形之间的偏移量；
// 180 表示两个 LFO 完全反相。
// 取其它值时，你会听到
// 更多整体音量被压低
// 的时刻。
// - shape：调高 shape 会把波形推向
// 上下限，使其形状「变硬」
//------------------------------------------------------------
import("stdfaust.lib");

autopan(amount, rate, phase, shape) = _*gainLeft, _*gainRight
with {

    // 一个饱和器
    // 假设 x 是 -1 到 1 之间的信号。该函数
    // 把输出推向 -1 和 1。随着 shape
    // 参数从 0 变到 1，正弦输入会
    // 越来越接近方波。若 shape 为零，
    // 函数不改变输入信号。
    saturator(shape, x) = result
    with {
        // 可以把 tanh 换成别的饱和器
        result = x, ma.tanh(x*10.) : it.interpolate_linear(shape);
    };

    phase2Gain(phase) = os.oscp(rate, phase)
        : saturator(shape) // 注释掉这一行即可去掉饱和器
        : it.remap(-1., 1., 1.-amount, 1.);

    gainLeft = 0. : phase2Gain;
    gainRight = phase : ma.deg2rad : phase2Gain;
};

amount = hslider("[0]Amount[style:knob]", 0., 0., 1., .001);
rate = hslider("[1]Rate[style:knob][unit:Hz][scale:log]", 1., .05, 90., .001);
phase = hslider("[2]Phase[style:knob][unit:deg]", 180., 0., 360., 15) : si.smoo;
shape = hslider("[3]Shape[style:knob]", 0., 0., 1., 0.001) : si.smoo;

process = hgroup("Auto Pan", autopan(amount, rate, phase, shape));

import("stdfaust.lib");

drumkit(gate, ks, ss, hs) = k_seq, s_seq, h_seq :> /(3)
    with {
        N = outputs(ks);
        env = en.ar(0.001, 0.05);
        kick(g) = g : env : *(os.osc(100));
        snare(g) = g : env : *(no.noise : fi.low_shelf(-60, 3000));
        hihat(g) = g : env : *(no.noise : fi.highpass(3, 10000));
        sequencer(t) = t : ba.selectn(N, gate : ba.pulse_countup_loop(N - 1, 1)) : *(gate : mem);
        k_seq = sequencer(ks) : kick;
        s_seq = sequencer(ss) : snare;
        h_seq = sequencer(hs) : hihat;
    };

drumkit_ui(n) = drumkit(attach(gate, nn), ks, ss, hs)
    with {
        gate = ba.pulse(ba.tempo(bpm));
        ks = par(i, n, checkbox("v:Drum kit/h:[0]kick/%2i"));
        ss = par(i, n, checkbox("v:Drum kit/h:[1]snare/%2i"));
        hs = par(i, n, checkbox("v:Drum kit/h:[2]hihat/%2i"));
        bpm = hslider("v:Drum kit/[4]BPM", 350, 10, 800, 1);

inlooper

// --------------------------inlooper----------------------------
//
// # inlooper
//
// 一个简单的循环器，可从单声道输入采集、回放、倒放并以可调速度循环。
// 到达采集区段末尾时会绕回开头。
// 循环的起点和终点位置以及播放方向均可控制。
// 水平条形图显示循环内的当前播放位置。
//
// 用法
//
// 
// _ : inlooper(length) : _;
// 
// 其中：
//
// - length：循环时长，单位为样本（常量数值表达式）。
//
//------------------------------------------------------------

import("stdfaust.lib");

declare name      "inlooper";
declare version   "1.0";
declare author    "Grame";
declare license   "BSD";
declare copyright "(c)GRAME 2025";

inlooper(length, x) = attach(rwtable(length, 0.0, rec_index, x, loop_index), pos)
with {

    // 用户控件
    speed = hslider("h:Controls/[3]Speed", 1, 0.2, 5, 0.01); // 控制播放速度的滑块
    record = button("h:Controls/[4]Record"); // 开始/停止录制的按钮
    play = checkbox("h:Controls/[5]Play/Stop"); // 开始/停止回放的复选框
    reverse = checkbox("h:Controls/[6]Reverse"); // 倒放复选框

    // 循环起点与终点位置，带平滑（按采集到的输入长度缩放）
    loop_start = length * hslider("[1]Loop_start", 0, 0, 1, 0.01) : si.smoo;
    loop_end = length * hslider("[2]Loop_end", 1, 0, 1, 0.01) : si.smoo;

    // 跟踪录制位置的索引计数器
    rec_index = (+(1) : %(length)) ~ *(record);

    // 循环内的当前播放位置
    loop_index = min_loop_pos + ma.modulo(index, max(ma.EPSILON, loop_length))
    with {

        // 循环位置的下限
        min_loop_pos = min(loop_start, loop_end);

        // 循环时长
        loop_length = abs(loop_end - loop_start);

        // 带方向的播放速度
        read_speed = speed * ba.if(reverse, -1, 1);

        // 跟踪播放位置的索引计数器
        index = windex
        letrec {
            'windex = windex + play * read_speed;
        };

    };

    // 以水平条形图显示播放位置
    pos = (loop_index / length) : hbargraph("[7]Position", 0, 1);

};

process = inlooper(MAX_LENGTH)
with {
    MAX_LENGTH = 5 * 44100; // 44100 Hz 下的 5 秒，必须是常量
};

paradigma

// 使用稀疏卷积滤波器和匹配冲激响应的噪声进行调制合成
// ========================================================================
//
// 一个在 x、y、z 轴上振荡的「3D」振荡器，x、y 的半径差分
// 与可选的、匹配冲激响应的同步噪声一起进入三角滤波器，随后是磁拾音失真。
//
// 适合各种半自然音色，参数少而含义明确，调音很快。
//
// y 轴振荡由 MIDI 音高设定，x 与 z 按纯律比值失谐。
// x 比值为负时振荡器会重新调音。
// 反馈作用于各个振荡分量。
//  
//
// 三份加权（微分）副本带时变偏移，在有损积分器中相加
// （稀疏卷积），之后是峰值谐振滤波器。
//
// 卷积表若积分两次会得到（变化的）三角冲激响应，
// 滚降为 -12 dB/倍频程，并带有变化的等间隔陷波。
// 此处只使用一个积分器。
// 变化的滤波器形状给出伪自然移动的陷波，并带频谱倾斜，
// 并相对时域响应峰值把相位反向因果地摊开。
// 这类似若干原声乐器中扩散的波形或变化的激励。
//
// symmetry 参数控制下降沿，在近乎瞬时的陡沿
// 与对称三角脉冲之间变化，对应不同的激励与音色。
//
// 「adaptive」参数决定滤波器跟随音高与包络的程度。
// Q 增加一个额外的峰值谐振，随后是额外的高通和磁拾音整形器。
// 另有一个单极点低通硬连接到力度（和音高）。
//
// 包络硬连接到各振荡的幅度和滤波器的上升时间。
//
// LFO 连接到音高。
//
// 灵感来自声音合成史，即 Trautonium、Mini Moog、Roland Juno、相位调制合成、
// Variophon 管乐器合成器、物理建模，以及 Thomas D. Rossing 的工作。
//
// gm 2020, 2026-8-24 (this version)
//
//
//
// 参考文献：
// Kot, Vítězslav. (2006). DIGITAL SOUND EFFECTS ECHO AND REVERB BASED ON NON EXPONENTIALLY DECAYING COMB FILTER.
// https://en.wikipedia.org/wiki/Variophon
// Parker, Julian & Zavalishin, Vadim & Le Bivic, Efflam. (2016).
// Reducing The Aliasing Of Nonlinear Waveshaping Using Continuous-Time Convolution.
// Nicholas G. Horton, Thomas R. Moore. (2008). Modelling The Magnetic Pickup Of An Electric Guitar.
// Frei, Beat. Digital Sound Generation I & II, ICST Zurich University of the Arts
// Smith, J.O. Physical Audio Signal Processing,http://ccrma.stanford.edu/~jos/pasp/, online book, 2010 edition

rto1sel = oscgrp(hslider("[1]x[style:knob]", -12, -36, 36, 1));
rto2sel = oscgrp(hslider("[2]z[style:knob]", 7, -36, 36, 1));
fbka = oscgrp(hslider("[3]Feedback[style:knob]", 0, 0, 1, 0.01)<:*:*(0.5/ma.PI));
detune = oscgrp(hslider("[4]Detune[style:knob]", 0.1, 0, 0.5, 0.005)/ma.SR);
nxamnt = oscgrp(hslider("[5]Osc / Noise[style:knob]", 0.7, 0, 1, 0.005)<:*);
constnx = oscgrp(checkbox("[6]continous"));
nxselect = oscgrp(hslider("[7]NoiseColor[style:knob]", 0, 0, 2, 1));

nxseed = seeds, nxselect:rdtable;

// LFO 与包络参数
lfogrp(x) = hgroup("[3]Envelope & LFO", x);
enva = (lfogrp(ba.db2linear(hslider("[1]A[style:knob]", 20, 15, 66, 1))/1000));
envd = (lfogrp(ba.db2linear(hslider("[2]D[style:knob]", 71, 26, 100, 1))/1000)*envpscal);
envs = (lfogrp(hslider("[3]S[style:knob]", 0, 0, 1, 0.01)));
envr = (lfogrp(ba.db2linear(hslider("[4]R[style:knob]", 42, 26, 100, 1))/1000)*envpscal);
lfof = lfogrp(hslider("[5]LFO Hz[style:knob]", 3.5, 0.1, 12, 0.1));
lfvibra = lfogrp(hslider("[6]Vibrato[style:knob]", 0.05, 0, 1, 0.01))<:*;

env = en.adsre(enva, envd*envpscal, envs, envr*envpscal, gate);
envlfo = en.adsre(enva, 0, 1, envr*envpscal, gate);
envg = env*gain;
envgr3 = env*(gain^(1/3));

lfosn = qsin(mphasor(lfof/ma.SR))*envlfo;

// 三角滤波器参数
fltgrp(x) = hgroup("[4]Filter", x);
wid = fltgrp(hslider("[1]RisingEdge[style:knob]", 1.67, 1, 2.64, 0.001))<:_, _:*:_*(-1):2^_;
widsym = fltgrp(hslider("[2]Symmetry[style:knob]", 0, 0, 1, 0.01));
fiq = fltgrp(hslider("[3]Q[style:knob]", 1.25, 0.5, 3.87, 0.01))<:*;
fiadapt = fltgrp(hslider("[4]Adaption[style:knob]", 2.5, 1, 3, 0.25));

hpoffs = fltgrp(hslider("[5]HighPass[style:knob]", -2, -2, 2, 0.03125)):2^_;
pickupdrive = fltgrp(hslider("[6]Pickup[style:knob]", -2, -2, 2, 0.03125)):2^_;

// 调制频率比值
rto1oct = rto1sel/12:floor;
rto1semi = rto1sel+36:_%12;
rto1a = frtonum, rto1semi:rdtable;
rto1b = frtodiv, rto1semi:rdtable;
rto1 = (rto1a/rto1b)*(2^rto1oct);
rto1r = min((1/rto1), 1);

rto2oct = rto2sel/12:floor;
rto2semi = rto2sel+36:_%12;
rto2a = frtonum, rto2semi:rdtable;
rto2b = frtodiv, rto2semi:rdtable;
rto2 = (rto2a/rto2b)*(2^rto2oct);
rto2r = min((1/rto2), 1);

// x 振荡低于基频时重新调音
retune = ba.if((rto1oct+rto1semi)<0, rto1, 1);

// 音高
lg2f = ma.log2(f/440);
// 按音高的轻微失谐
stretch = 0.003*lg2f;
// 按音高的缩放
envpscal = (-1.5*lg2f):ba.db2linear;
// 含调制与失谐的 f
fplus = f*bend+lfosn*lfvibra*f*0.5/12*envg+stretch;

w = retune*f/ma.SR;
w2 = rto1*w;
w3 = rto2*w;
wplus = retune*fplus/ma.SR;

// 反馈缩放
fbk1 = fbka*(0.5-w)^4;
fbk2 = fbka*(0.5-w2)^4*rto1r;
fbk3 = fbka*(0.5-w3)^4*rto2r;

// 按频率的调制衰减
redux = octcurve(w);
redux1 = octcurve(rto1*w);
redux2 = octcurve(rto2*w);
modep = envgr3*redux;
modep1 = envgr3*redux1;
modep2 = envgr3*redux2;

// 振荡器
// ==========
// 在 -0.5 与 0.5 之间回绕
wrap(n) = n-(floor(n+0.5));
// 基于 Bhaskara I 的近似余弦曲线
qsincurve(x) = 1-((x*x)<:*(1.2253517*16), (_<:*:*(-3.60562732*16)):>_);
// 类余弦的半圆
qsin(x) = x+(0.5):wrap<:(abs:-(0.25):qsincurve), _:ma.copysign;
// 完整正弦波，输入 -0.5...0.5
// 反馈与调制深度衰减曲线
fbcurve(x) = 1/x<:_, 1:max:ma.log2:_-6<:_, 0:max:_*(1/10);
octcurve(x) = 1/x<:_, 1:max:ma.log2:_-2<:_, 0:max:_*(1/6)<:*;
// 基本相位累加器 -0.5...0.5
mphasor(fw) = (+(fw)~(wrap));
// 基本正弦振荡器
oscsn(fw, off) = mphasor(fw)+off:qsin;
// 带反馈和 IIR 反馈滤波器的振荡器
osc1(fw, off) = ((fw), +(off):(oscsn))~(*(fbka):fi.pole(0.5):_*fbcurve(fw));

// 3D 调制
// 一个点绕 3 个轴旋转，x、y 半径的微分随后作为信号
oscy(fw, off) = (osc1(fw, off)*osc1(fw*rto2+2*detune, 0.75+off)*modep2)*modep;
oscx(fw, off) = (osc1(fw*rto1+detune, 0.25+off)*osc1(fw*rto2+2*detune, 0.25+off)*modep2)*modep1;
// 2D 半径 -> 微分器（信号稍后会被积分），加上（同步的）噪声
oscxy(fw, off) = (oscy(fw, off)<:*), (oscx(fw, off)<:*):+:sqrt<:_-_':_*(1-nxamnt)+sncnx(fw, off)*envg*nxamnt;

// 同步噪声波形
// 异或随机，全局 constnx 标志为 1 时递归，否则仅用 nxseed + step 作为种子
xornx(step) = (*(constnx):+(nxseed+step):int<:_, (_<<13):xor<:_, (_>>17):xor<:_, (_<<5):xor:_&536870911<:_, (float:_*(1/2147483648):wrap))~_:!, _*(1/(constnx+1));
sncwin(ph) = ba.if((abs(ph)<4*w), (abs(ph)/(4*w))-0.25:qsin*(0.5)+0.5, 1);
// 同步或连续噪声，因过渡简单（8 个样本的余弦渐变到零）而不抗混叠
sncnx(fw, off) = mphasor(fw):_+off:wrap<:(_<:_+0.5<:_, _+fw<:_/fw, _/fw<:(_<:_, _), _:_, floor, floor<:_, (_<:_, _), _:(_-_, xornx, xornx):(_<:_, _), _, _:route(4, 4, 1, 1, 3, 2, 2, 3, 4, 4):(1-_)*_, _*_:>_*5), sncwin:_, ba.if(constnx, 1, _):*;

// 上升沿可变、下降沿相对的三角滤波器
// =================================================================
octavesfromtop = 1/w<:_, 1:max:ma.log2;
widbrighten = octavesfromtop-4<:_, 0:max:1/(2^_);
// 用于提亮低音的曲线（== 减小三角宽度）
widredux = octavesfromtop-4<:_, 0:max: // 用于降低高音亮度（== 增大宽度）的曲线
_*(1/6)<:_, 1:min:sqrt*widbrighten;

// 到最大宽度的差，以倍频程计，高八度递减
dwo = (0.25/wid):max(_, 1):ma.log2:_*widredux;

minwid = 4/(ma.SR/f);
// 三角边的最小时间
maxwid = 0.5;
// 三角边的最大时间

// 按包络缩放
// 1. 上升沿
wid2 = wid*(fiadapt^(dwo*(1-envg))):min(_, maxwid):max(_, minwid);
// 2. 相对上升沿的下降沿
wid2e = (1/512+wid2*widsym):min(_, maxwid):max(_, minwid);
fiw = w/wid2;
// 上升沿的 t 增量
fiwtail = w/wid2e;
// 下降沿的 t 增量

fiwcompens = 0.25*wid2e/(2/512);
// 增益补偿

// 三角系数
apg0 = fiw*fiwcompens;
apg1 = -apg0-fiwtail*fiwcompens;
apg2 = fiwtail*fiwcompens;

// 积分频率约 25 Hz
igpole = 1.0-25.0/ma.SR;
// 谐振频率
resf = (fplus/(wid2+wid2e)):min(_, (0.166*ma.SR));

// 额外的随音高与力度变化的单极点低通，以及全局声部电平
lpfw = w+(5000/ma.SR-w)*gain*gain;
lpp = ma.E^(-2*ma.PI*lpfw);
lpa = 1.5*(1-lpp);

// 类拾音器失真
// =======================
// （x 距离 = 0.25）
pickup(x) = x, // in < 1.2e-4 时正常
(x, (x^2:_+0.25:_^(3/2)):/), // ILO 抗混叠：
(0.25^(3/2)/(sqrt(x*x+1)):ma.neg:_+0.25^(3/2)): // 选择
ba.if((_:abs:_<=1.2e-4), _, _):_*(0.25^(4/3));

// 基本合成器声部：3 份相移加权副本、随力度变化的低通、积分
synthvox(fw, ph2, ph3, g1, g2, g3) = (oscxy(fw, 0):_*g1), (oscxy(fw, ph2):_*g2), (oscxy(fw, ph3):_*g3):>_:_*lpa:fi.pole(lpp):fi.pole(igpole);

// 声音
// =====
process = synthvox(wplus, wid2, wid2e, apg0, apg1, apg2):fi.svf.peak(resf, fiq):fi.svf.hp(hpoffs*fplus/(wid2+wid2e):min(_, ma.SR*0.166), 0.707):_*pickupdrive:pickup*(min(1/pickupdrive, 1))<:_, _;

effect = re.dattorro_rev_default;

sflooper

// --------------------------sflooper----------------------------
//
// # sflooper
//
// 一个简单的循环器，可在声音文件内回放、倒放并以可调速度循环。
// 它自动适配输入声音文件的声道数，并在到达
// 文件末尾时绕回开头。
// 循环的起点和终点位置以及播放方向均可控制。
// 水平条形图显示循环内的当前播放位置。
//
// 用法
//
// 
// sflooper(sound) : si.bus(outputs(sound)-2);
// 
// 其中：
//
// - sound：输入的声音文件。
//
// 示例：
//
// 
// sound = soundfile("sound[url:{'loop.wav'}]",2);
// sflooper(sound);
// 
//------------------------------------------------------------

import("stdfaust.lib");

declare name      "looper";
declare version   "1.0";
declare author    "Grame";
declare license   "BSD";
declare copyright "(c)GRAME 2025";

// 播放按钮的状态用于控制 loop_reader 索引的递增和实际输出信号的生成。

sflooper(sf) = (0, play * loop_index) : sf : !, !, attach(_,pos), si.bus(outputs(sf)-3)
with {

    // 用户控件
    speed = hslider("h:Controls/[3]Speed", 1, 0.2, 5, 0.01); // 控制播放速度的滑块
    play = checkbox("h:Controls/[4]Play/Stop"); // 开始/停止回放的复选框
    reverse = checkbox("h:Controls/[5]Reverse"); // 倒放复选框

    // 循环起点与终点位置，带平滑（按声音文件长度缩放）
    loop_start = length * hslider("[1]Loop_start", 0, 0, 1, 0.01) : si.smoo;
    loop_end = length * hslider("[2]Loop_end", 1, 0, 1, 0.01) : si.smoo;

    // 声音文件长度
    length = (0, 0) : sf : (_, si.block(outputs(sf) - 1));  

    // 循环内的当前播放位置
    loop_index = min_loop_pos + ma.modulo(index, max(ma.EPSILON, loop_length))
    with {

        // 循环位置的下限
        min_loop_pos = min(loop_start, loop_end);

        // 循环时长
        loop_length = abs(loop_end - loop_start);

        // 带方向的播放速度
        read_speed = speed * ba.if(reverse, -1, 1);

        // 跟踪播放位置的索引计数器
        index = windex
        letrec {
            'windex = windex + play * read_speed;
        };

    };

    // 以水平条形图显示播放位置
    pos = (loop_index / length) : hbargraph("[6]Position", 0, 1);

};

declare soundfiles "https: // raw.githubusercontent.com/sletz/faust-sampler/main";
sound = soundfile("sound[url:{'violon.wav'}]", 2);

process = sflooper(sound);

matrix(M,N) = tgroup("Matrix: %M x %N", par(in, N, _)
              <: par(out, M, mixer(N, out))) with {
  fader(in) = ba.db2linear(vslider("Input %in", -10, -96, 4, 0.1));
  mixer(N,out) = hgroup("Output %out", par(in, N, *(fader(in)) ) :> _ );
};

Bd = par(i,p,mem) : B; // 传统定义所需的输入延迟
vsum(N) = si.bus(2*N) :> si.bus(N); // 两个 N 维向量的向量和

// 演示非零初始状态，遵循传统定义：
impulse = 1-1'; // 若初始状态为零，设 impulse = 0 或简化代码
x0 = par(i,N,i*impulse); // 初始状态 = (0,1,2,3,...,N-1)

system = si.bus(p) <: D, (Bd : vsum(N)~(A), x0 : vsum(N) : C) :> si.bus(q);

process = system;
