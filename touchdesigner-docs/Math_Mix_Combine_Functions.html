float TDLoop(float val, float low, float high)
{
	float v = (val - low) / (high - low);
	return mix(low, high, fract(v));
}

float TDZigZag(float val, float low, float high)
{
	float v = (val - low) / (high - low);
	bool flip = int(floor(v)) % 2 == 1 ? true : false;
	v = fract(v);
	
	if (flip)
		v = 1.0 - v;
		
	return mix(low, high, v);
}
