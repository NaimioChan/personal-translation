这段代码在原始 16 位 PCM 文件中生成一个扫描的抗混叠锯齿波。
它基于一个五阶多项式的四重微分。多项式的谐波（以及混叠谐波）每倍频程衰减 5*6 dB。微分器校正频谱与波形，而混叠谐波仍会被衰减。

/* clair.c         部分考试 2b
   T.Rochebois
   02/03/98
*/
#include <stdio.h>
#include <math.h>
main()
{
  double phase=0,dphase,freq,compensation;
  double aw0=0,aw1=0,ax0=0,ax1=0,ay0=0,ay1=0,az0=0,az1=0,sortie;
  short aout;
  int sr=44100;       //采样率 (Hz)
  double f_debut=55.0;//起始频率 (Hz)
  double f_fin=sr/6.0;//终止频率 (Hz)
  double octaves=log(f_fin/f_debut)/log(2.0);
  double duree=50.0;  //时长 (s)
  int i;
  FILE* f;
  f=fopen("saw.pcm","wb");
  for(i=0;i<duree*sr;i++)
  {
    //指数扫频
    //可替换为你想要的任何方式。
    freq=f_debut*pow(2.0,octaves*i/(duree*sr));
    dphase=freq*(2.0/sr);     //归一化相位增量
    phase+=dphase;            //相位递增
    if(phase>1.0) phase-=2.0; //相位回绕 (-1,+1)

    //多项式计算（在 -1 和 +1 处高度连续）
    //        7       1  3    1   5
    //P(x) = --- x - -- x  + --- x
    //       360     36      120
    aw0=phase*(7.0/360.0 + phase*phase*(-1/36.0 + phase*phase*(1/120.0)));
    // 四点差分（一阶高通滤波器）
    ax0=aw1-aw0; ay0=ax1-ax0; az0=ay1-ay0; sortie=az1-az0;
    //补偿四点差分器的衰减
    //这可在“控制率”下计算，再按采样率
    //进行线性插值。
    compensation=1.0/(dphase*dphase*dphase*dphase);
    // 补偿与输出
    aout=(short)(15000.0*compensation*sortie);
    fwrite(&aout,1,2,f);
    //差分器的历史状态
    aw1=aw0; ax1=ax0; ay1=ay0; az1=az0;
  }
  fclose(f);
}
