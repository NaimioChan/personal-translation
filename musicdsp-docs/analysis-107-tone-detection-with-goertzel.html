使用 Goertzel 的音调检测

  • 作者或来源：on.biu.ii@rnepse
  • 类型：Goertzel
  • 创建时间：2004-04-07 09:37:10
  • 关联文件：http://www.ii.uib.no/~espenr/tonedetect.zip

Goertzel 本质上是对频谱的局部做 DFT，而不是像通常用 FFT 那样对整段频谱做。所以如果你只想查看某些频率的功率，它可能更合适。听说它很适合 DTMF 检测。

WNk 的计算并非 100% 正确，但看起来能用 ;) 另外代码是 C++ 的，若要用 C 编译，可能得做些小调整。

/** Goertzel 算法音调检测
 *
 * 本程序在样本中搜索音调（正弦波），并报告它在各个频率上测得的 dB 值。
 * 稍加阈值判断即可扩展为输出检测真/假的结果。
 * 我远不能确定 Goertzel 的实现 100% 正确，但它确实能用 :)
 * 
 * 提示：SAMPLERATE、BUFFERSIZE、FREQUENCY、NOISE 和 SIGNALVOLUME 都会影响所报告的 dB 结果。请
 * 调整这些参数，找到最适合你的应用的设置。另外，它对噪声（至少是白噪声）相当敏感，这
 * 有点遗憾。我也不知道 Goertzel 是否真适合用浮点值表示频率……而且用 44100 作为
 * 采样率来检测 6000 Hz 音调确实有点傻，我知道 :)
 * 
 * 作者：Espen Riskedal, espenr@ii.uib.no，2002 年 7 月
 */

#include <iostream>
#include <cmath>
#include <cstdlib>

using std::rand;
// 数学部分
using std::cos;
using std::abs;
using std::exp;
using std::log10;
// iostream 相关
using std::cout;
using std::endl;

#define PI 3.14159265358979323844
// 可按需修改这些宏定义
#define SAMPLERATE 44100
#define BUFFERSIZE 8820
#define FREQUENCY 6000
#define NOISE 0.05
#define SIGNALVOLUME 0.8

/**  Goertzel 算法用二阶滤波器计算输入信号的第 k 个 DFT 系数。 
 *   http://ptolemy.eecs.berkeley.edu/papers/96/dtmf_ict/www/node3.html。 
 *   基本上它只对我们想检测的那个频率做 DFT，其余频率不做（FFT 会计算所有频率）。
 */
float goertzel(float *x, int N, float frequency, int samplerate) {
    float Skn, Skn1, Skn2;
    Skn = Skn1 = Skn2 = 0;

    for (int i=0; i<N; i++) {
	Skn2 = Skn1;
	Skn1 = Skn;
	Skn = 2*cos(2*PI*frequency/samplerate)*Skn1 - Skn2 + x[i];
    }

    float WNk = exp(-2*PI*frequency/samplerate); // 此版本忽略复数部分
    //float WNk = exp(-2*j*PI*k/N);
    return (Skn - WNk*Skn1);
}

/** 生成指定频率的音调 
 *  来源：http://groups.google.com/groups?hl=en&lr=&ie=UTF-8&oe=UTF-8&safe=off&selm=3c641e%243jn%40uicsl.csl.uiuc.edu
 */
float *makeTone(int samplerate, float frequency, int length, float gain=1.0) {
    //y(n) = 2 * cos(A) * y(n-1) - y(n-2)
    //A = (目标频率) * 2 * PI / (采样频率)
    //A 的单位是弧度。
    // 目标频率必须 <= 采样频率的 1/2。
    float *tone = new float[length];
    float A = frequency*2*PI/samplerate;

    for (int i=0; i<length; i++) {
	if (i > 1) tone[i]= 2*cos(A)*tone[i-1] - tone[i-2];
	else if (i > 0) tone[i] = 2*cos(A)*tone[i-1] - (cos(A));
	else tone[i] = 2*cos(A)*cos(A) - cos(2*A);
    }

    for (int i=0; i<length; i++) tone[i] = tone[i]*gain;

    return tone;
}

/** 给样本添加白噪声 */
void *addNoise(float *sample, int length, float gain=1.0) {
    for (int i=0; i<length; i++) sample[i] += (2*(rand()/(float)RAND_MAX)-1)*gain;
}

/** 返回信号功率/dB */
float power(float value) {
    return 20*log10(abs(value));
}

int main(int argc, const char* argv) {
    cout << "Samplerate: " << SAMPLERATE << "Hz\n";
    cout << "Buffersize: " << BUFFERSIZE << " samples\n";
    cout << "Correct frequency is: " << FREQUENCY << "Hz\n";
    cout << " - signal volume: " << SIGNALVOLUME*100 << "%\n";
    cout << " - white noise: " << NOISE*100 << "%\n";

    float *tone = makeTone(SAMPLERATE, FREQUENCY, BUFFERSIZE, SIGNALVOLUME);
    addNoise(tone, BUFFERSIZE,NOISE);

    int stepsize = FREQUENCY/5;

    for (int i=0; i<10; i++) {
	int freq = stepsize*i;
	cout << "Trying freq: " << freq << "Hz  ->  dB: " << power(goertzel(tone, BUFFERSIZE, freq, SAMPLERATE)) << endl;
    }
    delete tone;

    return 0;
}
