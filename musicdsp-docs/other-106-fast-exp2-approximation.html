快速 exp2 近似

  • 作者或来源：Laurent de Soras (moc.ecrofmho@tnerual)
  • 创建时间：2002-07-29 18:42:23

定点运算中 exp2 的部分近似。它精确地是：
[0 ; 1[ -> [0.5 ; 1[
f : x |-> 2^(x-1)
要得到完整的 exp2 函数，必须把数的整数部分与小数部分分开。整数部分可以用位移处理。用该函数处理小数部分，再把两个结果相乘。
最大误差只有 0.3%，对两次乘法来说相当不错！你还能得到一阶导数的连续性。

-- Laurent

// val 是 0x0 - 0xFFFF（[0 ; 1[）范围内的 16 位定点值
// 返回 0x80000000 - 0xFFFFFFFF（[0.5 ; 1[）范围内的 32 位定点值
unsigned int fast_partial_exp2 (int val)
{
   unsigned int   result;

   __asm
   {
      mov eax, val
      shl eax, 15           ; eax = input [31/31 bits]
      or  eax, 080000000h   ; eax = input + 1  [32/31 bits]
      mul eax
      mov eax, edx          ; eax = (input + 1) ^ 2 [32/30 bits]
      mov edx, 2863311531   ; 2/3 [32/32 bits], rounded to +oo
      mul edx               ; eax = 2/3 (input + 1) ^ 2 [32/30 bits]
      add edx, 1431655766   ; + 4/3 [32/30 bits] + 1
      mov result, edx
   }

   return (result);
}
