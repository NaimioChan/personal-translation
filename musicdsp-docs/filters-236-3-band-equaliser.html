三段均衡器

  • 作者或来源：Neil C
  • 创建时间：2006-08-29 20:34:25

简单的三段均衡器，低频与高频频率可调……

算法相当快，输出质量好（所有增益设为 1.0 时听起来是声学透明的）

用法……

1. 首先需要为均衡器声明状态

EQSTATE eq;

2. 现在初始化状态（假设输出采样率为 48Khz）

set_3band_state(eq,880,5000,480000);

现在你的 EQ 频段如下（近似！）

低频段 = 0Hz 到 880Hz
中频段 = 880Hz 到 5000Hz
高频段 = 5000Hz 到 24000Hz

3. 把增益设为某些值……

  eq.lg = 1.5;  // 低音提升 50%
  eq.mg = 0.75; // 中频衰减 25%
  eq.hg = 1.0;  // 高频段不动

4. 现在可以对样本做 EQ 了

out_sample = do_3band(eq,in_sample)

//| 结构体 |
// ------------

typedef struct
{
  // 滤波器 #1（低频段）

  double  lf;       // 频率
  double  f1p0;     // 极点 ...
  double  f1p1;     
  double  f1p2;
  double  f1p3;

  // 滤波器 #2（高频段）

  double  hf;       // 频率
  double  f2p0;     // 极点 ...
  double  f2p1;
  double  f2p2;
  double  f2p3;

  // 采样历史缓冲区

  double  sdm1;     // 采样数据减 1
  double  sdm2;     //                   2
  double  sdm3;     //                   3

  // 增益控制

  double  lg;       // 低频增益
  double  mg;       // 中频增益
  double  hg;       // 高频增益

} EQSTATE;

//| 导出 |
// ---------

//----------------------------------------------------------------------------
//
//                                3 Band EQ :)
//
// EQ.C - Main Source file for 3 band EQ
//
// (c) Neil C / Etanza Systems / 2K6
//
// Shouts / Loves / Moans = etanza at lycos dot co dot uk 
//
// This work is hereby placed in the public domain for all purposes, including
// use in commercial applications.
//

// this software.
//
//----------------------------------------------------------------------------

// 说明：
//
// - 原始滤波器代码由 Paul Kellet 编写（musicdsp.pdf）
//
// - 使用 4 个一阶滤波器串联，可提供 24dB/oct 的斜率
//
// - 现在带有 P4 非正规数修正 :)


//----------------------------------------------------------------------------

// ----------
//| 包含 |
// ----------

#include <math.h>
#include "eq.h"


// -----------
//| 常量 |
// -----------

static double vsa = (1.0 / 4294967295.0);   // 极小量（非正规数修正）


// ---------------
//| 初始化 EQ |
// ---------------

// 推荐频率为 ...
//
//  lowfreq  = 880 Hz
//  highfreq = 5000 Hz
//
// 将 mixfreq 设为系统实际使用的采样率（例如 48KHz）

void init_3band_state(EQSTATE* es, int lowfreq, int highfreq, int mixfreq)
{
  // 清除状态 

  memset(es,0,sizeof(EQSTATE));

  // 将低/中/高频增益设为 1

  es->lg = 1.0;
  es->mg = 1.0;
  es->hg = 1.0;

  // 计算滤波器截止频率

  es->lf = 2 * sin(M_PI * ((double)lowfreq / (double)mixfreq)); 
  es->hf = 2 * sin(M_PI * ((double)highfreq / (double)mixfreq));
}


// ---------------
//| 处理单个采样 |
// ---------------

// - 采样可以是任意范围 :)
//
// 注意输出取决于各频段的增益设置 
// （尤其是低频），因此输出前可能需要削波，不过你 
// 早就知道这一点了 :)

double do_3band(EQSTATE* es, double sample)
{
  // 局部变量

  double  l,m,h;      // 低 / 中 / 高 —— 采样值

  // 滤波器 #1（低通）

  es->f1p0  += (es->lf * (sample   - es->f1p0)) + vsa;
  es->f1p1  += (es->lf * (es->f1p0 - es->f1p1));
  es->f1p2  += (es->lf * (es->f1p1 - es->f1p2));
  es->f1p3  += (es->lf * (es->f1p2 - es->f1p3));

  l          = es->f1p3;

  // 滤波器 #2（高通）

  es->f2p0  += (es->hf * (sample   - es->f2p0)) + vsa;
  es->f2p1  += (es->hf * (es->f2p0 - es->f2p1));
  es->f2p2  += (es->hf * (es->f2p1 - es->f2p2));
  es->f2p3  += (es->hf * (es->f2p2 - es->f2p3));

  h          = es->sdm3 - es->f2p3;

  // 计算中频段（signal - (low + high)）

  m          = es->sdm3 - (h + l);

  // 缩放、混合并存储

  l         *= es->lg;
  m         *= es->mg;
  h         *= es->hg;

  // 移动历史缓冲区 

  es->sdm3   = es->sdm2;
  es->sdm2   = es->sdm1;
  es->sdm1   = sample;                

  // 返回结果

  return(l + m + h);
}

***********************************************
  TEq=Class
  private
    lf,f1p0,f1p1,f1p2,f1p3:double;
    hf,f2p0,f2p1,f2p2,f2p3:double;
    sdm1,sdm2,sdm3:double;
    vsa:double;
    lg,mg,hg:double;
    Function do_3band(sample:Smallint):Smallint;
  public
    constructor create;
    destructor destroy;override;
    procedure init_3band_state(lowfreq,highfreq,mixfreq:integer;BassGain,MidGain,HighGain:Double);
    procedure Equalize(const Data: Pointer; DataSize: DWORD);
  end;


constructor TEQ.create;
begin
  inherited create;
  vsa := (1.0 / 4294967295.0);
  lg := 1.0;
  mg := 1.0;
  hg := 1.0;

end;


destructor TEQ.destroy;
begin
  inherited destroy;
end;

procedure TEQ.init_3band_state(lowfreq,highfreq,mixfreq:integer;BassGain,MidGain,HighGain:Double);
begin

 {(880,5000,44100,1.5,0.75,1.0)
 eq.lg = 1.5; // Boost bass by 50%
 eq.mg = 0.75; // Cut mid by 25%
 eq.hg = 1.0; // Leave high band alone }

 lg := 1+(BassGain/100);
 mg := 1+(MidGain/100);
 hg := 1+(HighGain/100);

// Calculate filter cutoff frequencies

 lf := 2 * sin(PI * (lowfreq / mixfreq));
 hf := 2 * sin(PI * (highfreq / mixfreq));
end;

Function TEQ.do_3band(sample:Smallint):Smallint;
var l,m,h:double;
    res:integer;
begin

// Filter #1 (lowpass)

  f1p0 := f1p0 + (lf * (sample - f1p0)) + vsa;
  f1p1 := f1p1 + (lf * (f1p0 - f1p1));
  f1p2 := f1p2 + (lf * (f1p1 - f1p2));
  f1p3 := f1p3 + (lf * (f1p2 - f1p3));

  l := f1p3;

// Filter #2 (highpass)

  f2p0 := f2p0 + (hf * (sample - f2p0)) + vsa;
  f2p1 := f2p1 + (hf * (f2p0 - f2p1));
  f2p2 := f2p2 + (hf * (f2p1 - f2p2));
  f2p3 := f2p3 + (hf * (f2p2 - f2p3));

  h := sdm3 - f2p3;

// Calculate midrange (signal - (low + high))

  m := sdm3 - (h + l);

// Scale, Combine and store

  l := l * lg;
  m := m * mg;
  h := h * hg;

// Shuffle history buffer

  sdm3 := sdm2;
  sdm2 := sdm1;
  sdm1 := sample;

// Return result
  res := trunc(l+m+h);
  if res > 32767 then res := 32767 else if res < -32768 then res := -32768;

  result := res;
end;

procedure TEQ.Equalize(const Data: Pointer; DataSize: DWORD);
var pSample: PSmallInt;
begin
  pSample := Data;
  while DataSize > 0 do
  begin
    pSample^ := do_3band(pSample^);
    Inc(pSample);
    Dec(DataSize, 2);
  end;
end;
