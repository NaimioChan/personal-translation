Double 转 Int

  • 作者或来源：many people, implementation by Andy M00cho
  • 类型：指针转换（向零舍入，即截断）
  • 创建时间：2002-01-17 03:04:41

-字面上就是平台无关的。只要你有 IEEE FP 数，这就能用，前提是你不期望返回大于 16 位的有符号整数 :)
-仅对 [-32768.0,32767.0] 范围内的 FP 数正确工作
-FPU 必须处于双精度模式

typedef double lreal;
typedef float  real;
typedef unsigned long uint32;
typedef long int32;

   //2^36 * 1.5，(52-_shiftamt=36) 利用有限精度实现向下取整
   //16.16 定点表示

const lreal _double2fixmagic = 68719476736.0*1.5;
const int32 _shiftamt        = 16;

#if BigEndian_
        #define iexp_                           0
        #define iman_                           1
#else
        #define iexp_                           1
        #define iman_                           0
#endif //BigEndian_

// Real2Int
inline int32 Real2Int(lreal val)
{
   val= val + _double2fixmagic;
   return ((int32*)&val)[iman_] >> _shiftamt;
}

// Real2Int
inline int32 Real2Int(real val)
{
   return Real2Int ((lreal)val);
}

For the x86 assembler freaks here's the assembler equivalent:

__double2fixmagic    dd 000000000h,042380000h

fld    AFloatingPoint Number
fadd   QWORD PTR __double2fixmagic
fstp   TEMP
movsx  eax,TEMP+2

On PC this may be faster/easier:

int ftoi(float x)
{
  int res;
  __asm
  {
    fld x
    fistp res
  }
  return res;
}

int dtoi(double x)
{
  return ftoi((float)x);
}
