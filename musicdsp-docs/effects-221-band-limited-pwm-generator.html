// SelfPMpwm.cpp

// 抗混叠 PWM 振荡器

// 基于 Thierry Rochebois（98）的教程代码。
// 其本身受 Norio Tomisawa（81）的美国专利 4249447 启发。
// 注释由 P.Sernine（06）添加/翻译。

// 本程序生成 44100Hz 的裸 PCM 单声道 wave 文件。
// 它基于 Tomisawa 的自相位调制正弦波发生器。
// Rochebois 用一个公共相位累加器驱动两个半 Tomisawa 振荡器。
// 每个半 Tomisawa 振荡器产生一个频带受限的锯齿波
// （频带限制程度取决于反馈系数 B）。
// 这些半振荡器按所需脉冲宽度做相位偏移，
// 最后合并得到 PW 信号。
// 注意：抗“游移”滤波器是良好实现
// Tomisawa 方法的关键特性。
#include <math.h>
#include <stdio.h>
const float pi=3.14159265359f;
int main()
{
  float freq,dphi; //!< 频率（Hz）和相位增量（rad/sample）
  float dphif=0;   //!< 经滤波（抗咔哒声）的相位增量
  float phi=-pi;   //!< 相位
  float Y0=0,Y1=0; //!< 反馈记忆
  float PW=pi;     //!< 脉宽 ]0,2pi[
  float B=2.3f;    //!< 反馈系数
  FILE *f=fopen("SelfPMpwm.pcm","wb");
  // 音序（'a'=mi=E）
  // 如偏好其他旋律可自行修改。
  static char seq[]="aiakahiafahadfaiakahiahafahadf"; //!< 音序
  int note=sizeof(seq)-2;  //!< 音序中的音符编号
  int octave=0;     //!< 八度编号
  float env,envf=0; //!< 包络与滤波后的包络
  for(int ns=0;ns<8*(sizeof(seq)-1)*44100/6;ns++)
  {
//waveform control --------------------------------------------------
    //频率
    //freq=27.5f*powf(2.0f,8*ns/(8*30*44100.0f/6)); //sweep
    freq=27.5f*powf(2.0f,octave+(seq[note]-'a'-5)/12.0f);
    //freq*=(1.0f+0.01f*sinf(ns*0.0015f));        //vibrato
    dphi=freq*(pi/22050.0f);   // 相位增量
    dphif+=0.1f*(dphi-dphif);
    //音符与包络触发
    if((ns%(44100/6))==0)
    {
      note++;
      if(note>=(sizeof(seq)-1))// 音序循环
      {
        note=0;
        octave++;
      }
      env=1; //包络设置
      //PW=pi*(0.4+0.5f*(rand()%1000)/1000.0f); //random PW
    }
    env*=0.9998f;              // 指数包络
    envf+=0.1f*(env-envf);     // 去咔哒声包络
    B=1.0f;                    // 反馈系数
    //试试这个，可得到不错的贝斯音色：
    //B*=envf*envf;              // 反馈由包络控制
    B*=2.3f*(1-0.0001f*freq);  // 反馈限制
    if(B<0)
      B=0;

//waveform generation -----------------------------------------------
    //公共相位
    phi+=dphif;                 // 相位增量
    if(phi>=pi)
      phi-=2*pi;               // 相位回绕

    // "phase"    半 Tomisawa 发生器 0
    // B*Y0 -> 自相位调制
    float out0=cosf(phi+B*Y0); // 半输出 0
    Y0=0.5f*(out0+Y0);         // 防振荡滤波器

    // "phase+PW" 半 Tomisawa 发生器 1
    // B*Y1 -> 自相位调制
    // PW   -> 相位偏移
    float out1=cosf(phi+B*Y1+PW); // 半输出 1
    Y1=0.5f*(out1+Y1);            // 防振荡滤波器

    // 合成、包络与输出
    short s=short(15000.0f*(out0-out1)*envf);
    fwrite(&s,2,1,f);          // 文件输出
  }
  fclose(f);
  return 0;
}
