x86/SSE CPU 的快速立方根、平方根与倒数

  • 作者或来源：moc.noicratse@ajelak
  • 创建时间：2005-05-29 18:36:40

所有这些方法都用 SSE 指令或位操作技巧得到立方根、平方根或倒数的粗略近似，再用一步或多步 Newton-Raphson 近似加以精化。

每个函数的名字都表明其近似精度水平，注释则描述它相对常规版本的性能。

// 这些函数以不同的精度近似倒数、平方根和 
// 立方根，速度远快于直接使用 1/x、sqrtf(x)、 
// 以及 powf( x, 1.0f/3.0f ) 的方法。 
// 全部需要支持 SSE 的 CPU & OS。
// 与 powf() 方案不同，立方根还能正确处理
// 负数输入。  

#define REALLY_INLINE __forceinline

// Pentium M 上约 34 个时钟周期，powf 约 360 个 
REALLY_INLINE float cuberoot_sse_8bits( float x )
{
	float z;
	static const float three = 3.0f;
	_asm
	{
		mov		eax, x				// x 的位表示
		movss	xmm2, x				// x2: x
		movss	xmm1, three			// x1: 3
		// 用整数运算实现的魔数浮点立方根。
		// 指数被除以三，方式是让 
		// 余数位被挤入规格化尾数的 
		// 高位。 
		mov		ecx, eax			// x 的副本
		and		eax, 0x7FFFFFFF		// x 的指数 & 尾数，偏置 127 表示
		sub     eax, 0x3F800000		// x 的指数 & 尾数，二进制补码表示
		sar     eax, 10				// 
		imul    eax, 341			// 341/1024 ~= .333
		add		eax, 0x3F800000		// 转回偏置 127 表示
		and     eax, 0x7FFFFFFF		// 重新掩码
		and		ecx, 0x80000000		// 原符号与尾数
		or      eax, ecx			// 掩码后的新指数，以及原符号与尾数
		mov		z, eax				// 

		// 用 SSE 精化第一次近似
		movss	xmm0, z				;// x0: z
		movss	xmm3, xmm0			;// x3: z
		mulss	xmm3, xmm0			;// x3: z*z
		movss	xmm4, xmm3			;// x4: z*z
		mulss   xmm3, xmm1			;// x3: 3*z*z
		rcpss	xmm3, xmm3			;// x3: ~ 1/(3*z*z)
		mulss   xmm4, xmm0			;// x4: z*z*z
		subss	xmm4, xmm2			;// x4: z*z*z-x
		mulss	xmm4, xmm3			;// x4: (z*z*z-x) / (3*z*z)
		subss	xmm0, xmm4			;// x0：z' 精度约 0.3% 以内
		movss	z, xmm0
	}

	return z;
}

// Pentium M 上约 60 个时钟周期，powf 约 360 个
REALLY_INLINE float cuberoot_sse_16bits( float x )
{
	float z;
	static const float three = 3.0f;
	_asm
	{
		mov		eax, x				// x 的位表示
		movss	xmm2, x				// x2: x
		movss	xmm1, three			// x1: 3
		// 用整数运算实现的魔数浮点立方根。
		// 指数被除以三，方式是让 
		// 余数位被挤入规格化尾数的 
		// 高位。 
		mov		ecx, eax			// x 的副本
		and		eax, 0x7FFFFFFF		// x 的指数 & 尾数，偏置 127 表示
		sub     eax, 0x3F800000		// x 的指数 & 尾数，二进制补码表示
		sar     eax, 10				// 
		imul    eax, 341			// 341/1024 ~= .333
		add		eax, 0x3F800000		// 转回偏置 127 表示
		and     eax, 0x7FFFFFFF		// 重新掩码
		and		ecx, 0x80000000		// 原符号与尾数
		or      eax, ecx			// 掩码后的新指数，以及原符号与尾数
		mov		z, eax				// 

		// 用 SSE 精化第一次近似
		movss	xmm0, z				;// x0: z
		movss	xmm3, xmm0			;// x3: z
		mulss	xmm3, xmm0			;// x3: z*z
		movss	xmm4, xmm3			;// x4: z*z
		mulss   xmm3, xmm1			;// x3: 3*z*z
		rcpss	xmm3, xmm3			;// x3: ~ 1/(3*z*z)
		mulss   xmm4, xmm0			;// x4: z*z*z
		subss	xmm4, xmm2			;// x4: z*z*z-x
		mulss	xmm4, xmm3			;// x4: (z*z*z-x) / (3*z*z)
		subss	xmm0, xmm4			;// x0：z' 精度约 0.3% 以内

		movss	xmm3, xmm0			;// x3: z
		mulss	xmm3, xmm0			;// x3: z*z
		movss	xmm4, xmm3			;// x4: z*z
		mulss   xmm3, xmm1			;// x3: 3*z*z
		rcpss	xmm3, xmm3			;// x3: ~ 1/(3*z*z)
		mulss   xmm4, xmm0			;// x4: z*z*z
		subss	xmm4, xmm2			;// x4: z*z*z-x
		mulss	xmm4, xmm3			;// x4: (z*z*z-x) / (3*z*z)
		subss	xmm0, xmm4			;// x0：z'' 精度约 0.001% 以内

		movss	z, xmm0
	}

	return z;
}

// Pentium M 上约 6 个时钟周期，单精度 sqrtf 约 24 个
REALLY_INLINE float squareroot_sse_11bits( float x )
{
	float z;
	_asm
	{
		rsqrtss xmm0, x
		rcpss	xmm0, xmm0
		movss	z, xmm0			// z ~= sqrt(x) to 0.038%
	}
	return z;
}

// Pentium M 上约 19 个时钟周期，单精度 sqrtf 约 24 个
REALLY_INLINE float squareroot_sse_22bits( float x )
{
	static float half = 0.5f;
	float z;
	_asm
	{
 		movss	xmm1, x			// x1: x 
		rsqrtss xmm2, xmm1		// x2: ~1/sqrt(x) = 1/z
		rcpss	xmm0, xmm2		// x0: z == ~sqrt(x) to 0.05%

		movss	xmm4, xmm0		// x4: z
		movss	xmm3, half
 		mulss	xmm4, xmm4		// x4: z*z
		mulss	xmm2, xmm3		// x2: 1 / 2z
		subss	xmm4, xmm1		// x4: z*z-x
		mulss	xmm4, xmm2		// x4: (z*z-x)/2z
		subss	xmm0, xmm4		// x0: z' to 0.000015%

		movss	z, xmm0		   
	}
	return z;
}

// Pentium M 上约 12 个时钟周期，单精度除法约 16 个
REALLY_INLINE float reciprocal_sse_22bits( float x )
{
	float z;
	_asm
	{
		rcpss	xmm0, x			// x0: z ~= 1/x
		movss	xmm2, x			// x2: x
		movss	xmm1, xmm0		// x1: z ~= 1/x
		addss	xmm0, xmm0		// x0: 2z
		mulss	xmm1, xmm1		// x1: z^2
		mulss	xmm1, xmm2		// x1: xz^2
		subss	xmm0, xmm1		// x0: z' ~= 1/x to 0.000012%

		movss	z, xmm0		   
	}
	return z;
}
