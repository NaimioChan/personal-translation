需要支持 RDTSC 的 CPU

// 使用 rdtsc 的分块处理基准测试代码
// 适合测量 DSP 分块处理相关性能
// （基于 Intel 的论文）
// 64 位精度
// VeryUglyCode(tm)，作者 Arguru

// 全局变量
UINT time,time_low,time_high;

// 在进入循环等之前调用此函数
void bpb_start()
{
	// 读取时间戳到 EAX
	__asm rdtsc;
	__asm mov time_low,eax;
	__asm mov time_high,edx;
}

// 在循环结束后立即调用下面的函数
// 返回每个采样平均消耗的周期数
UINT bpb_finish(UINT const num_samples)
{
	__asm rdtsc 
	__asm sub eax,time_low;
	__asm sub edx,time_high;
	__asm div num_samples;
	__asm mov time,eax;
	return time;
}

__asm sub eax,time_low
