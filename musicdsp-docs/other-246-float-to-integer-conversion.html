浮点转整数

  • 作者或来源：Peter Schoffhauzer
  • 创建时间：2007-02-19 10:03:07

fld x/fistp i 指令可用于把浮点数快速转换为整数。默认情况下，数字会舍入到最近的整数。然而，有时我们需要的不是这个。

FPU 控制字的第 12 与第 11 位决定 FPU 的舍入方式。四种舍入状态是：

00 = 就近舍入（默认）
01 = 向下舍入（朝 -infinity）
10 = 向上舍入（朝 +infinity）
11 = 向零截断

状态字通过 fldcw/fstcw 指令加载/存储。按需设置舍入模式后，float2int() 函数会一直使用该舍入模式，直到控制模式被重置。ceil() 与 floor() 函数对每条指令都设置舍入模式，非常慢。因此，在处理一个块之前先设置舍入模式（向下或向上），再使用 float2int()，要快得多。

参考：Raymond Filiatreault 的 SIMPLY FPU
http://www.website.masmforum.com/tutorials/fptute/index.html

MSVC（大概其他编译器也是）在 <float.h> 中定义了修改 FPU 控制字的函数：_control87/_controlfp。等效指令为：

_controlfp(_RC_CHOP, _MCW_RC); // 设置舍入模式为截断
_controlfp(_RC_UP _MCW_RC); // 设置舍入模式为向上（ceil）
_controlfp(_RC_DOWN, _MCW_RC); // 设置舍入模式为向下（floor）
_controlfp(_RC_NEAR, _MCW_RC); // 设置舍入模式为就近（默认）

注意 FPU 舍入模式也会影响其他计算，所以同样的代码可能给出不同结果。

另外，在支持 SSE 指令的平台上（大概所有现代 CPU 都支持），可以用 cvtss2si、cvttss2si、cvtps2pi 或 cvttps2pi 指令把单精度浮点数截断或舍入为整数。这里不详细讨论，但我提供了 truncateSSE 函数，它总是把单精度浮点数截断为整数，无论当前舍入模式如何。

（另外我认为 SSE 的舍入模式可能与 FPU 控制字中设置的舍入模式不同，但我至今没有测试过。）

#ifndef __FPU_ROUNDING_H_
#define __FPU_ROUNDING_H_

static short control_word;
static short control_word2;

// 按当前舍入模式将浮点数舍入为整数 
inline int float2int(float x) {
	int i;
	__asm {
		fld x
		fistp i
	}
	return i;
}

// 设置舍入模式为就近舍入
inline void set_round2near() {
	__asm  	{
		fstcw	control_word		// 保存 FPU 控制字 
		mov 	dx, word ptr [control_word]
 		and		dx,0xf9ff			// 就近舍入（默认）
		mov		control_word2, dx
		fldcw	control_word2		// 加载修改后的控制字 
	}
}

// 设置舍入模式为向下舍入
inline void set_round_down() {
	__asm  {
		fstcw	control_word		// 保存 FPU 控制字 
		mov 	dx, word ptr [control_word]
 		or		dx,0x0400			// 向 -inf 舍入
		and		dx,0xf7ff			
		mov		control_word2, dx
		fldcw	control_word2		// 加载修改后的控制字 
	}
}

// 设置舍入模式为向上舍入
inline void set_round_up() {
	__asm  {
		fstcw	control_word		// 保存 FPU 控制字 
		mov 	dx, word ptr [control_word]
 		or		dx,0x0800			// 向 +inf 舍入
		and		dx,0xfbff			
		mov		control_word2, dx
		fldcw	control_word2		// 加载修改后的控制字 
	}
}

// 设置舍入模式为截断
inline void set_truncate() {
	__asm  {
		fstcw	control_word		// 保存 FPU 控制字 
		mov 	dx, word ptr [control_word]
 		or		dx,0x0c00			// 舍入：截断
		mov		control_word2, dx
		fldcw	control_word2		// 加载修改后的控制字 
	}
}

// 恢复原始舍入模式
inline void restore_cw() {
	__asm fldcw	control_word
}

// 使用 SSE 截断为整数
inline long truncateSSE(float x) {
	__asm cvttss2si eax,x
}

#endif

I've seen a lot of talk about using the function lrintf() when converting from float to int, regarding bypassing some 'slow' opcodes in what standard compilers put in for something like:

int myint = (int) myfloat;

in otherwords the following code would be faster:
int myint = lrintf(myfloat);

this lrintf function is available on GNU C/C++
