Pdfsm

确定性有限状态机

ListPattern : Pattern : AbstractFunction : Object

参见：Pfsm

说明

Pdfsm 是一个带信号输入的确定性有限状态机（由 ccos 编写）。

类方法

Pdfsm.new(list, startState: 0, repeats: 1)

参数：

list

一个列表，由用于给出输入信号以确定状态转换的流，以及随后针对每个状态的字典条目组成，这些条目将目标状态和 yield 流映射到对应的输入信号。

startState

起始状态的整数索引。

repeats

一个整数，表示模式应循环的次数。当 signal stream 结束，或为信号值的目标状态给出 nil 时，一次循环结束，详见下文。

实例方法

示例

列表结构：

[

信号流

可以是任何可作为关联集合键的对象的流。整数、符号等……每次重复时都会对其调用 asStream。

states

states 应该是 IdentityDictionary、Event 或其他关联集合的实例。

]
列表语法：

[

信号流，
( // state 0,
信号值：[目标状态，返回流或模式]，
信号值：[目标状态，返回流或模式]
),
... // 状态 1 ... N
]

可以给出任意数量的状态，并按给出的顺序进行索引。

如果 fsm 处于状态 x 并接收到一个信号值 y，它会在为 x 提供的状态字典中查找 y；如果没有 y 条目，则查找 \default 条目并使用它。

然后将下一个状态设置为 目标状态，生成的流由返回流或模式给出。除非目标状态被指定为 nil，或者为某个目标状态指定了但你未提供对应的字典——在两种情况下，当前循环都会结束，并执行所有剩余的重复。如果某个信号没有给出信号值，且未提供 \default，则会出现运行时错误。

(
    p = Pdfsm(
        [
            ( // 状态 0
            Pseq([\foo, \bar], 2), // foobar 信号
                \foo : [1, Pseq([0, 1], 2)]
            ),
            ( // 状态 1
                \bar : [0, 3]
            )
        ],
        0,
        2
    ).asStream;

    11.do({ p.next.postln });
)

SynthDef(\help_Pdfsm1,
    { |out = 0, freq = 440, sustain = 0.05|
        var env;
        env = EnvGen.kr(Env.perc(0.01, sustain, 0.2), doneAction: Done.freeSelf);
        Out.ar(out, SinOsc.ar([freq, freq + 0.1.rand2], 0, env))
    }).add;
)

(
var p;
p = Pdfsm(
    [
        Prand([0, 1, 2], inf),    // signalStream

        IdentityDictionary[    // 状态 0
            0 -> [2, Pseq([67, 68, 69], 2)],
            1 -> [0, 66],
            2 -> [1, 65]
        ],
        IdentityDictionary[    // 状态 1
            1 -> [1, Pseq([69, 68, 67], 2)],
            \default -> [0, 70]
        ],
        IdentityDictionary[
            0 -> [0, 71],
            1 -> [0, 72],
            2 -> [nil]    // signalStream 是无限长的，
                    // 所以 fsm 只因这个 nil 而结束
                    // 2 -> [nil, nil] 也可以
        ]
    ],
    1,                 // startState
    1                 // 重复
).asStream;

Routine({
    var freq;
    while({ (freq = p.next.postln).notNil }, {
        Synth(\help_Pdfsm1, [\freq, freq.midicps]);
        0.1.wait;
    })
}).play;
)

SynthDef(\help_Pdfsm2,
    { |out, freq, gate = 1|
        var n = 8, env, osc;
        env = Linen.kr(gate, 0.01, 1, 0.03, 2);
        osc = {
            Mix.fill(n, { |i|
                FSinOsc.ar(freq + Rand(-2.0, 2.0), Rand(0, 0.05pi)) ring4:
                FSinOsc.ar(freq * (i+1));
            })
        }.dup * FSinOsc.kr(Rand(1.5, 4.5), { Rand(-0.1pi, 0.1pi) }.dup, 0.6, env*0.4);
        Out.ar(out, env * osc / (n*4))
}).add;
)

var n = 3, base, penult;

base = [3, 4, 4, 0];

for(1, n, { |i|
    penult = Pbind(\degree, Pshuf(base - (i*5), 2), \dur, Pseq([0.2], 2));
    Pset(
        \instrument, \help_Pdfsm2,
        Pdfsm(
            [
                Pseq([    // signalStream
                    Pn(1, 22 + i),
                    Pn(0, 4),
                    Pn(1, 8),
                    Pseq([
                        Pn(0, 3),
                        Prand([0, 1], 8),
                        Pn(1, 8)
                    ], 3),
                    Pn(2, 2)
                ], 1),
                (    // 状态 0
                    0 : [0, Pbind(\degree, Pseq(base - i, 1), \dur, Pxrand([0.2, 0.3], 4))],
                    1 : [1, Pbind(
                            \degree, Pseq(base.reverse - (i*2), 2),
                            \dur, Pseq([0.2, 0.21], 1)
                            )
                        ],
                    2 :    [2, penult]
                ),
                (    // 状态 1
                    0 : [0, Pbind(\degree, Pshuf(base * i.neg, 8), \dur, Pseq([0.08], 8))],
                    1 : [0, Pbind(\degree, Pseq(base - (i*3), 3+i), \dur, Pseq([0.11], 3+i))],
                    2 : [2, penult]
                ),
                (    // 状态 2
                    \default : [2, Pbind(
                                    \degree, Prand(base - (i*7), 5),
                                    \dur, Prand([0.6, 0.8], 5)
                                )
                            ]
                )
            ],
            i % 2         // startState
        )
    ).play;
})
