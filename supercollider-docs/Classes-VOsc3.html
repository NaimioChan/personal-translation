VOsc3

三个可变波表振荡器。

PureUGen : UGen : AbstractFunction : Object

另请参阅：COsc、Osc、OscN、VOsc

说明

一个波表查找振荡器，可以平滑地扫过多个波表。table 的小数值将在两个相邻波表之间插值。虽然 VOsc 以单一频率循环遍历其波表，但此单元生成器包含三个不同频率的振荡器，混合在一起。

此振荡器至少需要两个填充了波表格式信号的缓冲区。这会将信号预处理为振荡器可以高效使用的形式。所有波表必须分配为相同的大小，且该大小必须是 2 的幂。

使用缓冲区设置你的表

波表可以通过 Buffer: Buffer Fill Commands（例如 Buffer: -sine1）填充缓冲区来生成。设置 wavetable: true 以确保数据格式正确。

同样地，你可以通过发送 \b_gen 消息中的 \sine1、\sine2、\sine3 之一来创建 Buffer 对象（参见 Server Command Reference: Wave Fill Commands 及下方示例），同样记得启用 wavetable 格式。

你可以使用 Buffer: *allocConsecutive 来确保缓冲区编号连续以便进行插值。

使用信号设置你的表

这也可以通过创建一个 Signal 对象并向其发送 Signal: -asWavetable 消息、将其保存到磁盘、然后让服务器从那里加载来实现。

类方法

VOsc3.ar(bufpos, freq1: 110.0, freq2: 220.0, freq3: 440.0, mul: 1.0, add: 0.0)

VOsc3.kr(bufpos, freq1: 110.0, freq2: 220.0, freq3: 440.0, mul: 1.0, add: 0.0)

参数：

bufpos

缓冲区索引。可以在大小相同的相邻波表缓冲区之间连续扫过。

freq1

第 1 个振荡器的频率，单位为赫兹。（非插值控制速率）

freq2

第 2 个振荡器的频率，单位为赫兹。（非插值控制速率）

freq3

第 3 个振荡器的频率，单位为赫兹。（非插值控制速率）

mul

输出将乘以该值。

add

此值将被添加到输出中。

实例方法

示例

// 分配并填充供 VOsc3 使用的缓冲区
(
s.waitForBoot({

    var numBufs = 8;

    // 分配连续缓冲区表
    ~bufs = Buffer.allocConsecutive(numBufs, s, 1024, 1);
    s.sync;

    ~bufs.do({ |buf, i|
        var n, a;

        // 生成谐波振幅数组
        n = (i+1)**2;
        a = Array.fill(n, { |j| ((n-j)/n).squared.round(0.001) });

        // 填充表
        // 参数 '7'：\sine1 波形填充方法的标志
        // 请参阅 Server Command Reference 中的 "Wave Fill Commands"
        s.performList(\sendMsg, \b_gen, buf.bufnum, \sine1, 7, a);

        s.sync;
    });
    "Buffers prepared.".postln;
})
)

// 播放 synth，用 MouseX 索引缓冲区
(
x = SynthDef("help-VOsc", { |out = 0, bufoffset = 0, freq = 240|
    // 鼠标 x 控制波表位置
    var bufindex = MouseX.kr(0, ~bufs.size - 1) + bufoffset;

    Out.ar(out,
        VOsc3.ar(
            bufindex,
            freq + [0, 1],
            freq + [0.37, 1.1],
            freq + [0.43, -0.29],
            0.3
        )
    )
}).play(s, [\out, 0, \bufoffset, ~bufs.first.bufnum, \freq, 240]);
)

// 用新的谐波即时填充缓冲区
(
~bufs.do({ |buf, i|
    var a;
    a = Array.fill(i, 0) ++ [0.5, 1, 0.5];
    s.performList(\sendMsg, \b_gen, buf.bufnum, \sine1, 7, a);
});
)

// 新的谐波……
(
~bufs.do({ |buf, i|
    var a, n;
    n = (i+1)*8;
    a = Array.fill(n, 0);
    (n>>1).do({ |i| a.put(n.rand, 1) });

    s.performList(\sendMsg, \b_gen, buf.bufnum, \sine1, 7, a);
});
)

// 新的谐波……
(
~bufs.do({ |buf, i|
    var a, n;
    n = (i+1)**2;
    a = Array.fill(n, { 1.0.rand2 });

    s.performList(\sendMsg, \b_gen, buf.bufnum, \sine1, 7, a);
});
)

// 新的谐波……
(
var a = Array.fill(64, { 1.0.rand2 });
var n = [1, a.size].interpolate(~bufs.size-1);
~bufs.do({ |buf, i|
    n[i].postln;
    s.performList(\sendMsg, \b_gen, buf.bufnum, \sine1, 7, a.keep(n[i].asInteger).postln);
});
)

// cleanup
(
x.free;
~bufs.do(_.free);
)
