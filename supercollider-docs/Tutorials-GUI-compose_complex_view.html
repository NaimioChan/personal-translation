GUI：使用基本组件组合视图

关于用基本 GUI 组件组合视图的通用示例。

另请参阅：GUI: Create a Window、GUI: Interfacing an instrument with a View

GUI 教程

此帮助文件是 GUI 教程 的一部分。

这是 SuperCollider GUI 系统初学者介绍的第二部分。第一部分介绍创建窗口，而第二部分讨论将 Synth 与 GUI 界面连接。

本教程是关于什么的？

本教程介绍使用 SuperCollider 系统时你可能需要构建的不同界面，并讨论它们的特殊性。如果你需要关于视图组合基础的更详细教程，应先阅读 GUI：将乐器与 View 对接。

SuperCollider 实现了不同的 GUI 组件，允许你组合复杂的软件界面。

这些组件的列表可在文档中查看，方法是选择 Browse > GUI > Views。

你可以把这些组件看作一些基础构建块，让你能够构建出更大的结构。

组合一个带标题和可编辑文本字段的 Slider

这是关于视图组合的最简单示例：一个 Slider、一个 TextField 和一个 StaticText。这里没什么可多说的，代码本身基本就能说明问题。唯一需要记住的是，当编辑时，让 Slider 和 TextField 相应地相互更新。

请注意，此示例是“不安全”的：没有任何机制阻止在 TextField 中输入高数值：任何浮点数都是有效的。在使用 TextField 时，你仍需负责将值裁剪到正确的范围。在 TextField 中输入非有效数字会将值设为 0.0（因为对非浮点的 String 调用 .asFloat 会返回 0.0）。

(
var value = 1;
var displayValue = { |value|
    value.postln;
};

var sliderView = UserView()
.bounds_(Rect(0, 0, 400, 64));
var slider;
var text = StaticText()
.align_(\center)
.string_("Slider Name");
var field;

slider = Slider()
.orientation_(\horizontal)
.action_({
    value = slider.value;
    field.string_(
        value.trunc(0.01).asString);
    displayValue.value(value);
});

field = TextField()
.align_(\center)
.string_(value.asString)
.maxWidth_(64)
.action_({
    value = field.string.asFloat;
    slider.value_(value);
    displayValue.value(value);
});

sliderView.layout_(
    HLayout()
    .add(
        VLayout()
        .add(text)
        .add(slider),
    )
    .add(field)
);

sliderView.front;
)

使用单个 Function 集中管理 Views 操作

此示例是前一示例的延续，解释了为什么 UI 组件不应直接负责它们最终执行的操作。它还简要讨论了焦点和键盘输入响应。

开始使用 SuperCollider 的 GUI 系统时常见的错误之一，是把 View 的 action 函数当作真正负责修改软件的的函数来使用。即使在与该算法关联的 View 只有一个时它能可靠工作，这种做法也会降低代码可读性。当多个 View 绑定到同一个算法时，这几乎总会引发问题，正如在上一个例子中所见。

下一个示例展示了如何将算法集中到单个 Function 中，该函数负责将修改应用到软件，然后相应地更新各 View。在这种上下文中，View 的动作函数只负责将其提供的数据转换为正确的格式，然后调用这个集中函数。

这里，多个视图使用 updateValue 函数来修改值。此函数也会同时更新每个视图。有趣的是，该函数会在值越界时对其进行裁剪：这可以防止用户在使用 TextField 时指定错误的值。无论错误值来源如何，都会被裁剪。

Slider 和 TextField 都只有非常小的动作函数：它们只将值转换为正确的格式，然后调用 updateValue。除了可读性之外，只有在我们必须更改算法时才需要修改 updateValue。View 的 action 函数一旦设置就不再受任何更改的影响。

在这个特定示例中，我还实现了一个 switch 语句，它允许窗口响应键盘输入。由于 Slider 和 TextField 都原生响应键盘输入事件，我们必须操作 View 的焦点来指定事件的上下文。

(
// 指定默认值及其范围
var valueDefault = 440;
var valueMin = 440;
var valueMax = 880;
// 引用值
// 用于键盘事件
var currentValue = valueDefault;

var window = Window("", Rect(100, 100, 400, 400));

var slider = Slider()
.orientation_(\horizontal)
.maxHeight_(64)
.value_(
    valueDefault.linlin(valueMin, valueMax, 0, 1)
);

var textField = TextField()
.string_(valueDefault.asString)
.align_(\center);

// 中心化函数：
var updateValue = { |value|
    // 将值裁剪到正确的范围
    currentValue = value.clip(valueMin, valueMax);

    // 这里是你实现实际控制的地方
    // 例如
    // synth.set(\freq, value);

    // 现在我们可以更新我们的界面

    // 将值调整回滑块的范围
    slider.value_(
        currentValue.linlin(valueMin, valueMax, 0, 1
    ));

    // 将值转换为字符串格式以供 textField 使用
    textField.string_(currentValue.trunc(0.01).asString);
};

// 滑块需要在调用函数之前
// 将其值映射到正确的范围
slider.action_({
    updateValue.value(
        slider.value.linlin(0, 1, valueMin, valueMax)
    );
});

// textField 在调用该函数之前需要将其值
// 转换为浮点格式
textField.action_({
    updateValue.value(textField.string.asFloat);
});

// 点击背景会移除
// slider 和 textField 的焦点，
// 从而可以使用方向键
// 直接控制数值
window.view.mouseDownAction_({
    slider.focus(false);
    textField.focus(false);
});

// 设置窗口视图
// 使其响应箭头
// 键事件
window.view.keyDownAction_({ |view, char, mod, unicode, keycode|
    switch(keycode)
    // 左箭头检测
    { 65361 } { updateValue.value(currentValue * 0.99); }
    // 右箭头检测
    { 65363 } { updateValue.value(currentValue * 1.01); }
    // 向上箭头检测
    { 65362 } { updateValue.value(currentValue * 1.1); }
    // 下箭头检测
    { 65364 } { updateValue.value(currentValue * 0.9); };
});

window.layout_(
    VLayout()
    .add(slider)
    .add(textField)
);

// 让滑块失去焦点，
// 它默认是获得焦点的
window.view.focus(true);

window.front;
)

注意： 还有其他几种将算法与图形用户界面连接的方法。其中最为广泛使用的一种是 Model-View-Controller 设计模式。关于 Model-View-Controller 范式的 SuperCollider 实现，请参见 SimpleController。

创建 Tab 视图

一个提供简单标签系统的接口，类似于网页浏览器中的标签页，允许在多个网页之间切换。

为此，我们可以组合两个简单的构建块：一些 Button，以及一个 StackLayout。

StackLayout 允许你将多个图形上下文层叠在一起，就像把一些图画堆叠在一起那样。你可以选择只显示其中一个，或全部显示；在全部显示时，你可以看到某个视图位于另一个视图之下。这类似于绘图软件中的图层系统。

(
var tabView = UserView();
var stackLayout = StackLayout();
var buttonLayout = HLayout();

var tabNames = [
    "Tab One",
    "Tab Two",
    "Tab Three"
];
var tabs = List(0);

tabNames.do({ |tabName, tabIndex|
    var button = Button()
    .string_(tabName)
    // 将按钮映射到 stackLayout 索引：
    .action_({ stackLayout.index_(tabIndex); });

    // Tabs 在此处创建
    // 因为这只是个示例。
    // 在实际代码中，
    // 你的视图很可能
    // 早已创建好了。
    var tab = UserView()
    .background_(Color.rand);

    buttonLayout.add(button);
    stackLayout.add(tab);
});

// 允许指定起始标签页
stackLayout.index_(0);

tabView.layout_(
    VLayout()
    .add(buttonLayout, 1)
    .add(stackLayout, 9)
);

tabView.front;
)

实现视图最小化器

此示例演示如何切换视图的可见性，就像使用窗口最小化功能一样。

这是可行的，因为布局在其子元素的可见性被修改时会重绘其子元素。

窗口首先被垂直分成两部分：一个主视图，以及一个“包含”隐藏视图的小菜单。此菜单仅在非空时可见。

对于我们的每个子视图（在我们的示例中，是一个 MultiSliderView 和一个 StaticText），我们创建两个 Button：一个用于隐藏视图，位于它所隐藏的视图相同的布局中；另一个位于菜单内，用于将其重新显示出来。它们控制视图的可见性，同时也控制菜单的可见性。

(
var window = Window(
    "Visibility Toggle",
    Rect(100, 100, 600, 400)
);

// 我们的主显示
var mainView = UserView()
.layout_(HLayout());

// 限制最大高度
// 因为它只是一个菜单
var visibilityView = UserView()
.maxHeight_(64)
.layout_(HLayout());

// 我们引用
// 隐藏视图的数量
// 以便在没有视图被隐藏时
// 隐藏“可见性菜单”
var numberOfHiddenView = 0;

// 一个 multiSlider 示例
var multiSliderView;
var multiSliderHideButton;
var multiSliderShowButton;
var multiSlider;

// 一个文本示例
var textView;
var textHideButton;
var textShowButton;
var text;

// 设置 multiSlider 视图
multiSliderView = UserView()
.layout_(VLayout());

multiSlider = MultiSliderView()
.value_([1, 2, 3, 4, 5, 6, 7, 8].reciprocal)
.drawLines_(true)
.elasticMode_(1);

multiSliderHideButton = Button()
.string_("Hide MultiSlider")
.maxHeight_(64);

// 启动时 Show 按钮不可见
multiSliderShowButton = Button()
.visible_(false)
.string_("Show MultiSlider")
.maxHeight_(64);

multiSliderView.layout.add(
    multiSliderHideButton, 1);
multiSliderView.layout.add(
    multiSlider, 9);

// 将子视图插入主视图
mainView.layout.add(multiSliderView, 1);
// 将 'show' 按钮添加到可见性视图
visibilityView.layout.add(multiSliderShowButton);

// 设置文本视图
textView = UserView()
.layout_(VLayout());

text = StaticText()
.string_("Lorem ipsum dolor sit amet, consectetur adipiscing elit. Sed non risus. Suspendisse lectus tortor, dignissim sit amet, adipiscing nec, ultricies sed, dolor. Cras elementum ultrices diam.")
.align_(\center);

textHideButton = Button()
.string_("Hide Text")
.maxHeight_(64);

// 启动时 Show 按钮不可见
textShowButton = Button()
.visible_(false)
.string_("Show Text")
.maxHeight_(64);

textView.layout.add(
    textHideButton, 1);
textView.layout.add(
    text, 9);

// 将子视图插入主视图
mainView.layout.add(textView, 1);
// 将 'show' 按钮添加到可见性视图
visibilityView.layout.add(textShowButton);

// 构建顶层
window.layout_(
    VLayout()
    .add(mainView)
    .add(visibilityView)
);

// 隐藏可见性视图，
// 因为开始时所有内容都是可见的
visibilityView.visible_(false);


// --- 以下是可见性逻辑：---
// 当按下隐藏按钮时
multiSliderHideButton.action_({
    // 隐藏视图
    multiSliderView.visible_(false);
    // 设置 show 按钮可见
    multiSliderShowButton.visible_(true);
    // 增加隐藏视图的数量
    numberOfHiddenView = numberOfHiddenView + 1;
    // 如果恰好隐藏了一个项目，
    // 这意味着我们必须显示之前隐藏的
    // 可见性视图
    if(numberOfHiddenView == 1)
    { visibilityView.visible_(true); };
});

// 现在显示按钮的反向逻辑：
// 当按下显示按钮时
multiSliderShowButton.action_({
    // 显示视图
    multiSliderView.visible_(true);
    // 隐藏显示按钮
    multiSliderShowButton.visible_(false);
    // 减少隐藏视图的数量
    numberOfHiddenView = numberOfHiddenView - 1;
    // 如果没有隐藏项，
    // 隐藏可见性视图
    if(numberOfHiddenView == 0)
    { visibilityView.visible_(false); };
});

// 文本视图同理：
textHideButton.action_({
    textView.visible_(false);
    textShowButton.visible_(true);
    numberOfHiddenView = numberOfHiddenView + 1;
    if(numberOfHiddenView == 1)
    { visibilityView.visible_(true); };
});

textShowButton.action_({
    textView.visible_(true);
    textShowButton.visible_(false);
    numberOfHiddenView = numberOfHiddenView - 1;
    if(numberOfHiddenView == 0)
    { visibilityView.visible_(false); };
});

// Done
window.front;
)
