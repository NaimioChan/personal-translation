PV_Max

最大幅度。

PV_MagMul : PV_ChainUGen : WidthFirstUGen : UGen : AbstractFunction : Object

另请参阅：FFT、IFFT、PV_Add、PV_CopyPhase、PV_MagMul、PV_Min、PV_Mul

说明

输出复制两个输入中幅度最大的频段。

类方法

PV_Max.new(bufferA, bufferB)

来自父类：PV_MagMul

参数：

bufferA

FFT 缓冲区 A。

bufferB

FFT 缓冲区 B。

实例方法

示例

s.boot;
b = Buffer.read(s, ExampleFiles.child);

(
var exBuf;
Dialog.getPaths({ |paths| // 获取第二个声音文件；
    paths.do({ |p| exBuf = Buffer.read(s, p);
        SynthDef("help-max", { |out = 0, soundBufnum1 = 2, soundBufnum2 = 3|
            var inA, chainA, inB, chainB, chain ;
            inA = PlayBuf.ar(1, soundBufnum1, BufRateScale.kr(soundBufnum1), loop: 1);
            inB =  PlayBuf.ar(1, soundBufnum2, BufRateScale.kr(soundBufnum2), loop: 1);
            chainA = FFT(LocalBuf(2048), inA);
            chainB = FFT(LocalBuf(2048), inB);
            chain = PV_Max(chainA, chainB);
            Out.ar(out, 0.1 * IFFT(chain).dup);
        }).play(s, [\soundBufnum1, b.bufnum, \soundBufnum2, exBuf.bufnum]);
    })
}, {
    "cancelled".postln;
});
)

b.free;
