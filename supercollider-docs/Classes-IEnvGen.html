IEnvGen

用于从 Env 轮询值的包络生成器

UGen : AbstractFunction : Object

参见：Env

说明

用于从包络中轮询取值的包络发生器。IEnvGen 从 index 点开始播放断点包络。包络是 Env 类的实例。

类方法

IEnvGen.ar(envelope, index, mul: 1, add: 0)

IEnvGen.kr(envelope, index, mul: 1, add: 0)

参数：

envelope

Env 的一个实例（在 UGen 的整个生命周期内保持不变）

index

在 Env 中访问的一个点

mul

add

未文档化的类方法

IEnvGen.convertEnv(env)

实例方法

示例

(
{
    var env =  Env([0, 0.6, 0.3, 1.0, 0], [0.1, 0.02, 0.4, 1.1], [\lin, \exp, -6, \sin]);
    var envgen = IEnvGen.kr(env, MouseX.kr(0, env.times.sum));
    env.plot;
    SinOsc.ar(envgen * 500 + 440)
}.play;
)

// 用 SinOsc 建立索引……鼠标控制 SinOsc 的振幅
// 使用偏移量，使 SinOsc 的负值也映射到 Env 中
(
{
    var sin = SinOsc.ar(440, 0, MouseX.kr(0, 1));
    // 使用 offset 使 SinOsc 的负值映射进 Env
    var env = Env([-1, -0.7, 0.7, 1], [0.8666, 0.2666, 0.8668], \lin, offset: -1.0);
    IEnvGen.ar(env, sin) * 0.1
}.play;
)

// 用输入信号的 Amplitude 作为索引，控制 SinOsc 的频率（使用 SoundIn）
(
{
    var point = Amplitude.ar(SoundIn.ar(0), 0.01, 0.2);
    // 使用 offset 使 SinOsc 的负值映射进 Env
    var env = Env.xyc([[0, 330, \exp], [0.5, 440, \exp], [1.0, 1760]]);
    SinOsc.ar(IEnvGen.kr(env, point)) * 0.2
}.play;
)
