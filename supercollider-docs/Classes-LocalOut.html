LocalOut

写入合成器本地的总线。

AbstractOut : UGen : AbstractFunction : Object

另请参阅：LocalIn

说明

LocalOut 写入封闭 synth 本地总线。这些总线应由 LocalIn ugen 定义。channelsArray 的通道数必须与 LocalIn 中声明的通道数相同。这些总线类似全局总线，但如果你想实现使用反馈处理环路的自包含效果，会更方便。

警告： 写入 LocalOut 的音频直到下一个周期才会被相应的 LocalIn 读取，即晚一个块大小的采样。因此，在使用 LocalIn 创建延迟时间短于音高阈值（即 < 0.05 秒或 > 20Hz）的反馈延迟，或需要采样精确对齐时，务必将此额外延迟考虑在内。参见下面的谐振器示例。

类方法

LocalOut.ar(channelsArray)

LocalOut.kr(channelsArray)

参数：

channelsArray

用于写出的通道数组或单个输出。一旦 SynthDef 构建完成，就无法更改其大小。

未文档化的类方法

LocalOut.numFixedArgs

示例

(
{
    var source, local;

    source = Decay.ar(Impulse.ar(0.3), 0.1) * WhiteNoise.ar(0.2);
    local = LocalIn.ar(2) + [source, 0]; // 读取反馈，加到源上
    local = DelayN.ar(local, 0.2, 0.2); // 延迟声音

    // 反转声道以产生乒乓效果，并施加衰减因子
    LocalOut.ar(local.reverse * 0.8);

    local
}.play;
)



(
z = SynthDef("tank", { |out|
    var local, in;

    in = Mix.fill(12, {
        Pan2.ar(
            Decay2.ar(Dust.ar(0.05), 0.1, 0.5, 0.1)
                * FSinOsc.ar(IRand(36, 84).midicps).cubed.max(0),
            Rand(-1, 1))
    });
    in = in + Pan2.ar(Decay2.ar(Dust.ar(0.03), 0.04, 0.3) * BrownNoise.ar, 0);

    4.do { in = AllpassN.ar(in, 0.03, { Rand(0.005, 0.02) }.dup, 1) };

    local = LocalIn.ar(2) * 0.98;
    local = OnePole.ar(local, 0.5);

    local = Rotate2.ar(local[0], local[1], 0.23);
    local = AllpassN.ar(local, 0.05, { Rand(0.01, 0.05) }.dup, 2);

    local = DelayN.ar(local, 0.3, [0.19, 0.26]);
    local = AllpassN.ar(local, 0.05, { Rand(0.03, 0.15) }.dup, 2);

    local = LeakDC.ar(local);
    local = local + in;

    LocalOut.ar(local);

    Out.ar(out, local);
}).play;
)



(
z = SynthDef("tape", { |out|
    var local, in, amp;

    in = SoundIn.ar([0, 1]);

    amp = Amplitude.kr(Mix.ar(in));
    in = in * (amp > 0.02); // 噪声门

    local = LocalIn.ar(2);
    local = OnePole.ar(local, 0.4);
    local = OnePole.ar(local, -0.08);

    local = Rotate2.ar(local[0], local[1], 0.2);

    local = DelayN.ar(local, 0.25, 0.25);

    local = LeakDC.ar(local);
    local = ((local + in) * 1.25).softclip;

    LocalOut.ar(local);

    Out.ar(out, local * 0.1);
}).play;
)
