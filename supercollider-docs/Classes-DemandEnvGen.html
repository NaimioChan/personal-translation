DemandEnvGen

需求率包络发生器

UGen : AbstractFunction : Object

参见：Demand、Duty、TDuty

说明

播放由 demand UGen 给出的断点包络轮廓（levels、times、shapes）。当下一个节点到达时，会调用下一个值。

类方法

DemandEnvGen.ar(level, dur, shape: 1, curve: 0, gate: 1.0, reset: 1.0, levelScale: 1.0, levelBias: 0.0, timeScale: 1.0, doneAction: 0)

DemandEnvGen.kr(level, dur, shape: 1, curve: 0, gate: 1.0, reset: 1.0, levelScale: 1.0, levelBias: 0.0, timeScale: 1.0, doneAction: 0)

参数：

level

一个 demand UGen 或任何其他返回电平值的 UGen。如果它结束，则

doneAction 被求值。

dur

一个需求 UGen 或任何其他返回时间值的 UGen。如果它结束，则

doneAction 被求值。

shape

一个 demand UGen 或任意其他 UGen：给定的数字是根据 Env 的形状编号。

curve

一个 demand UGen 或任何其他返回曲线值的 UGen。如果 shape 为 5，则此为曲线因子。

可能的值为：

0

平坦段

1

线性段，默认值

2

自然指数增长与衰减。在这种情况下，所有电平都必须非零且符号相同。

3

正弦 S 形分段。

4

形状像 Welch 窗两侧的正弦段。

一个 Float

所有段的曲率值。

一个 Float 数组

每个段的曲率值。

0

平坦段

1

线性段，默认值

2

自然指数增长与衰减。在这种情况下，所有电平都必须非零且符号相同。

3

正弦 S 形分段。

4

形状像 Welch 窗两侧的正弦段。

一个 Float

所有段的曲率值。

一个 Float 数组

每个段的曲率值。

gate

如果 gate 为 x ≥ 1，则 UGen 运行。

如果 gate 为 0 > x > 1，则 UGen 在下一电平（doneAction）释放。

如果 gate 为 x < 0，则对 UGen 进行采样并保持。

reset

如果 reset 从非正数跨越到正数，UGen 会在下一个层级被重置。如果它 > 1，则立即重置。

levelScale

返回电平缩放值的 demand ugen

levelBias

返回电平偏移值的 demand UGen

timeScale

返回时间缩放值的 demand UGen

doneAction

在 level 或 duration 流结束时求值的 doneAction。详见 Done。

示例

// 具有随机时间的频率包络
(
    {
        var freq;
        freq = DemandEnvGen.ar(
                Dseq([204, 400, 201, 502, 300, 200], inf),
                Drand([1.01, 0.2, 0.1, 2], inf) * MouseY.kr(0.01, 3, 1),
                7 // 三次插值
            );
        SinOsc.ar(freq * [1, 1.01]) * 0.1

    }.play;
)

// 频率调制
(
    {
        var freq, list;
        list = { exprand(200, 1000.0) } ! 32;
        freq = DemandEnvGen.ar(
                { Dseq(list.scramble, inf) } ! 2,
                SampleDur.ir * MouseY.kr(1, 3000, 1),
                5, // 曲线插值
                MouseX.kr(-0.01, -4) // 曲线必须为负值才能实现快速插值。
            );
        SinOsc.ar(freq) * 0.1

    }.play;
)


//  门控
// 屏幕右侧的鼠标 x 坐标切换门控
(
    {
        var freq;
        freq = DemandEnvGen.kr(
                Dwhite(300, 1000, inf).round(100),
                0.1,
                5, 0.3, // curve: 0.3
                MouseX.kr > 0.5,
                1
            );
        SinOsc.ar(freq * [1, 1.21]) * 0.1

    }.play;
)

// 采样与保持（0.5 > gate > 0）
// 屏幕右侧的鼠标 x 切换 gate
// 鼠标 y 缩放频率
(
    {
        var freq;
        freq = DemandEnvGen.kr(
                Dwhite(300, 1000, inf).round(100),
                0.1,
                5, 0.3,
                MouseX.kr > 0.5 + 0.1
            );
        SinOsc.ar(freq * [1, 1.21]) * 0.1

    }.play;
)

// gate
// 鼠标在屏幕右侧切换 gate
// 鼠标按钮执行软重置
(
    {
        var freq;
        freq = DemandEnvGen.kr(
                Dseq([Dseries(400, 200, 5), 500, 800, 530, 4000, 900], 2),
                Dseq([0.2, 0.1, 0.2, 0.3, 0.1], inf),
                Dseq([1, 0, 0, 6, 1, 1, 0, 2], inf), // shapes
                0,
                MouseX.kr > 0.5, // gate
                MouseButton.kr > 0.5, // reset
                doneAction: Done.none
            );
        SinOsc.ar(freq * [1, 1.001]) * 0.1

    }.play;
)


//  gate
// 鼠标在屏幕右侧时切换 sample and hold
// 鼠标按键执行硬重置
(
    {
        var freq;
        freq = DemandEnvGen.kr(
                Dseq([Dseries(400, 200, 5), 500, 800, 530, 4000, 900], 2),
                0.1,
                3, 0,
                MouseX.kr > 0.5 - 0.1, // 门限：采样与保持
                MouseButton.kr > 0.5 * 2, // 硬重置
                doneAction: Done.none
            );
        SinOsc.ar(freq * [1, 1.001]) * 0.1

    }.play;
)





// 带 doneAction 的短序列，线性
(
    {
        var freq;
        freq = DemandEnvGen.kr(
                Dseq([1300, 500, 800, 300, 400], 1),
                0.2,
                1,
                doneAction: Done.freeSelf
            );
        SinOsc.ar(freq * [1, 1.01]) * 0.1

    }.play;
)

// 带 doneAction、step 的短序列
(
    {
        var freq;
        freq = DemandEnvGen.kr(
                Dseq([1300, 500, 800, 300, 400], 1),
                0.2,
                0,
                doneAction: Done.freeSelf
            );
        SinOsc.ar(freq * [1, 1.01]) * 0.1

    }.play;
)



// 线性斜坡
(
    {
        var freq;
        freq = DemandEnvGen.kr(
                Dseq([300, 800], 1),
                1,
                1
            );
        SinOsc.ar(freq * [1, 1.01]) * 0.1

    }.play;
)



// 随机门控：释放。门限低电平 > 0。
// 只有端点保留为释放电平
(
    {
        var freq;
        freq = DemandEnvGen.kr(
                Dseq([500, 800], inf),
                0.03,
                1, 0, // linear
                ToggleFF.kr(Dust.kr(5)) + 0.1 // gate


            );
        SinOsc.ar(freq * [1, 1.01]) * 0.1

    }.play;
)

// 随机门控：采样保持。门控低电平 = 0。
(
    {
        var freq;
        freq = DemandEnvGen.kr(
                Dseq([500, 800, 600], inf),
                0.03,
                1, 0, // linear
                ToggleFF.kr(Dust.kr(5)), // gate
                0 // reset

            );
        SinOsc.ar(freq * [1, 1.01]) * 0.1

    }.play;
)




// lfnoise1
(
    {
            DemandEnvGen.ar(
                Dwhite(-0.1, 0.1, inf),
                SampleDur.ir * MouseY.kr(0.5, 20),
                5,
                -4
            );


    }.play;
)

// lfbrownnoise
(
    {
            DemandEnvGen.ar(
                Dbrown(-0.1, 0.1, 0.1, inf),
                SampleDur.ir * MouseY.kr(1, 100, 1)
            );


    }.play;
)






// 硬同步锯齿波
(
    {

            DemandEnvGen.ar(
                Dseq([Dseries(-0.1, 0.01, 20)], inf),
                SampleDur.ir * MouseY.kr(1, 100, 1),
                1, 0,
                K2A.ar(1),
                Impulse.ar(MouseX.kr(1, SampleRate.ir * MouseX.kr(0.002, 1, 1), 1), 0, 1.5)

            )


    }.scope;
)


// 软同步锯齿波
(
    {

            DemandEnvGen.ar(
                Dseq([Dseries(-0.1, 0.01, 20)], inf),
                SampleDur.ir * MouseY.kr(1, 100, 1),
                1, 0,
                K2A.ar(1),
                Impulse.ar(MouseX.kr(1, SampleRate.ir * MouseX.kr(0.002, 1, 1), 1)) + [0, 0.3]

            )


    }.scope;
)



// 对锯齿波进行硬同步，附带一些随机元素
(
    {

            DemandEnvGen.ar(
                Dseq([Dseries(-0.1, 0.01, 20), Dseries(-0.1, 0.01, 20), Dwhite(-0.1, 0.1, 5)], inf),
                SampleDur.ir * MouseY.kr(1, 100, 1),
                3, 0,
                1,
                Impulse.ar(MouseX.kr(1, SampleRate.ir * MouseX.kr(0.002, 1, 1), 1), 0, 1.5)

            )


    }.scope;
)

// 对锯齿波进行软同步，附带一些随机元素
(
    {

            DemandEnvGen.ar(
                Dseq([Dseries(-0.1, 0.01, 20), Dseries(-0.1, 0.01, 20), Dwhite(-0.1, 0.1, 5)], inf),
                SampleDur.ir * MouseY.kr(1, 100, 1),
                1, 0, // 线性插值
                1,
                Impulse.ar(MouseX.kr(1, SampleRate.ir * MouseX.kr(0.002, 1, 1), 1))

            )


    }.scope;
)


// 多通道扩展
// 鼠标在屏幕右侧时切换 gate
// 鼠标 y 控制速度

(
    {
        var freq;
        freq = DemandEnvGen.kr(
                { Dseq([300, 800, Drand([1000, 460, 300], 1), 400], inf) + 3.0.rand } ! 2,
                MouseY.kr(0.001, 2, 1),
                5, -4,
                MouseX.kr > 0.5
            );
        SinOsc.ar(freq) * 0.1

    }.play;
)
