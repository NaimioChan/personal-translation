TDuty

将结果作为来自需求速率 UGen 的触发。

Duty : UGen : AbstractFunction : Object

另请参阅：Demand、Duty

说明

列表中的每个 UGen 都会被要求提供一个值，并根据一个持续时间值流将其作为触发器输出。列表中的单元生成器应为 'demand' 速率。

当 reset 输入有触发时，列表中的 demand rate UGen 和 duration 会被重置。reset 输入也可以是一个 demand UGen，提供重置时间的流。

类方法

TDuty.ar(dur: 1.0, reset: 0.0, level: 1.0, doneAction: 0, gapFirst: 0)

TDuty.kr(dur: 1.0, reset: 0.0, level: 1.0, doneAction: 0, gapFirst: 0)

参数：

dur

时间值。可以是需求 UGen 或任何信号。下一个触发值在持续时间之后获取。

reset

触发或重置时间值。触发时重置 UGen 列表和持续时间 UGen。重置输入也可以是一个需求 UGen，提供一个重置时间流。

level

提供输出值的 Demand UGen。

doneAction

当持续时间流结束时求值的 doneAction。参见

Done 以获取更多细节。

gapFirst

为 0（默认值）时，UGen 会立即进行第一次电平轮询，然后等待第一个持续时间值。为 1 时，UGen 最初轮询第一个持续时间值，等待该持续时间，然后轮询第一个电平（同时轮询下一个持续时间值）。

实例方法

示例

// 示例

s.boot;


// 播放一个小节奏

{ TDuty.ar(Dseq([0.1, 0.2, 0.4, 0.3], inf)) }.play; // 将 demand ugen 作为时长



// 振幅变化
(
    {
        var trig;
        trig = TDuty.ar(
                Dseq([0.1, 0.2, 0.4, 0.3], inf), // 将 demand ugen 作为时长
                0,
                Dseq([0.1, 0.4, 0.01, 0.5, 1.0], inf) // 将 demand ugen 用作振幅
            );
        Ringz.ar(trig, 1000, 0.1)

    }.play;
)

(
    {
        var trig;
        trig = TDuty.ar(
                MouseX.kr(0.001, 2, 1), // 使用控制率 ugen 作为时值
                0,
                Dseq([0.1, 0.4, 0.01, 0.5, 1.0], inf)
            );
        Ringz.ar(trig, 1000, 0.1)

    }.play;
)




// 将 demand ugen 用作音频振荡器

(
    {
        var a, trig, n = 5, m = 64;
        a = {
            var x;
            x = { 0.2.rand2 } ! m;
            x = x ++ ({  Drand({ 0.2.rand2 } ! n) } ! m.rand);
            Dseq(x.scramble, inf)
        } ! n;
        trig = TDuty.ar(
                MouseX.kr(1, 2048, 1) * SampleDur.ir * [1, 1.02],
                0,
                Dswitch1(a, MouseY.kr(0, n-1))
            );
        Ringz.ar(trig, 1000, 0.01)

    }.play;
)


// 单个脉冲

(
SynthDef("delta_demand", { |amp = 0.5, out|
    OffsetOut.ar(out,
        TDuty.ar(Dseq([0]), 0, amp, 2)
    )
}).add;
)

fork { 10.do { s.sendBundle(0.2, ["/s_new", "delta_demand", -1]); 1.0.rand.wait } };


// 脉冲链
(
SynthDef("delta_demand2", {
    OffsetOut.ar(0,
        TDuty.ar(Dgeom(0.05, 0.9, 20), 0, 0.5, 2)
    )
}).add;
)

fork { 10.do { s.sendBundle(0.2, ["/s_new", "delta_demand2", -1]); 1.0.rand.wait } };



// 多通道扩展

(
    {
        var t;
        t = TDuty.ar(
                Drand([Dgeom(0.1, 0.8, 20), 1, 2], inf) ! 2,
                0,
                [Drand({ 1.0.rand } ! 8, inf), Dseq({ 1.0.rand } ! 8, inf)] * 2
            );
        x = Ringz.ar(t, [400, 700], 0.1) * 0.1;

    }.play;
)
