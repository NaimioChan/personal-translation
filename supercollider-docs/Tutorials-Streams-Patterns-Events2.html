Patterns

通常我们希望从单个流规格创建多个流。模式正是从单个规格创建多个 Stream 的方式，就像饼干模具一样。模式可以是任何通过响应 asStream 消息创建 Stream 的对象。同样，类 Object 中有一个 asStream 的默认实现，它只是将接收者作为其自身的流返回。因此，任何对象默认都是一个模式，当收到 asStream 消息时，它会将自身作为流返回。

(
a = 7.asStream;
a.postln;
a.next.postln;
)

模式及其子类

有一个名为 Pattern 的类，为模式这一概念提供了更多功能。

Pfunc 是一种返回 FuncStream 的模式。其提供的函数参数与提供给 FuncStream 的相同。

(
var a, b;
a = Pfunc.new({ #[1, 2, 3, 4].choose });
b = a.asStream;            // 从模式创建一个流
5.do({ b.next.postln; });    // 从流中打印 5 个值
)

Prout 是一个返回 Routine 的模式。传递给它的函数参数与传递给 Routine 的相同。

(
var a, b, c;
a = Prout.new({
        3.do({ arg i; 3.rand.yield; })
    });
// 从模式生成两个流
b = a.asStream;
c = a.asStream;
4.do({ b.next.postln; });    // 打印第一个流中的 4 个值
4.do({ c.next.postln; });    // 从第二个流打印 4 个值
)

Pseries 是一个生成等差数列的模式。

(
var a, b;
a = Pseries.new(10, 3, 8);    // 流从 10 开始，步长为 3，长度为 8
b = a.asStream;
9.do({ b.next.postln; });    // 从流中打印 9 个值
)

Pgeom 是一个生成几何级数的模式。

(
var a, b;
a = Pgeom.new(10, 3, 8);    // 流从 10 开始，以因子 3 步进，长度为 8
b = a.asStream;
9.do({ b.next.postln; });    // 从流中打印 9 个值
)

Pattern 上的数学运算

模式对数学运算符的响应也是返回模式，这些模式对 asStream 的响应是返回经过相应修改的流。

对模式应用一元运算符

(
var a, b, c;
a = Pseries.new(0,1,10);    // a 是一个流从 0 计数到 9 的模式
b = a.squared;            // 模式 b 是模式 a 的平方
c = b.asStream;
12.do({ c.next.postln; });
)

对模式使用二元运算符

(
var a, b, c;
a = Pseries.new(0,1,10);    // a 是一个流从 0 计数到 9 的模式
b = a + 100;            // 向模式 a 添加一个常量值
c = b.asStream;
12.do({ c.next.postln; });
)

对模式进行过滤操作

模式也会响应 collect、select 和 reject 消息，返回一个新的 Pattern。

collect 消息返回一个 Pattern，其 Stream 会被一个函数修改，方式与发送给 Collection 的 collect 消息返回一个修改后的 Collection 相同。

(
var a, b, c;
// a 是一个流从 0 计数到 9 的模式
a = Pseries.new(0,1,10);
// b 是一个 pattern，其 stream 为偶数值加 100
b = a.collect({ arg item; if (item.even, { item + 100 },{ item }); });
c = b.asStream;
6.do({ c.next.postln; });
)

select 消息创建一个模式，其流只传递用户提供的函数返回 true 的项。

(
var a, b, c;
// a 是一个流从 0 计数到 9 的模式
a = Pseries.new(0,1,10);
// b 是一个 pattern，其 stream 只返回奇数值
b = a.select({ arg item; item.odd; });
c = b.asStream;
6.do({ c.next.postln; });
)

reject 消息创建一个模式，其流只传递由用户提供的函数返回 false 的项。

(
var a, b, c;
// a 是一个流从 0 计数到 9 的模式
a = Pseries.new(0,1,10);
// b 是一个模式，其流只返回非奇数值
b = a.reject({ arg item; item.odd; });
c = b.asStream;
6.do({ c.next.postln; });
)

用模式创作音乐

这是第 1 部分所给示例的一个变体，它使用 Pattern 创建随机旋律流的两个实例。

(
    s = Server.local;
    SynthDef(\help_SPE2, { arg i_out=0, sustain=1, freq;
        var out;
        out = RLPF.ar(
            LFSaw.ar( freq ),
            LFNoise1.kr(1, 36, 110).midicps,
            0.1
        ) * EnvGen.kr( Env.perc, levelScale: 0.3,
            timeScale: sustain, doneAction: Done.freeSelf );
        //out = [out, DelayN.ar(out, 0.04, 0.04) ];
        4.do({ out = AllpassN.ar(out, 0.05, [0.05.rand, 0.05.rand], 4) });
        Out.ar( i_out, out );
    }).send(s);
)
(
// 将流视为音高序列
    var pattern, streams, dur, durDiff;
    dur = 1/7;
    durDiff = 3;
    pattern = Prout.new({
        loop({
            if (0.5.coin, {
                #[ 24,31,36,43,48,55 ].do({ arg fifth; fifth.yield });
            });
            rrand(2,5).do({
                // 变化的琶音
                60.yield;
                #[63,65].choose.yield;
                67.yield;
                #[70,72,74].choose.yield;
            });
            // 随机高音旋律
            rrand(3,9).do({ #[74,75,77,79,81].choose.yield });
        });
    });
    streams = [
        (pattern - Pfunc.new({ #[12, 7, 7, 0].choose })).midicps.asStream,
        pattern.midicps.asStream
    ];
    Routine({
        loop({
            Synth( \help_SPE2, [ \freq, streams.at(0).next, \sustain, dur * durDiff ] );
            durDiff.do({
                Synth( \help_SPE2, [ \freq, streams.at(1).next, \sustain, dur ] );
                dur.wait;
            });
        })
    }).play
)
