LocalBuf

分配一个局部于该 synth 的缓冲区

WidthFirstUGen : UGen : AbstractFunction : Object

另请参阅：Buffer、SetBuf、ClearBuf

类方法

LocalBuf.new(numFrames: 1, numChannels: 1)

分配一个本地缓冲区

参数：

numFrames

帧数（默认：1）

numChannels

多通道缓冲区的通道数（默认：1）

返回值：

一个新缓冲区——该 ugen 输出其缓冲区编号，因此可用于任何需要缓冲区编号输入的其他 ugen 中。

LocalBuf.newFrom(list)

从给定的值列表分配一个新的缓冲区

参数：

list

当 numChannels > 1 时，该列表可以是二维的。随后会通过展平将其重塑为缓冲区的当前格式。

返回值：

一个新缓冲区

讨论：

由于 newFrom 由 as 消息调用，因此可以将数组转换为 LocalBuf：

[1, 2, 3].as(LocalBuf)

实例方法

.set(values, offset: 0)

用值列表设置缓冲区槽。

讨论：

如果列表小于 numFrames，则只会设置缓冲区的一部分。当 numChannels > 1 时，列表可以是二维的。offset 是起始索引（默认值：0）

警告： SynthDef 在一个定义中最多允许 65536 (2**16) 个唯一常量值。一个由不同值组成的超大数组可能会破坏 SynthDef 的二进制格式。如果你需要一个用信号数据预初始化的大型缓冲区，强烈建议改用 Buffer。

如果缓冲区很大但包含较少数量的唯一值，例如 Array.fill(88200, { #[0, 0.25, 0.5, 0.75, 1.0].choose })，这没有问题。SynthDef 会为常量列表压缩大型数组。

SynthDef(\bigLocalBuf, {
    LocalBuf(88200).set(Array.fill(88200, { #[0, 0.25, 0.5, 0.75, 1.0].choose }))
}).add;

SynthDescLib.at(\bigLocalBuf).constants;
// prints: FloatArray[1, 88200, 0, 0.75, 0.5, 0.25]

.clear

将缓冲区槽位设为零。

讨论：

当随机访问缓冲区槽（例如使用 BufRd）或避免覆盖它们时，这一点很重要。对于较大的缓冲区，Clear 不是一种高效的实时操作，因此只应在确实需要时使用——但那时它又是必不可少的：每个新的 synth 都会“创建”一个 LocalBuf，并且它可能会重用旧的空间。因此，如果较早的 synth 已经结束，这部分内存可能与新 synth 的相同。

示例

// 示例：FFT

(
{
var in, chain;
    in = WhiteNoise.ar(0.1.dup);
    chain = FFT({ LocalBuf(2048, 1) }.dup, in);
    chain = PV_BrickWall(chain, SinOsc.kr([0.1, 0.11]));
    IFFT(chain) // 逆 FFT
}.play;
)

// 生成一些基于 FFT 的合成器：
(
SynthDef(\fftgrain, { |out, sustain = 1, rate = 0.2|
    var in, chain;
    in = WhiteNoise.ar(0.1).dup;
    chain = FFT({ LocalBuf(128, 1) }.dup, in);
    chain = PV_BrickWall(chain,
        SinOsc.kr(rate * XLine.kr(1, 15 * [1, 1.6], sustain), Rand(0, pi))
    );
    Out.ar(out, IFFT(chain) * XLine.kr(1, 0.001, sustain, doneAction: Done.freeSelf)) // 逆 FFT
}).add;
)

(
Pbind(
    \instrument, \fftgrain,
    \rate, Pwhite().linexp(0, 1, 0.01, 300),
    \legato, Pwhite(1, 3.0, inf),
    \dur, Prand([0.2, 1, 1.2], inf)
).play
)

// IndexL
(
{
    var buf = LocalBuf.newFrom((0..5).scramble);
    var freq = IndexL.kr(buf, MouseX.kr(0, BufFrames.kr(buf))).poll * 100 + 40;
    Saw.ar(freq * [1, 1.1]) * 0.1
}.play;
)

// DetectIndex
(
{
    var buf1 = LocalBuf.newFrom((0..5).scramble);
    var buf2 = LocalBuf.newFrom((0..5).scramble - 1);
    var buf3 = LocalBuf.newFrom((0..5).scramble + 1);
    var index = DetectIndex.kr([buf1, buf2], SinOsc.kr([0.85, 0.8], 0, 6).trunc).poll;
    var freq = IndexL.kr([buf2, buf3], index).poll * 40 + 40;
    Saw.ar(freq) * 0.1
}.play;
)


// DegreeToKey
// 调式空间
// 鼠标 x 控制多利亚调式中的离散音高
(
play({
    var mix;

    mix =

    // 主音
    SinOsc.ar(
        (
            DegreeToKey.kr(
                [0, 2, 3.2, 5, 7, 9, 10].as(LocalBuf),
                MouseX.kr(0, 15),        // 鼠标索引到音阶
                12,                    // 每八度 12 个音
                1,                    // mul = 1
                72                    // 偏移 72 个音符
            ).poll
            + LFNoise1.kr([3, 3], 0.04)    // 添加一些低频立体声失谐
        ).midicps,                        // 将 MIDI 音符转换为赫兹
        0,
        0.1)

    // 持续音五度
    + RLPF.ar(LFPulse.ar([48, 55].midicps, 0.15),
        SinOsc.kr(0.1, 0, 10, 72).midicps, 0.1, 0.1);

    // 添加一些 70 年代欧洲太空摇滚回声
    CombN.ar(mix, 0.31, 0.31, 2, 1, mix)
})
)
