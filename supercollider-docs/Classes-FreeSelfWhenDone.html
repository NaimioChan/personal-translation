FreeSelfWhenDone

当 UGen 结束时释放所属的 synth

UGen : AbstractFunction : Object

另请参阅：Done、PauseSelfWhenDone、Done

说明

某些 UGen 在播放完成时会设置一个“done”标志。当该标志设为 true 时，FreeSelfWhenDone 将释放所属的 synth。

有关这些 UGen 的完整列表，请参见 Done。

请注意，这些 UGen 中有许多具有 doneActions，这是实现相同功能的另一种方式。详情请参阅 Done。

注意： 对带有完成标志的 UGen 使用二元运算时必须小心，因为这会返回一个 BinaryOpUGen，从而阻止访问完成标志。参见下面的示例。

类方法

FreeSelfWhenDone.kr(src)

参数：

src

要检查是否完成的 UGen。

实例方法

示例

s.boot;

// 简单示例
(
{ var env;
env = Line.kr(0, 1, 1);
FreeSelfWhenDone.kr(env); // 在行末释放合成器
SinOsc.ar(200, 0, 0.5) * env
}.play;
)

// 前一个示例可以工作，因为 FreeSelfWhenDone 作用于 Line
// 这个版本则不行
(
{ var env, output;
env = Line.kr(0, 1, 1);
output = SinOsc.ar(200, 0, 0.5) * env;
output.postln; // 输出是一个 BinaryOpUGen，它没有 'done' 标志
FreeSelfWhenDone.kr(output); // 永远不会完成
output
}.play;
)

// 录制四秒
b = Buffer.alloc(s, 44100 * 4.0, 1);
(
SynthDef("help-RecordBuf", { |out = 0, bufnum = 0|
    var formant, recbuf;
    formant = Formant.ar(XLine.kr(400, 1000, 4), 2000, 800, 0.125);
    recbuf = RecordBuf.ar(formant, bufnum, recLevel: Line.kr(1, 1), loop: 0);
    // RecordBuf 不会循环，因此你可以检查它的 'done' 状态
    FreeSelfWhenDone.kr(recbuf);
}).play(s, [\out, 0, \bufnum, b]);
)

// 播放回来
(
SynthDef("help-RecordBuf play", { |out = 0, bufnum = 0|
    var playbuf;
    playbuf = PlayBuf.ar(1, bufnum);
    FreeSelfWhenDone.kr(playbuf); // 当 PlayBuf 结束时释放 synth
    Out.ar(out, playbuf);
}).play(s, [\out, 0, \bufnum, b]);
)
