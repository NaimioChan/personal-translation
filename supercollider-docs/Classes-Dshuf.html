Dshuf

需求率随机序列生成器

ListDUGen : DUGen : UGen : AbstractFunction : Object

类方法

Dshuf.new(list, repeats: 1)

来自父类：ListDUGen

参数：

list

值或其他 ugens 的数组

repeats

重复次数

讨论：

结构相关：Pshuf

实例方法

示例

(
{
    var a, freq, trig;
    a = Dseq([Dshuf([1, 3, 2, 7, 8.5], 3)], inf);
    trig = Impulse.kr(MouseX.kr(1, 40, 1));
    freq = Demand.kr(trig, 0, a) * 30 + 340;
    SinOsc.ar(freq) * 0.1

}.play;
)

// 音频速率
(
{
    var a, freq, trig;
    a = Dseq([Dshuf({ 10.rand } ! 81, 5)], inf).poll;
    trig = Impulse.ar(MouseX.kr(1, 10000, 1));
    freq = Demand.ar(trig, 0, a) * 30 + 340;
    SinOsc.ar(freq) * 0.1

}.play;
)

// 嵌入结构
(
{
    var a, freq, trig;
    a = Dseq([Dshuf([Drand([1, 2, 3], 1), 3, Drand([20, 23, 56], 1), 7, 8.5], 8)], inf);
    trig = Impulse.kr(MouseX.kr(1, 40, 1));
    freq = Demand.kr(trig, 0, a) * 30 + 340;
    SinOsc.ar(freq) * 0.1

}.play;
)
