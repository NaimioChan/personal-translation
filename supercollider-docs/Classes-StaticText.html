StaticText

显示不可编辑文本的视图

说明

显示不可编辑文本的视图

类方法

实例方法

Data

.string

视图显示的文本。

参数：

一个 String。

.object

如果 -setBoth 为 true，则设置此变量时也会将 -string 设置为按 as String 解释的值。

参数：

任意对象，通常是以字符串形式显示时有意义的对象，例如 Float。

.setBoth

一个变量，表示设置 -object 时是否也会设置 -string。

参数：

一个布尔值。

Appearance

.align

显示文本的对齐方式。可能的值参见 Alignment。

.font

用于显示文本的字体。

参数：

一个 Font。

.stringColor

用于显示文本的颜色。

参数：

一个 Color。

.background

设置此变量会用给定的颜色为文本下方视图占据的整个区域着色。

参数：

一个 Color。

示例

基本示例

(
w = Window.new.front;
a = StaticText(w, Rect(10, 10, 200, 20));
a.string = "Rolof's Rolex";
)

// 调整外观、对齐方式和内容
a.background = Color.grey;
a.align = \center;
a.font = Font("Monaco", 11);
a.string = "Your Rolex";

在 Synth 中监控数值

(
var w, a, r, b, q;
w = Window("Frequency Monitor", Rect(200, Window.screenBounds.height - 200, 300, 150)).front;

a = StaticText(w, Rect(45, 10, 200, 20)).background_(Color.rand);
a.string = " Current Frequency ";

Button.new(w, Rect(45, 70, 200, 20)).states_([["close", Color.black, Color.rand]]).action_({ w.close });

s.waitForBoot({

    b = Bus.new(\control, 0, 1);

    q = SynthDef(\Docs_FreqMonitor, {
        var freq, snd;
        freq = LFNoise0.ar(2, 400, 650);
        snd = SinOsc.ar(freq, 0, 0.2);
        Out.ar(0, snd);
        Out.kr(b.index, freq); // 将频率输出到控制总线
    }).play;

    r = Routine {
        {
            // 将 StaticText 的值设置为控制总线中的值。
            // 设置 GUI 值是异步的，因此你必须在系统时钟中使用 .defer。
            // 另外你必须检查窗口是否仍然打开，因为 Routine 在窗口关闭后
            // 至少还会继续执行一步。
            b.get({ |v|
                {
                    if(w.isClosed.not) {
                        a.string = " Current Frequency: " ++ v.round(0.01)
                    }
                }.defer
            });

            0.01.wait;
        }.loop

    }.play
});

CmdPeriod.doOnce({ w.close });
w.onClose = { r.stop; q.free; b.free }; // 如果窗口关闭则进行清理
)

动态文本

(
w = Window.new.front;
w.view.background = Color.white;
a = Array.fill(20, { StaticText(w, Rect(w.bounds.extent.x.rand, w.bounds.extent.y.rand, 160, 16))
    .string_("Rolof's Rolex".scramble)
    .align_(\center)
    .stringColor_(Color.rand)
    .font_(Font([
        "Helvetica-Bold",
        "Helvetica",
        "Monaco",
        "Arial",
        "Gadget",
        "MarkerFelt-Thin"
    ].choose, 11))
});

r = {
    inf.do { |i|
        thisThread.randSeed_(1284);
        a.do { |item|
            // 设置 GUI 值是异步的，因此必须使用 .defer
            {
                item.bounds = Rect(
                    5 + w.bounds.extent.x.rand * abs(cos(i * 0.01)),
                    w.bounds.extent.y.rand * sin(i * 0.01),
                    160,
                    20
                )
            }.defer;
        };
        0.15.wait;
    }
}.fork;
CmdPeriod.doOnce({ w.close });
w.onClose_({ r.stop });
)
