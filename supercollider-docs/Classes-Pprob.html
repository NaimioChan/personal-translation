Pprob

具有任意概率分布的随机值

Pattern : AbstractFunction : Object

参见：Ppoisson

说明

在实例化时创建一个积分表（CPU 密集），随后由流使用以高效地生成随机值。

类方法

Pprob.new(distribution, lo: 0.0, hi: 1.0, length: inf, tableSize)

参数：

distribution

期望的概率分布（直方图）。

lo

结果值的下界。

hi

结果值的上界。

length

要重复的值的数量。

tableSize

将表重采样到此大小。如果分布的大小小于 64，则会（线性地）重采样到此最小大小。

(distribution)

设置分布，表格将被重新计算。

(tableSize)

设置重采样大小，表格会被重新计算。

示例

// 一致性测试
(
var a = Pprob([0,0,0,0,1,1,1,1,3,3,6,6,9].scramble);
var b = a.asStream;
b.nextN(800).sort.plot("sorted distribution");
b.nextN(800).sort.plot("sorted distribution, again");
)


// 比较：模拟 linrand
(
var a, b, x, y;
a = Pprob([1, 0]);
x = Pfunc({ 1.0.linrand });

b = a.asStream;
y = x.asStream;

postf("Pprob mean: % linrand mean: % \n", b.nextN(800).mean, y.nextN(800).mean);

b.nextN(800).sort.plot("this is Pprob");
y.nextN(800).sort.plot("this is linrand");
)


// 比较效率

bench { Pprob([0, 1]) } // 这一操作开销相当大
bench { 16.do { Pseq([0, 1] ! 32) } }

x = Pprob([0, 1]).asStream;
y = Pseq([0, 1], inf).asStream;

bench { 100.do { x.next } }; // 这非常高效
bench { 100.do { y.next } };



// 声音示例
(
SynthDef(\help_sinegrain,
    { |out = 0, freq = 440, sustain = 0.05|
        var env;
        env = EnvGen.kr(Env.perc(0.01, sustain, 0.2), doneAction: Done.freeSelf);
        Out.ar(out, SinOsc.ar(freq, 0, env))
    }).add;
)


(
var t;
a = Pprob([0, 0, 1, 0, 1, 1, 0, 0], 60, 80);
t = a.asStream;
Routine({
    loop({
    Synth(\help_sinegrain, [\freq, t.next.midicps]);
    0.01.wait;
    })
}).play;
)

a.distribution = [0, 1];
a.distribution = [1, 0];
a.distribution = [0, 0, 0, 0, 1, 0];
a.distribution = [0, 1, 0, 0, 0, 0];

// 更高的分辨率会带来更精确的分布：
