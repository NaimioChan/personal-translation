Pspawner

从例程中动态控制多个事件流

Prout : Pattern : AbstractFunction : Object

参见：Pspawn

说明

Pspawner 允许例程动态启动和停止子模式。

类方法

Pspawner.new(routineFunc)

来自父类：Prout

参数：

routineFunc

该函数定义了一个 Routine，它接收一个 Spawner 作为其唯一参数。对子模式的所有控制都通过 spawner 进行。

Spawner 响应以下消息：

par

与例程并行地开始一个事件流。如果 delta 非零，模式将在“现在”之后 delta 拍开始，前提是现在 + delta 晚于 Spawner 将生成的下一个事件。该方法返回该流。可以从任何对象调用。

seq

运行整个模式，然后将控制权返回给例程。

wait

等待 dur 秒，然后将控制权返回给例程。

suspend

找到 Spawner 中的流并停止它，如果未找到该流则返回 nil，否则返回该流。

suspendAll

停止 Spawner 的所有子流。

注意： 这些也记录在 Spawner 中。

示例

// 示例 1：一个简单的 Pspawner

(
Pspawner({ |sp|

// 平行的类 in 脉冲将贯穿整个示例
    sp.par(Pbind(*[degree: [0, 7], octave: 7, dur: 0.2, db: Pseq([-20, -24, -22, -24], inf)]));

// 带暂停的序列音阶
    sp.seq(
        Pbind(*[degree: Pseq((0..7).mirror), dur: 0.2])
    );
    sp.wait(0.4);
    sp.seq(
        Ppar([
            Pbind(*[degree: Pseq((0..7).mirror), dur: 0.2, octave: 4]),
            Pbind(*[degree: Pseq((0..7).reverse.mirror), dur: 0.2])
        ])

    );
    sp.wait(3);
    sp.seq(
        Pbind(*[degree: Pseq((0..7).mirror), dur: 0.2, mtranspose: (0, 2..14)])
    );

// 以 0.4 秒间隔重叠的比例

    10.do {
        sp.par(
            Pbind(*[degree: Pseq((0..7).mirror), dur: 0.2])
        );

        sp.wait(0.4)
    };
    sp.wait(1.6);
    sp.suspendAll;
}).play
)


(
// 示例 2：以 4 秒间隔创建 6 个流
// 然后按创建顺序删除它们

Pspawner({ |sp|
    var streams, stream;
    // 启动 pattern，收集产生的事件流
    streams = [2, 3, 4, 6, 7, 8].collect { |i|
        stream = sp.par(Pbind(*[
            harmonic: i,
            ctranspose:     [0, 1, 3]/40,
            octave:    Pbrown(2, 8, 2), dur: 1/i, db: -30
        ]));
        sp.wait(4);
        stream;
    };

    // 现在逐个停止这些流
    streams.do { |s| sp.suspend(s); sp.wait(4) };
}).play
)

(
// 示例 3：定义一个 Pspawner 并使用 Pattern 操作
p = Pspawner({ |sp|
    var pat = Pbrown(-7, 7, 3);

    sp.par(Pbind(*[octave: 5, degree: pat, dur: 1/4]));
    sp.wait(2);
    sp.par(Pbind(*[octave: 6, degree: pat, dur: 1/8]));
    sp.wait(3);
    sp.par(Pbind(*[octave: 7, degree: pat, dur: 1/6, db: -20]));
    sp.wait(4);
    sp.par(Pbind(*[octave: 4, degree: pat, dur: 1/4]));

    sp.wait(8);
    sp.suspendAll;
});
)

// 播放基本模式
p.play;

(
// 用 Pchain 操作基本 pattern
Pchain(
    Pbind(*[mtranspose: Pkey(\mtranspose) + Pdup(8, Prand([0, [0, 3, -2], [0, 2, 4]], inf))]), // 将一些音符组成三和弦
    Pbind(*[ctranspose: Pwhite(-0.1, 0.1)]), // 给音符添加随机失谐
    Pbind(*[\scale, Scale.minor]), // 改变比例
    Pn(Pseq([p, (type: \rest, dur: 1.0)])), // 暂停 1 秒后重复模式
    Pbind(*[
        db: Pstep(Pseq([-10, -13, -13, -11, -13, -13], inf), 0.1) - 10,
        mtranspose: Pstep(Pwhite(-7, 7), Prand([5, 4, 2], inf)) // 随机调式移调
    ])
).play(protoEvent: Event.default)
)

// 示例 4：从单独的代码修改 Pspawner 的内容

(
a = Pspawner({ |sp|
    c = sp;             // 将 Spawner 存储在全局变量中
    100.do{ sp.wait(1) }
});
a.play;
)

(                    // 在 Pspawner 运行之前，c 将无效
b = c.par(                // 现在在 spawner 中启动一个模式
    Pbind(*[degree: Pseq((0..6) ++ (7..1), inf), dur: 0.2])
);
)
c.suspend(b)                // 我们可以挂起并恢复该流
c.par(b)
c.suspend(b)
(
b = c.par(                // 或直接启动一个新模式
    Pbind(*[degree: Pseq((0..6) ++ (7..1), inf), dur: 0.2])
);
)

// 示例 5：Spawner 可以直接以 Pspawner 的方式使用。
// 这允许外部代码无论 spawner 是否已运行都能访问它
(
c = Spawner({ |sp|
    100.do{ sp.wait(1) }
});
b = c.par(                // 现在在 spawner 中启动一个模式
    Pbind(*[degree: Pseq((0..6) ++ (7..1), inf), dur: 0.2])
);
c.play;                    // 在这种情况下，c 始终有效
)
c.suspend(b)                // 我们可以挂起并恢复该流
c.par(b)
c.suspend(b)
(
b = c.par(                // 或直接启动一个新模式
    Pbind(*[degree: Pseq((0..6) ++ (7..1), inf), dur: 0.2])
);
)

(
Pspawner({ |sp|
    (1..5).do { |i|
        sp.par(Pbind(*[
            octave: i + 2,
            degree:    Pwhite(0, 7), dur: 1/i, db: -30
        ]));
        sp.wait(4);
    };
    sp.suspendAll;
}).play
)
