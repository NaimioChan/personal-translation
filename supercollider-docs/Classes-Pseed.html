Pseed

在子模式中设置随机种子

FilterPattern : Pattern : AbstractFunction : Object

另请参阅：Random Seed

说明

设置结果流的随机生成器种子。

类方法

Pseed.new(randSeed, pattern)

参数：

randSeed

整数、返回整数的模式或流。

注意： randSeed 始终被视为模式/流。如果你提供一个单一的常量种子值，它将表现为一个无限长度的流。这会导致子模式被无限次嵌入。比较：// Pwhite 永远重复它的三个值 Pseed(1000, Pwhite(1, 10, 3)).asStream.nextN(10); // Pwhite 只运行一次：// 输出流由三个值组成，然后无限地 'nil' Pseed(Pn(1000, 1), Pwhite(1, 10, 3)).asStream.nextN(10);

pattern

实例方法

未文档化的实例方法

.embedInStream(inval)

.randSeed

.randSeed = value

示例

a = Pseed(1972, Prand([1, 2, 3], inf));

b = a.asStream;
10.do({ b.next.post });

c = a.asStream;
10.do({ c.next.post });


// 使用种子模式作为输入：

a = Pseed(Pseq([1812, 1912], inf), Prand([1, 2, 3], 5));

b = a.asStream;
2.do({ 5.do({ b.next.post }); "".postln });

c = a.asStream;
2.do({ 5.do({ c.next.post }); "".postln });


// 外层线程是独立的：

a = Pseed(Prand([1534, 1600, 1798, 1986, 2005], inf), Pshuf([1, Prand([7, 9], 2), 1, 2, 3], 1));

// 返回随机流
b = a.asStream;
2.do({ 5.do({ b.next.post }); "".postln });

c = a.asStream;
2.do({ 5.do({ c.next.post }); "".postln });


// Pseed 在音乐中使用方式的一些示例。
(
SynthDef.new(\varsaw, {
    arg dur, attack = 0.01, release = 1.0,
    t_gate = 1, out, freq = 442, cutoff = 5500,
    rq = 1, pan = 0.0, amp = 0.5, width = 0.1;

    var env = EnvGen.kr(Env.perc(attack, release), t_gate, timeScale: dur, doneAction: 2);
    var sig = VarSaw.ar(freq: freq, mul: env, width: width);
    sig = RLPF.ar(sig, cutoff.clip(20.0, 20000.0), rq.clip(0.0, 1.0));
    sig = Pan2.ar(sig, pan);
    Out.ar(out, sig * amp);
}).add;
)

(
Pdef(\varsawPat,
    Pseed(490, // 这个数字是种子。
        Pbind(*[
            instrument: \varsaw,
            attack: Plprand(0.01, 1.0),
            release: Pwrand([0.5, 8], [8, 1].normalizeSum, inf),
            dur: 1 / Pstutter(Phprand(5, 11), Plprand(5, 11)),
            freq: 100 * Plprand(1, 9),
            width: Phprand(0.0, 0.5),
            cutoff: Phprand(20, 5500),
            pan: Pmeanrand(-1.0, 1.0),
        ])
    )
).play
)

// 重新评估上述模式以听到相同的起点。无限的变体从单个数字展开。
// 更改种子以获得另一个无限的确定性选择流。


(
SynthDef(\drum, { |out, dur, t_gate = 1, pew = 1, sustain = 1, pan, fed = 0, tun, amp = 1|
    var env, freq, sig;
    tun = ((tun>0)*tun) + ((tun<1)*3);
    freq = (tun*10).midicps;
    env = EnvGen.ar(Env.linen(0.01, 0, 1, 1, -3), t_gate, timeScale: dur, doneAction: 2);
    sig = LPF.ar(SinOscFB.ar(XLine.ar(freq.expexp(10, 2000, 1000, 8000), freq, 0.025/pew), fed), 9000);
    sig = Pan2.ar(sig, pan);
    Out.ar(out, sig * amp);
}).add
)

(
Pdef(\drumPat,
    Pseed(9223372036854775807, // 这是最大的 64 位数字。
        Psync(
            Pbind(*[
                instrument: \drum,
                tun: Pstutter(Plprand(4, 8), Plprand(0.0, 8.0).round(1/8)),
                amp: Plprand(-7.5.dbamp, -1.5.dbamp),
                dur: 1 / Pstutter(Phprand(5, 11), Pwhite(5, 11)),
                legato: Plprand(0.75, 4.0),
                pan: Prand([Phprand(0.0, 0.5, 1), Plprand(0.5, 1.0, 1)], inf),
                // pew: 0.4,
                // pew: Phprand(0.4, 10.0),
            ]), 1, 2.0, // 循环长度是这里的第二个数字。
        )
    )
).play(quant: 1);
)

/*
当 Pseed 像上面例子那样环绕整个模式时，参数值会自上而下地生成。
取消注释静态的 pew 值不会改变模式，但取消注释下面那行带有随机选择的内容则会改变。
pew 参数现在接收到的值原本会传给 tun 参数，于是这条替代时间线随着时间的推移
越来越偏离。
*/

(
Pdef(\varsawPat,
    Pbind(*[
        instrument: \varsaw,
        attack: Plprand(0.01, 1.0),
        release: Pwrand([0.5, 8], [8, 1].normalizeSum, inf),
        dur: 1 / Pseed(490, Pstutter(Phprand(5, 11), Plprand(5, 11))), // 用于单个参数的 Pseed。
        amp: Plprand(-18.dbamp, -12.dbamp),
        freq: 100 * Plprand(1, 9),
        width: Plprand(0.0, 0.5),
        cutoff: Phprand(20, 5500),
        pan: -1,
    ])
).play(quant: 1);

Pdef(\drumPat,
    Pbind(*[
        instrument: \drum,
        tun: Pstutter(Plprand(4, 8), Plprand(0.0, 8.0).round(1/8)),
        fed: Plprand(0.0, 1.0),
        amp: Plprand(-18.dbamp, -12.dbamp),
        dur: 1 / Pseed(490, Pstutter(Phprand(5, 11), Plprand(5, 11))), // 相同的种子，相同的节奏。
        legato: Plprand(0.75, 4.0),
        pan: 1,
    ])
).play(quant: 1);
)

/*
在上面的例子中，Pseed 被用作一种“数据共享”策略。这两个模式实际上并没有共享数据。
它们都使用相同的种子来为 dur 参数生成所有未来的随机选择。
两个模式中的所有其他参数都不受影响。
*/
