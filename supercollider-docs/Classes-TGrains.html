TGrains

缓冲区颗粒合成器。

MultiOutUGen : UGen : AbstractFunction : Object

参见：GrainBuf

说明

触发器从缓冲区生成颗粒。每个颗粒都有一个 Hanning 包络

(sin2(x) for x from 0 to π)

并且可以声像定位到多声道输出上。

类方法

TGrains.ar(numChannels, trigger: 0, bufnum: 0, rate: 1, centerPos: 0, dur: 0.1, pan: 0, amp: 0.1, interp: 4)

参数：

numChannels

输出通道数。必须是非零正整数。此值在编译 SynthDef 时固定，因此不能赋值给 SynthDef 参数。

trigger

每次触发时，以下参数会被采样并用作新 grain 的参数。当信号从非正值变为正值时会发生触发。

如果触发是音频速率，则颗粒将以采样精度开始。

bufnum

要使用的缓冲区索引。它必须是单声道缓冲区。

rate

1.0 为正常速率，2.0 为升高一个八度，0.5 为降低一个八度，-1.0 为反向正常速率……等等。

centerPos

颗粒包络在缓冲区中达到最大振幅的位置，以秒为单位。

dur

颗粒的持续时间（秒）。

pan

决定将输出声像定位到何处。

  • 如果 numChannels = 1，则忽略 pan 参数。
  • 如果 numChannels = 2，声像处理类似于 Pan2。
  • 如果 numChannels > 2，则声像处理与 PanAz 相同。

amp

颗粒的振幅。

interp

1、2 或 4。决定颗粒使用 (1) 无插值、(2) 线性插值还是 (4) 三次插值。

实例方法

示例

s.boot;
b = Buffer.read(s, ExampleFiles.child);

(
{
    var trate, dur, rate;
    trate = MouseY.kr(2, 200, 1);
    dur = 4 / trate;
    rate = Dseq([10, 1, 1, 0.5, 0.5, 0.2, 0.1], inf);
    TGrains.ar(2, Impulse.ar(trate), b, rate, MouseX.kr(0, BufDur.kr(b)), dur, Dseq([-1, 1], inf), 0.1, 2);
}.scope(zoom: 4);
)

(
{
    var trate, dur, clk, pos, pan;
    trate = MouseY.kr(8, 120, 1);
    dur = 12 / trate;
    clk = Impulse.kr(trate);
    pos = MouseX.kr(0, BufDur.kr(b)) + TRand.kr(0, 0.01, clk);
    pan = WhiteNoise.kr(0.6);
    TGrains.ar(2, clk, b, 1, pos, dur, pan, 0.1);
}.scope(zoom: 4);
)

// 4 个声道
(
{
    var trate, dur, clk, pos, pan;
    trate = MouseY.kr(8, 120, 1);
    dur = 12 / trate;
    clk = Impulse.kr(trate);
    pos = MouseX.kr(0, BufDur.kr(b)) + TRand.kr(0, 0.01, clk);
    pan = WhiteNoise.kr(0.6);
    TGrains.ar(4, clk, b, 1, pos, dur, pan, 0.1);
}.scope(4, zoom: 4);
)

(
{
    var trate, dur, clk, pos, pan;
    trate = MouseY.kr(8, 120, 1);
    dur = 4 / trate;
    clk = Dust.kr(trate);
    pos = MouseX.kr(0, BufDur.kr(b)) + TRand.kr(0, 0.01, clk);
    pan = WhiteNoise.kr(0.6);
    TGrains.ar(2, clk, b, 1, pos, dur, pan, 0.1);
}.scope(zoom: 4);
)

(
{
    var trate, dur, clk, pos, pan;
    trate = LinExp.kr(LFTri.kr(MouseY.kr(0.1, 2, 1)), -1, 1, 8, 120);
    dur = 12 / trate;
    clk = Impulse.ar(trate);
    pos = MouseX.kr(0, BufDur.kr(b));
    pan = WhiteNoise.kr(0.6);
    TGrains.ar(2, clk, b, 1, pos, dur, pan, 0.1);
}.scope(zoom: 4);
)

(
{
    var trate, dur, clk, pos, pan;
    trate = 12;
    dur = MouseY.kr(0.2, 24, 1) / trate;
    clk = Impulse.kr(trate);
    pos = MouseX.kr(0, BufDur.kr(b)) + TRand.kr(0, 0.01, clk);
    pan = WhiteNoise.kr(0.6);
    TGrains.ar(2, clk, b, 1, pos, dur, pan, 0.1);
}.scope(zoom: 4);
)

(
{
    var trate, dur, clk, pos, pan;
    trate = 100;
    dur = 8 / trate;
    clk = Impulse.kr(trate);
    pos = Integrator.kr(BrownNoise.kr(0.001));
    pan = WhiteNoise.kr(0.6);
    TGrains.ar(2, clk, b, 1, pos, dur, pan, 0.1);
}.scope(zoom: 4);
)

(
{
    var trate, dur, clk, pos, pan;
    trate = MouseY.kr(1, 400, 1);
    dur = 8 / trate;
    clk = Impulse.kr(trate);
    pos = MouseX.kr(0, BufDur.kr(b));
    pan = WhiteNoise.kr(0.8);
    TGrains.ar(2, clk, b, 2 ** WhiteNoise.kr(2), pos, dur, pan, 0.1);
}.scope(zoom: 4);
)

(
{
    var trate, dur;
    trate = MouseY.kr(2, 120, 1);
    dur = 1.2 / trate;
    TGrains.ar(2, Impulse.ar(trate), b, (1.2 ** WhiteNoise.kr(3).round(1)), MouseX.kr(0, BufDur.kr(b)), dur, WhiteNoise.kr(0.6), 0.1);
}.scope(zoom: 4);
)

// 将 demand ugen 作为输入
(
{
    var trate, dur, z, d;
    trate = MouseX.kr(1, 100, 1);
    d = { Dwhite(0.1, 0.2, 1) };
    z = {
        Drand([Dgeom(0.1, 1 + d.value, Diwhite(20, 40)), Dgeom(1, 1 - d.value, Diwhite(20, 40))])
    };
    TGrains.ar(2,
        Impulse.ar(trate),
        bufnum: 10,
        rate: Dseq([1, 1, z.value, 0.5, 0.5, 0.2, 0.1, 0.1, 0.1, 0.1], inf) * 2 + 1,
        centerPos: Dseq(z.dup(8), inf),
        dur: Dseq([1, d.value, 1, z.value, 0.5, 0.5, 0.1, z.value] * 2, inf) / trate,
        pan: Dseq([1, 1, 1, 0.5, 0.2, 0.1, 0, 0, 0], inf) * 2 - 1,
        amp: Dseq([1, 0, z.value, 0, 2, 1.0, 1, 0.1, 0.1], inf)
    );
}.scope(zoom: 4);
)

b.free
