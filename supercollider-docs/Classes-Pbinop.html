Pbinop

二元运算符模式

Pattern : AbstractFunction : Object

另请参阅：Pnaryop、Punop、BinaryOpFunction

说明

返回一个流，该流将二元运算符应用于接收者的流值。通常，这是将二元运算符（即带一个参数的方法）应用于模式的结果。

二元运算符的示例有：+, -, /, *, min, max, hypot ...

类方法

Pbinop.new(operator, a, b, adverb)

参数：

operator

要应用的操作符

a

一个模式（或兼容的模式输入）

b

一个模式（或兼容的模式输入）

adverb

实例方法

未文档化的实例方法

.a

.a = value

.adverb

.adverb = value

.asStream

.b

.b = value

.operator

.operator = value

示例

(
var a;
a = Pbinop(\hypot, Pseries(0, 1, 12), Pseries(3, -1, 12));
a.asStream.all;
)

// 这与以下相同：
(
var a;
a = Pseries(0, 1, 12).hypot(Pseries(3, -1, 12));
a.asStream.all;
)

// 也可写作：
(
var a;
a = Pseries(0, 1, 12) hypot: Pseries(3, -1, 12);
a.asStream.all;
)

// 一些常见情况：
Pseq([1, 2, 3]) + 2;
Pseq([1, 2, 3]) / Pseq([3, 4, 5, 6]);
max(Pwhite(-10, 10, inf), Pseq([0, 2, 3, 4]));
// 声音示例
(
SynthDef(\help_sinegrain,
    { |out = 0, freq = 440, sustain = 0.05, amp = 0.1|
        var env;
        env = EnvGen.kr(Env.perc(0.01, sustain, 0.2, amp), doneAction: Done.freeSelf);
        Out.ar(out, SinOsc.ar(freq, 0, env))
    }).add;
)

(
var a;
a = Pn(Pbinop(\hypot, Pseries(0, 1, 34), Pseries(3, -1, 34)), inf).asStream;
{
    a.do { |val|
        Synth(\help_sinegrain, [\freq, a * 200 + 300].postln);
        0.05.wait;
    }
}.fork;
)

(
Pbind(
    \dur, 0.01,
    \instrument, \help_sinegrain,
    \note, Pn(Pbinop(\hypot, Pwhite(0, 12, 13), Pseries(3, -1, 12)))
).play;
)



// 这些与以下相同：

(
var a;
a = Pn(Pseries(0, 1, 34) hypot: Pseries(3, -1, 34), inf).asStream;
{
    a.do { |val|
        Synth(\help_sinegrain, [\freq, a * 200 + 300].postln);
        0.05.wait;
    }
}.fork;
)

Pbind(

    \note, Pn(Pwhite(0, 12, 13) hypot: Pseries(3, -1, 12))
