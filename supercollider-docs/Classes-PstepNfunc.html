PstepNfunc

组合模式

Pattern : AbstractFunction : Object

参见：PstepNadd

子类：PstepNadd

说明

通过求值一个函数（深度优先遍历）来组合任意数量的模式。当一个流结束时，它会从其模式重新创建，直到顶层流结束。

示例

(
f = { |vals|
    vals.postln;
};
x = PstepNfunc(f, [
        Pseq([1, 2, 3]), Pseq([4, 5, 6]), Pseq([7, 8, 9])
    ]).asStream;
50.do({ x.next });
)

(
f = { |vals|
    var r;
    r = vals.copy.removeAt(0);
    vals.do({ |item|  r = item / r.squared * 10 });
    r
};
x = PstepNfunc(f,
    [
        Pseq([1, 2, 3], inf),
        Pseq([2, pi, 1]),
        Pseq([0.1, 3, 0.2, 3])
    ]
    ).asStream;

50.do({ x.next.postln });
)

// 注意，如果最后一个 pattern 循环，它会卡在那个 pattern 上：
(
f = { |vals|
    vals.postln;
};
x = PstepNfunc(f, [Pseq([1, 2, 3]), Pseq([10, 20, 30, 40]), Pseq([100, 200, 300], inf)]).asStream;
50.do({ x.next });
)


(
f = { |vals|
    vals.inject(1, { |x, y| x * y })
};
x = PstepNfunc(f,
    [
        Pseq([1, 2, 3], inf),
        Pseq([2, pi, 1]),
        Pseq([0.1, 3, 0.2, 3])
    ]
    ).asStream;

50.do({ x.next.postln });
)
