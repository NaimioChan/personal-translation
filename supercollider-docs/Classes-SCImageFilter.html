SCImageFilter

与 SCImage 配合使用的过滤器类

参见：SCImage、SCImageKernel

说明

目前，此类表示可应用于 SCImage 的 CoreImage 滤镜。所有内置滤镜及文档：http://developer.apple.com/documentation/GraphicsImaging/Reference/CoreImageFilterReference/Reference/reference.html

类方法

.new

f = SCImageFilter.new(\CIStarShineGenerator);
f.attributes;

// 或者，如果你已经知道此 synth 的属性，也可以采用类似 Synth 的风格
f = SCImageFilter.new(\CIStarShineGenerator, [\center, [200, 200], \radius, 200*0.05]);
f.values; //

参数：

filterName

过滤器名称 Symbol 或 String

args

过滤器名称 Symbol 或 String

.filterCategories

返回一个 Dictionary，包含按类别关联的所有过滤器。

// 获取过滤器类别（字典）
(
    SCImageFilter.filterCategories.keysDo({ |cat|
        cat.postln;
    });
    "ok".postln;
)

// 获取某个类别的 Filters - 返回一个 SymbolArray
SCImageFilter.filterCategories.at(\CICategoryGeometryAdjustment);
SCImageFilter.filterCategories.at(\CICategoryGenerator);

// 查找非内置插件通常是在 (/Library/Graphics/Image Units/) 中加载的插件
// 所有插件都在启动时加载
// 你可以在 http://www.noiseindustries.com/products/ 找到免费插件
(
var n = 0;
SCImageFilter.filterCategories.do ({
    |symbolArray|
    symbolArray.do ({ |pluginName|
        if(pluginName.asString.beginsWith("CI").not, {
            ("External Filter Found: " ++ pluginName).postln;
            n = n+1;
        });
    });
});
(n + "plugins found").postln;
n;
)

实例方法

.attributes

返回一个 IdentityDictionary，其中包含每个关联：

  • 作为键的属性/特性名称。
  • 你应作为参数传入以设置该属性的 SuperCollider Class。

一旦你了解了这些属性，就可以像使用普通实例设置方法一样设置它们，使用名称并追加 '_'。

(
f = SCImageFilter.new(\CIStarShineGenerator);
f.attributes.keysValuesDo({ |k, v|
    ("CIStarShineGenerator responds to "++k.asString++"_("++v.asString++")").postln;
});
)

f.center_([200, 200]);
f.radius_(200*0.05);

.attributeRange

获取某个属性的数值范围和默认值。返回一个数组，形式为 [min, max, default]。min、max 或 default 可以是 Float、Color、Array 或 Nil。

// 一个 SCImageFilter
f = SCImageFilter.new(\CIFlashTransition);

f.dump;

f.attributes;

f.attributeRange(\time);

// 获取所有属性的 MIN - MAX 可能值
(
f.attributes.keysDo({ |attr|
    (attr ++ " = " + f.attributeRange(attr)).postln; // nil 结果表示这些属性没有最小/最大值
});
)

.values

返回你为每个属性设置的所有值。如果某个属性未显式设置值，则在应用于 SCImage 时将其设为默认值。

(
    f = SCImageFilter.new(\CIStarShineGenerator);
    f.center_([200, 200]);
    f.radius_(200*0.05);
    f.color_(Color.blue);
    f.crossWidth_(2.0);
    f.crossAngle_(0.0);
    f.crossOpacity_(-4.0);
    f.values.postln;
)

.set

设置此 SCImageFilter 的属性

(
    f = SCImageFilter.new(\CIStarShineGenerator);
    f.set(\center, [200, 200], \radius, 200*0.05, \color, Color.blue, \crossWidth, 2.0);
    f.values.postln;
)

.enable

是否在应用于 SCImage 时启用 Filter。

参数：

value

如果为 false，应用该滤波器将不执行任何操作。

示例

// thor 的快速实验请求
// 使用 filters 数组的示例，允许你在不就地应用的情况下使用滤波器
// 对于实时使用 / 测试或任何其他用途都很方便……

// 在这里缩小图像，否则配合 zoomblur 使用 addFilter 而非 applyFilter 会让你的电脑
// 吃不消
// 算是快速的交换测试

a = SCImage.new("/Library/Desktop Pictures/Plants/Peony.jpg").scalesWhenResized_(true).setSize(500, 400);
a.plot; // 先看看我 - 我很美

// 但我想要你变为色调分离效果
(
h = SCImageFilter.new(\CIColorPosterize);
a.addFilter(h); // 首次调用所需
a.plot; // 应该会有它
)

// 然后你就可以自由地直接访问和设置该 slot——更简单
(
a.filters[0] = SCImageFilter.new(\CIColorInvert);
a.plot;
)

// again
(
a.filters[0] = SCImageFilter.new(\CIZoomBlur);
a.plot;
)

// again
(
a.filters[0] = SCImageFilter.new(\CIRandomGenerator);
a.plot;
)

(
a.removeAllFilters;
a.plot(freeOnClose: true);
)


// ** 掩蔽示例 **
(
f = SCImageFilter.new(\CIColorMonochrome); // 创建一张灰度图像
g = SCImageFilter.new(\CISourceInCompositing); // 合成我们将使用
f.color_(Color.black);
f.intensity_(1.0);
a = SCImage.new(SCDoc.helpSourceDir +/+ "images/vduck2.jpg");
a.bounds;
b = SCImage.new(SCDoc.helpSourceDir +/+ "images/flowers2.jpg");
a.applyFilters([f, SCImageFilter(\CIColorInvert), SCImageFilter(\CIMaskToAlpha)]); // grayscale + invert + maskToAlpha = 创建遮罩
g.backgroundImage_(a); // 设置背景图像
b.applyFilters(g); // 创建蒙版图像
a.free;
w = b.plot(freeOnClose: true, background: Color.clear); // 设置为清除颜色以清晰地看到图像
)


// *** 使用 SCImage 的 .filters 属性实现非常简单的实时 FX ***
// SCImage.filters 提供了一种设置要在渲染链中应用的滤波器的方法
// 这对于在实时中修改某些滤波器属性很有用
// 但它们不会就地处理，而是在每次渲染时计算
// 这里滤波器计算在每次渲染调用时执行
// 所以这很方便但不是最优的
// SCPhotoshop :)
(
var width = 500, height = 500, centerVector;
var controller, specs;

centerVector = [width*0.5, height*0.5];
a = SCImage.new(500@500);
a.accelerated_(true);
f = SCImageFilter.new(\CIStarShineGenerator);
g = SCImageFilter.new(\CIPixellate);
h = SCImageFilter.new(\CIZoomBlur);

f.center_(centerVector);
f.radius_(width*0.05);
f.color_(Color.green);
f.crossWidth_(2.0);
f.crossAngle_(0.0);
f.crossOpacity_(-4.0);
g.center_(centerVector);
h.center_(centerVector);
h.amount_(50);

// 简单的示例控件
specs = [
    [f, \crossOpacity_, [-8, 0].asSpec, -4],
    [f, \crossAngle_, [-pi, pi].asSpec, 0.0],
    [h, \amount_, [0, 200].asSpec, 50],

]; // CIStarShine 不透明度，CIZoomBlur

controller = SCWindow.new("Close Me First !!!", Rect(400, 400, 300, specs.size * 40));
controller.view.decorator = FlowLayout(controller.view.bounds.insetBy(10), 10@10);
specs.size.do { |i|
    SCSlider.new(controller, Rect(0, 0, 150, 20))
    .action_({ |obj|
        specs[i][0].perform(specs[i][1], specs[i][2].map(obj.value));
        w.refresh;
    })
    .value_(specs[i][2].unmap(specs[i][3]));
    controller.view.decorator.nextLine;
};

// 添加滤波器
a.addFilter(f);
a.addFilter(g);
a.addFilter(h);

a.filters.do { |filt, i|
    if(i != 0, {
        SCButton.new(controller, Rect(0, 0, 20, 20))
        .action_({ |obj|
            filt.enable_(obj.value != 1);
            w.refresh;
        })
        .states_([["", Color.white], ["", Color.white, Color.black]]);
    });
};

// 绘图
w = a.plot(background: Color.black);
controller.front;
controller.onClose_({
    {
    w.close;
    }.defer(0.1);
});
)
