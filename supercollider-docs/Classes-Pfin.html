Pfin

限制嵌入流中的事件数量

FilterPattern : Pattern : AbstractFunction : Object

另请参阅：Pfinval、Pfindur、Pconst

子类：Pfinval

说明

限制给定模式或流的输出值数量。如果流能够输出更多值，Pfin 会提前终止它。

该数字可以以常数形式给出，也可以由函数或流即时计算得出。参见下面的“音高序列”示例。

类方法

Pfin.new(count, pattern)

将 pattern 的 count 个元素嵌入到流中。

示例

(
var a, b;
a = Pfin(5, Pseq(#[1, 2, 3], inf));
b = a.asStream;
9.do({ b.next.postln });
)


// Pfin 用作音高序列

(
SynthDef(\help_sinegrain,
    { |out = 0, freq = 440, sustain = 0.05|
        var env;
        env = EnvGen.kr(Env.perc(0.01, sustain, 0.2), doneAction: Done.freeSelf);
        Out.ar(out, SinOsc.ar(freq, 0, env))
    }).add;
)

(
var c, b;
c = Pn(Pfin({ rrand(3, 5) }, Pseq([1, 2, 3, 4, 5, 6], inf) * 4 + 65), inf);
b = c.asStream;
Routine({
    loop({
        Synth(\help_sinegrain, [\freq, b.next.midicps]);
        0.12.wait;
    })
}).play;
)

// 或者更简洁地说：

p = Pbind(
    \instrument, \help_sinegrain,
    \midinote, Pn(Pfin({ rrand(3, 5) }, Pseq([1, 2, 3, 4, 5, 6], inf) * 4 + 65), inf),
    \dur, 0.12
).play;

p.stop;
