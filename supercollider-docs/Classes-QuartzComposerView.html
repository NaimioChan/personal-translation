QuartzComposerView

用于渲染 Quartz Composer 合成作品的视图

注意： QuartzComposerView 在 SC 3.13 中已弃用，在后续使用 Qt 6.x 的版本中将不再可用。

说明

QuartzComposerView 允许在 macOS 上于 SC 中渲染 Quartz Composer 合成。Quartz Composer 是一个用于处理和渲染图形数据的可视化编程环境，作为 Apple 的 XCode 开发工具的一部分免费分发。QC 针对 macOS 图形系统进行了高度优化，通常应比 Pen 更高效。有关 QC 的更多信息，请参见：https://developer.apple.com/library/mac/documentation/GraphicsImaging/Conceptual/QuartzComposerUserGuide/qc_intro/qc_intro.html 以及 http://en.wikipedia.org/wiki/Quartz_Composer

你可以使用 setInputValue、getInputValue 和 getOutputValue 方法访问输入和输出端口，或者（由于实现中的一个小技巧）直接将端口键用作 getter 和 setter。因此，下面两行代码是等价的：

myQCView.setInputValue(\valueIn, 1);
myQCView.valueIn = 1;

类方法

实例方法

.path

以 String 形式获取或设置当前乐曲的路径。

.inputKeys

获取当前组合的输入端口键。

返回值：

一个 Array，由 Symbol 组成。

.outputKeys

获取当前合成输出端口的键。

返回值：

一个 Array，由 Symbol 组成。

.start

开始渲染已加载的合成作品。

.stop

停止渲染已加载的合成作品。

.setInputValue

设置由 key 指定的输入端口的值。

参数：

key

与端口键匹配的 String 或 Symbol。

value

值的类型必须与端口的类型相对应，但 Float、Integer 和 Boolean 会在需要时进行转换。（true = 1，false = 0）

.getInputValue

获取输入端口的当前值。返回对象的类型将与端口的类型相对应。

参数：

key

与端口键匹配的 String 或 Symbol。

.getOutputValue

获取输出端口的当前值。返回的对象类型将与端口的类型相对应。

参数：

key

与端口键匹配的 String 或 Symbol。

.maxFPS

设置合成渲染的最大每秒帧数。

参数：

rate

一个 Float 或 Integer，指定期望的最大帧率。值为 0 表示不限制。

.openInQC

在 Quartz Composer 中打开当前加载的合成。在任何已保存的更改生效之前，你需要将合成重新加载到视图中。

m.start;

// 获取输入和输出键的名称
m.inputKeys;
m.outputKeys;
m.openInQC; // 你可以在作品中看到已发布的输入和输出

// 你可以使用 setInputValue、getInputValue 和 getOutputValue 访问输入和输出端口，
// 或者直接使用键作为 getter 和 setter
m.setInputValue(\direction, 1);
m.direction = 0;
m.direction;
m.fontSize; // 以 QC 单位表示的字号
m.fontSize_(0.1);

m.string_("SCTV").fontName_("Courier");
m.startColor_(Color.green);
m.billboardEnable_(false);
m.billboardEnable_(true);
m.maxFPS_(5.0);
m.maxFPS_(10);
m.maxFPS_(0.0); // no max

m.getOutputValue(\interpResult); // 文本当前的旋转角度（以度为单位）
m.interpResult;
m.systemTime; // 在作曲中发布当前 System Time
m.endColor.class; // 没错，这是一个 SC Color 对象
m.endColor == Color.white;

// 在 QC 中做这个可能更高效，但是……
{
c = Color.blue;

p = 0.25; i = 0;
// m.setInputValue(\period, p);
// m.maxFPS_(p.reciprocal * 2);
SystemClock.sched(0.0, {
b.getn(0, 256, { |buf|
    var z, x;
    z = buf.clump(2).flop;
    z = [Signal.newFrom(z[0]), Signal.newFrom(z[1])];
    x = Complex(z[0], z[1]);
    // { m.setInputValue(\magnitudes, x.magnitude.resamp1(m.bounds.height * 0.5)) }.defer
    { m.setInputValue(\magnitudes, x.magnitude * 0.025); m.setInputValue(\count, i) }.defer;
    i = i + 1;
}); p
});
)

a.free;

w = Window("Level Meters", Rect(128, 64, 200, 400)).front;

// MouseX 控制噪声幅度
(
o = OSCresponder(s.addr, '/tr', { |time, resp, msg|
    { ~meters[msg[2]].level = (msg[3] + 0.01).explin(0.01, 1.01, 0, 1) }.defer;
}).add;
b = Buffer.read(s, ExampleFiles.child);
a = { var colum, noise, imp, delimp;
    imp = Impulse.kr(20);
    delimp = Delay1.kr(imp);
    colum = PlayBuf.ar(1, b, BufRateScale.kr(b), loop: 1);
    noise = PinkNoise.ar(MouseX.kr);
    // 测量 Peak
    SendTrig.kr(imp, 0, Peak.ar(colum, delimp));
    SendTrig.kr(imp, 1, Peak.ar(noise, delimp));
    [colum, noise];
}.play;
)

a.free;


/// 使用事件类型通过模式控制 QCView：

Event.addEventType(\quartz, { |server|
    var latency = server.latency;
    var view = ~view.postln;
    var key = ~key, value = ~value;
    AppClock.sched(latency, { view.postln; view.setInputValue(key, value) });
});
