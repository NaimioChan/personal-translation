编写一个基础光线追踪器

阅读时间：6 分钟。

不少读者来信说，很想看一个光线追踪的实际例子，问：「既然你说它这么简单，为什么不给我们看个真的？」于是我们暂时偏离原来那套一步步搭建渲染器的讲法，先拼一个基础的光线追踪器出来。这个小程序大约 300 行，只花了几个小时就写好了。它算不上我们的得意之作（希望如此）——毕竟赶得急——但我们的用意是表明：把底层概念真正吃透之后，写这样一个程序是很容易的。感兴趣的话，源码可以自取。

这个速成项目没有配上细致的注释，也肯定还有优化的余地。在这个光线追踪器版本里，我们把光源做成了一个可见的球体，这样它的反射就能在反射球体的表面上被观察到。至于透明玻璃球难以看清的问题——它们外观通透，不好分辨——我们的做法是给它们染上一点红色。这一决定参照了现实中透明玻璃的行为：它未必总那么容易被察觉，很大程度上取决于周围环境。不过要注意，这个初版生成的图像并不完美；比如那颗透明红球投下的阴影看起来实得不太真实。后续课程会深入到这些细节的改进，让视觉表现更准确。此外，我们还试着实现了一些功能，比如简化的 Fresnel 效果（用的是一种叫朝向比的方法）和折射，这些内容计划在后面详细展开。其中若有哪块看起来不清楚，放心，到时候都会讲明白。眼下，你手上有一个小而能跑的程序可以折腾。

要开始用这个程序，先把源码下载到本地。你需要一个 C++ 编译器，比如 clang++，来编译代码。这个程序编译起来很直接，不需要任何特殊的库。开一个终端窗口（Windows 上用 GitBash，Linux 或 macOS 上用标准终端），进到源文件所在的目录，运行下面这条命令（假设你用的是 gcc）：


c++ -O3 -o raytracer raytracer.cpp

如果你用的是 clang，则改用下面这条命令：


clang++ -O3 -o raytracer raytracer.cpp

要生成图像，在终端里输入 ./raytracer 运行程序。稍等片刻，程序会在你的电脑上生成一个名为 untitled.ppm 的文件。这个文件可以用 Photoshop、Preview（Mac 用户）或 Gimp 打开查看。另外，如何打开和查看 PPM 图像，我们会在后面的一课里讲到。

下面是用伪代码给出的传统递归光线追踪算法的一个示例实现：


#define MAX_RAY_DEPTH 3 
 
color Trace(const Ray &ray, int depth) 
{ 
    Object *object = NULL; 
    float minDistance = INFINITY;
    Point pHit; 
    Normal nHit; 
    for (int k = 0; k < objects.size(); ++k) { 
        if (Intersect(objects[k], ray, &pHit, &nHit)) { 
            float distance = Distance(ray.origin, pHit); 
            if (distance < minDistance) { 
                object = objects[k];
                minDistance = distance;
            } 
        } 
    } 
    if (object == NULL) 
        return backgroundColor; // 返回背景色而不是 0
    // 若物体材质是玻璃且深度小于 MAX_RAY_DEPTH，则把光线分裂
    if (object-&gtisGlass && depth < MAX_RAY_DEPTH) { 
        Ray reflectionRay, refractionRay;
        color reflectionColor, refractionColor; 
        float Kr, Kt; 
 
        // 计算反射光线
        reflectionRay = computeReflectionRay(ray.direction, nHit, ray.origin, pHit); 
        reflectionColor = Trace(reflectionRay, depth + 1); 
 
        // 计算折射光线
        refractionRay = computeRefractionRay(object-&gtindexOfRefraction, ray.direction, nHit, ray.origin, pHit); 
        refractionColor = Trace(refractionRay, depth + 1); 
 
        // 计算 Fresnel 效果
        fresnel(object-&gtindexOfRefraction, nHit, ray.direction, &Kr, &Kt); 
 
        // 按 Fresnel 效果合成反射色与折射色
        return reflectionColor * Kr + refractionColor * (1 - Kr); 
    } else if (!object-&gtisGlass) { // 检查物体是否非玻璃（漫反射/不透明）
        // 仅在物体不在阴影中时计算光照
        Ray shadowRay; 
        shadowRay.origin = pHit + nHit * bias; // 加一点偏移量以避免自相交
        shadowRay.direction = Normalize(lightPosition - pHit); 
        bool isInShadow = false; 
        for (int k = 0; k < objects.size(); ++k) { 
            if (Intersect(objects[k], shadowRay)) { 
                isInShadow = true; 
                break; 
            } 
        } 
        if (!isInShadow) {
            return object-&gtcolor * light.brightness; // 该点被照亮
        }
    } 
    return backgroundColor; // 若无相互作用则返回背景色
} 
 
// 逐像素的渲染循环
for (int j = 0; j < imageHeight; ++j) { 
    for (int i = 0; i < imageWidth; ++i) { 
        Ray primRay; 
        computePrimRay(i, j, &primRay); // 假设 computePrimRay 已正确设置光线的原点与方向
        pixels[i][j] = Trace(primRay, 0); 
    } 
}

极简光线追踪器

把光线追踪器压缩到能写在一张名片上，这个想法由研究者 Paul Heckbert 首创，堪称极简编程力量的明证。Heckbert 发起的这个新奇挑战，目标是用尽可能精简的 C/C++ 代码实现一个光线追踪器，相关内容详细写在他为《Graphics Gems IV》撰写的那一篇里。这一倡议在程序员中掀起了一股热潮，激励了许多人去做这项紧凑的编码练习。

其中一个有名的例子是 Andrew Kensler 写出的版本。他的成果在视觉上相当抓人，从他程序生成的图像就能看出来。尤其令人印象深刻的是他做出的景深效果：物体随着向远处退去而变得模糊。用一段极其简短的代码生成一幅复杂度相当可观的图像，确实了不起。


// minray > minray.ppm
#include &ltstdlib.h>
#include &ltstdio.h>
#include &ltmath.h>
typedef int i;typedef float f;struct v{f x,y,z;v operator+(v r){return v(x+r.x,y+r.y,z+r.z);}v operator*(f r){return v(x*r,y*r,z*r);}f operator%(v r){return x*r.x+y*r.y+z*r.z;}v(){}v operator^(v r){return v(y*r.z-z*r.y,z*r.x-x*r.z,x*r.y-y*r.x);}v(f a,f b,f c){x=a;y=b;z=c;}v operator!(){return*this*(1/sqrt(*this%*this));}};i G[]={247570,280596,280600,249748,18578,18577,231184,16,16};f R(){return(f)rand()/RAND_MAX;}i T(v o,v d,f&t,v&n){t=1e9;i m=0;f p=-o.z/d.z;if(.01&ltp)t=p,n=v(0,0,1),m=1;for(i k=19;k--;)for(i j=9;j--;)if(G[j]&1<&ltk){v p=o+v(-k,0,-j-4);f b=p%d,c=p%p-1,q=b*b-c;if(q&gt0){f s=-b-sqrt(q);if(s&ltt&&s&gt.01)t=s,n=!(p+d*t),m=2;}}return m;}v S(v o,v d){f t;v n;i m=T(o,d,t,n);if(!m)return v(.7,.6,1)*pow(1-d.z,4);v h=o+d*t,l=!(v(9+R(),9+R(),16)+h*-1),r=d+n*(n%d*-2);f b=l%n;if(b&lt0||T(h,l,t,n))b=0;f p=pow(l%r*(b&gt0),99);if(m&1){h=h*.2;return((i)(ceil(h.x)+ceil(h.y))&1?v(3,1,1):v(3,3,3))*(b*.2+.1);}return v(p,p,p)+S(h,r)*.5;}i main(){printf("P6 512 512 255 ");v g=!v(-6,-16,0),a=!(v(0,0,1)^g)*.002,b=!(g^a)*.002,c=(a+b)*-256+g;for(i y=512;y--;)for(i x=512;x--;){v p(13,13,13);for(i r=64;r--;){v t=a*(R()-.5)*99+b*(R()-.5)*99;p=S(v(17,16,8)+t,!(t*-1+(a*(R()+x)+b*(y+R())+c)*16))*3.5+p;}printf("%c%c%c",(i)p.x,(i)p.y,(i)p.z);}}

要运行这个程序，先把代码复制粘贴到一个新的文本文档里。把这个文件重命名为 minray.cpp 或你喜欢的任何名字。接着用命令 c++ -O3 -o minray minray.cpp 编译；如果你选择用 clang 编译器，就用 clang++ -O3 -o minray minray.cpp。编译好之后，用命令行 minray > minray.ppm 运行程序。这种做法的效果是把最终的图像数据直接输出到标准输出（你正在用的终端），再用 > 操作符重定向到一个文件，存成 PPM 文件。这个格式 Photoshop 能兼容，查看起来很方便。

这里展示这个程序，是为了表明光线追踪算法能压缩到多么紧凑的程度。代码中用到的几项技术，会在本系列后续课程里详细讲解和展开。
